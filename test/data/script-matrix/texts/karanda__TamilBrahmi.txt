𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀸𑀥𑀺𑀱𑁆𑀝𑀸𑀦𑀳𑁆𑀭𑀼𑀢𑀬 𑀓𑀼𑀳𑁆𑀬𑀥𑀸𑀢𑀼 𑀓𑀭𑀡𑁆𑀝𑀫𑀼𑀢𑁆𑀭𑀸 𑀥𑀸𑀭𑀡𑀻
𑀦𑀫𑀲𑁆𑀢𑁆𑀭𑁆𑀬𑀥𑁆𑀯𑀺𑀓𑀸𑀦𑀸𑀫𑁆 𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀸𑀦𑀸𑀫𑁆 .
𑀑𑀫𑁆 𑀧𑀼𑀯𑀺𑀧𑀯𑀦𑀯𑀭𑁂 𑀯𑀘𑀦𑀯𑀘𑀢𑀺 .
𑀲𑀼𑀭𑀼 𑀲𑀼𑀭𑀼 𑀥𑀭 𑀥𑀭 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀥𑀸𑀢𑀼 𑀥𑀭𑁂 𑀧𑀢𑁆𑀫𑀫𑁆 𑀧𑀯𑀢𑀺 .
𑀚𑀬𑀯𑀭𑁂 𑀫𑀼𑀢𑁆𑀭𑁂 .
𑀲𑁆𑀫𑀭𑀢𑀢𑀸𑀓𑀢𑀥𑀭𑁆𑀫𑀘𑀓𑁆𑀭𑀧𑁆𑀭𑀯𑀭𑁆𑀢𑀦 𑀯𑀚𑁆𑀭𑁂 𑀧𑁄𑀥𑀺𑀫𑀡𑁆𑀝𑀸𑀮𑀗𑁆𑀓𑀸𑀭𑀸𑀮𑀗𑁆𑀓𑁆𑀭𑀼𑀢𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀸𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀧𑁄𑀥𑀬 𑀧𑁄𑀥𑀬 𑀧𑁄𑀥𑀺 𑀧𑁄𑀥𑀺 𑀧𑀼𑀥𑁆𑀬 𑀧𑀼𑀥𑁆𑀬 .
𑀲𑀫𑁆𑀧𑁄𑀥𑀦𑀺 𑀲𑀫𑁆𑀧𑁄𑀥𑀬 .
𑀘𑀮 𑀘𑀮 𑀘𑀮𑀦𑁆𑀢𑀼 𑀲𑀭𑁆𑀯𑀸𑀯𑀭𑀡𑀸𑀦𑀺 .
𑀲𑀭𑁆𑀯𑀧𑀸𑀧𑀯𑀺𑀓𑀢𑁂 .
𑀳𑀼𑀭𑀼 𑀳𑀼𑀭𑀼 𑀲𑀭𑁆𑀯𑀰𑁄𑀓𑀯𑀺𑀓𑀢𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀳𑁆𑀭𑀼𑀢𑀬𑀯𑀚𑁆𑀭𑀺𑀡𑀺 .
𑀲𑀫𑁆𑀧𑀸𑀭 𑀲𑀫𑁆𑀧𑀸𑀭 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀓𑀼𑀳𑁆𑀬𑀥𑀸𑀭𑀡𑀻 𑀫𑀼𑀢𑁆𑀭𑁂 .
𑀧𑀽𑀢𑁂 𑀲𑀼𑀧𑀽𑀢𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀸𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑀥𑀸𑀢𑀼 𑀓𑀭𑁆𑀧𑁂 𑀲𑁆𑀯𑀸𑀳𑀸 .
𑀲𑀫𑀬𑀸𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 𑀲𑁆𑀯𑀸𑀳𑀸 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀳𑁆𑀭𑀼𑀢𑀬𑀥𑀸𑀢𑀼 𑀫𑀼𑀢𑁆𑀭𑁂 𑀲𑁆𑀯𑀸𑀳𑀸 .
𑀲𑀼𑀧𑁆𑀭𑀢𑀺𑀱𑁆𑀝𑀺𑀢𑀲𑁆𑀢𑀽𑀧𑁂 𑀢𑀢𑀸𑀓𑀢𑀸𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 𑀳𑀼𑀭𑀼 𑀳𑀼𑀭𑀼 𑀳𑀽𑀫𑁆 𑀳𑀽𑀫𑁆 𑀲𑁆𑀯𑀸𑀳𑀸 .
𑀑𑀫𑁆 𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑁄𑀱𑁆𑀡𑀻𑀱𑀥𑀸𑀢𑀼 𑀫𑀼𑀢𑁆𑀭𑀸𑀡𑀺 𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀲𑀥𑀸𑀢𑀼 𑀯𑀺𑀧𑀽𑀱𑀺𑀢𑀸𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 𑀳𑀽𑀫𑁆 𑀳𑀽𑀫𑁆 𑀲𑁆𑀯𑀸𑀳𑀸 ..
