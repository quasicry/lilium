𑀅𑀢 𑀓𑀮𑁆𑀯𑀓𑁆𑀱𑀬𑀫𑀢𑀺𑀭𑁆𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀧𑀓𑀯𑀦𑁆𑀢𑀫𑁂𑀢𑀢𑀯𑁄𑀘𑀢𑁆-𑀓𑀢𑀫𑁆 𑀧𑀓𑀯𑀦𑁆 𑀅𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑁄'𑀲𑁆𑀬𑀸𑀫𑁆 𑀲𑀳𑀸𑀬𑀸𑀫𑁆 𑀮𑁄𑀓𑀥𑀸𑀢𑁅 𑀧𑁆𑀭𑀯𑀺𑀘𑀭𑀢𑀺? 𑀓𑀢𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺? 𑀓𑀻𑀢𑁆𑀭𑀼𑀰𑀰𑁆𑀘𑀸𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑀲𑁆𑀬 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬𑁄𑀧𑀸𑀬𑀓𑁅𑀰𑀮𑁆𑀬𑀯𑀺𑀱𑀬𑀂? 𑀏𑀯𑀫𑀼𑀓𑁆𑀢𑁂 𑀧𑀓𑀯𑀸𑀦𑀓𑁆𑀱𑀬𑀫𑀢𑀺𑀫𑁆 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑀫𑁆 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀫𑁂𑀢𑀢𑀯𑁄𑀘𑀢𑁆-𑀲𑀦𑁆𑀢𑀺 𑀓𑀼𑀮𑀧𑀼𑀢𑁆𑀭 𑀮𑁄𑀓𑀥𑀸𑀢𑀯𑀂 𑀬𑁂𑀱𑁆𑀯𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀧𑀼𑀢𑁆𑀥𑀭𑀽𑀧𑁂𑀡 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀲𑀦𑁆𑀢𑀺 𑀮𑁄𑀓𑀥𑀸𑀢𑀯𑀂, 𑀬𑁂𑀱𑁆𑀯𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑀭𑀽𑀧𑁂𑀡 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀢𑁆 𑀧𑁆𑀭𑀢𑁆𑀬𑁂𑀓𑀧𑀼𑀢𑁆𑀥𑀭𑀽𑀧𑁂𑀡 𑀅𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀘𑁆𑀘𑁆𑀭𑀸𑀯𑀓𑀭𑀽𑀧𑁂𑀡 𑀅𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀢𑁆 𑀧𑁆𑀭𑀳𑁆𑀫𑀭𑀽𑀧𑁂𑀡𑀸𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀘𑁆𑀘𑀓𑁆𑀭𑀭𑀽𑀧𑁂𑀡𑀸𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺.. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀢𑁆 𑀓𑀦𑁆𑀥𑀭𑁆𑀯𑀭𑀽𑀧𑁂𑀡𑀸𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀓𑁂𑀱𑀸𑀫𑁆𑀘𑀺𑀢𑁆 𑀓𑀦𑁆𑀥𑀭𑁆𑀯𑀭𑀽𑀧𑁂𑀡𑀸𑀯𑀮𑁄𑀓𑀺𑀢𑁂𑀰𑁆𑀯𑀭𑁄 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀬𑀓𑁆𑀱𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀬𑀓𑁆𑀱𑀭𑀽𑀧𑁂𑀡 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀈𑀰𑁆𑀯𑀭𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑀻𑀰𑁆𑀯𑀭𑀭𑀽𑀧𑁂𑀡, 𑀫𑀳𑁂𑀰𑁆𑀯𑀭𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀫𑀳𑁂𑀰𑁆𑀯𑀭𑀭𑀽𑀧𑁂𑀡 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀘𑀓𑁆𑀭𑀯𑀭𑁆𑀢𑀺𑀭𑀸𑀚𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀘𑀓𑁆𑀭𑀯𑀭𑁆𑀢𑀺𑀭𑀸𑀚𑀭𑀽𑀧𑁂𑀡 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀧𑀺𑀰𑀸𑀘𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀧𑀺𑀰𑀸𑀘𑀭𑀽𑀧𑁂𑀡 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺. 𑀯𑁃𑀰𑁆𑀭𑀯𑀡𑀯𑁃𑀦𑁂𑀬𑀸𑀦𑀸𑀫𑁆 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀯𑁃𑀰𑁆𑀭𑀯𑀡𑀭𑀽𑀧𑁂𑀡 𑀥𑀭𑁆𑀫𑀫𑁆 𑀢𑁂𑀰𑀬𑀢𑀺.
