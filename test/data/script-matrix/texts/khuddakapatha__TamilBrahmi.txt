.. 𑀦𑀫𑁄 𑀢𑀲𑁆𑀲 𑀧𑀓𑀯𑀢𑁄 𑀅𑀭𑀳𑀢𑁄 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑀲𑁆𑀲 ..
𑀓𑀼𑀢𑁆𑀢𑀓𑀦𑀺𑀓𑀸𑀬𑁂
𑀓𑀼𑀢𑁆𑀢𑀓𑀧𑀸𑀝𑀧𑀸𑀴𑀺
1. 𑀲𑀭𑀡𑀢𑁆𑀢𑀬𑀫𑁆
𑀧𑀼𑀢𑁆𑀥𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀥𑀫𑁆𑀫𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀲𑀗𑁆𑀓𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺..
𑀢𑀼𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀧𑀼𑀢𑁆𑀥𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀢𑀼𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀥𑀫𑁆𑀫𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀢𑀼𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺..
𑀢𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀧𑀼𑀢𑁆𑀥𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀢𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀥𑀫𑁆𑀫𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺.
𑀢𑀢𑀺𑀬𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑀫𑁆 𑀲𑀭𑀡𑀫𑁆 𑀓𑀘𑁆𑀘𑀸𑀫𑀺..
𑀲𑀭𑀡𑀢𑁆𑀢𑀬𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
2. 𑀢𑀲𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆
1. 𑀧𑀸𑀡𑀸𑀢𑀺𑀧𑀸𑀢𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
2. 𑀅𑀢𑀺𑀦𑁆𑀦𑀸𑀢𑀸𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
3. 𑀅𑀧𑁆𑀭𑀳𑁆𑀫𑀘𑀭𑀺𑀬𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
4. 𑀫𑀼𑀲𑀸𑀯𑀸𑀢𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
5. 𑀲𑀼𑀭𑀸𑀫𑁂𑀭𑀬𑀫𑀚𑁆𑀚𑀧𑀫𑀸𑀢𑀝𑁆𑀝𑀸𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
6. 𑀯𑀺𑀓𑀸𑀮𑀧𑁄𑀚𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
7. 𑀦𑀘𑁆𑀘-𑀓𑀻𑀢-𑀯𑀸𑀢𑀺𑀢-𑀯𑀺𑀲𑀽𑀓𑀢𑀲𑁆𑀲𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
8. 𑀫𑀸𑀮𑀸-𑀓𑀦𑁆𑀥-𑀯𑀺𑀮𑁂𑀧𑀦-𑀥𑀸𑀭𑀡-𑀫𑀡𑁆𑀝𑀦-𑀯𑀺𑀧𑀽𑀲𑀦𑀝𑁆𑀝𑀸𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
9. 𑀉𑀘𑁆𑀘𑀸𑀲𑀬𑀦-𑀫𑀳𑀸𑀲𑀬𑀦𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
10. 𑀚𑀸𑀢𑀭𑀽𑀧-𑀭𑀚𑀢𑀧𑀝𑀺𑀓𑁆𑀓𑀳𑀡𑀸 𑀯𑁂𑀭𑀫𑀡𑀻-𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀲𑀫𑀸𑀢𑀺𑀬𑀸𑀫𑀺.
𑀢𑀲𑀲𑀺𑀓𑁆𑀓𑀸𑀧𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
3. 𑀢𑁆𑀯𑀢𑁆𑀢𑀺𑀫𑁆𑀲𑀸𑀓𑀸𑀭𑁄
𑀅𑀢𑁆𑀢𑀺 𑀇𑀫𑀲𑁆𑀫𑀺𑀫𑁆 𑀓𑀸𑀬𑁂 –
𑀓𑁂𑀲𑀸 𑀮𑁄𑀫𑀸 𑀦𑀓𑀸 𑀢𑀦𑁆𑀢𑀸 𑀢𑀘𑁄,
𑀫𑀫𑁆𑀲𑀫𑁆 𑀦𑁆𑀳𑀸𑀭𑀼 𑀅𑀝𑁆𑀝𑀺 𑀅𑀝𑁆𑀝𑀺𑀫𑀺𑀜𑁆𑀚𑀫𑁆 𑀯𑀓𑁆𑀓𑀫𑁆,
𑀳𑀢𑀬𑀫𑁆 𑀬𑀓𑀦𑀫𑁆 𑀓𑀺𑀮𑁄𑀫𑀓𑀫𑁆 𑀧𑀺𑀳𑀓𑀫𑁆 𑀧𑀧𑁆𑀧𑀸𑀲𑀫𑁆,
𑀅𑀦𑁆𑀢𑀫𑁆 𑀅𑀦𑁆𑀢𑀓𑀼𑀡𑀫𑁆 𑀉𑀢𑀭𑀺𑀬𑀫𑁆 𑀓𑀭𑀻𑀲𑀫𑁆 𑀫𑀢𑁆𑀢𑀮𑀼𑀗𑁆𑀓𑀫𑁆,
𑀧𑀺𑀢𑁆𑀢𑀫𑁆 𑀲𑁂𑀫𑁆𑀳𑀫𑁆 𑀧𑀼𑀧𑁆𑀧𑁄 𑀮𑁄𑀳𑀺𑀢𑀫𑁆 𑀲𑁂𑀢𑁄 𑀫𑁂𑀢𑁄,
𑀅𑀲𑁆𑀲𑀼 𑀯𑀲𑀸 𑀓𑁂𑀴𑁄 𑀲𑀺𑀗𑁆𑀓𑀸𑀡𑀺𑀓𑀸 𑀮𑀲𑀺𑀓𑀸 𑀫𑀼𑀢𑁆𑀢𑀦𑁆𑀢𑀺 ..
𑀢𑁆𑀯𑀢𑁆𑀢𑀺𑀫𑁆𑀲𑀸𑀓𑀸𑀭𑁄 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑁄.
4. 𑀓𑀼𑀫𑀸𑀭𑀧𑀜𑁆𑀳𑀸
1. ‘‘𑀏𑀓𑀫𑁆 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀲𑀧𑁆𑀧𑁂 𑀲𑀢𑁆𑀢𑀸 𑀆𑀳𑀸𑀭𑀝𑁆𑀝𑀺𑀢𑀺𑀓𑀸’’.
2. ‘‘𑀢𑁆𑀯𑁂 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀦𑀸𑀫𑀜𑁆𑀘 𑀭𑀽𑀧𑀜𑁆𑀘’’.
3. ‘‘𑀢𑀻𑀡𑀺 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀢𑀺𑀲𑁆𑀲𑁄 𑀯𑁂𑀢𑀦𑀸’’.
4. ‘‘𑀘𑀢𑁆𑀢𑀸𑀭𑀺 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀘𑀢𑁆𑀢𑀸𑀭𑀺 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀸𑀦𑀺’’.
5. ‘‘𑀧𑀜𑁆𑀘 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀧𑀜𑁆𑀘𑀼𑀧𑀸𑀢𑀸𑀦𑀓𑁆𑀓𑀦𑁆𑀥𑀸’’.
6. ‘‘𑀘 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀘 𑀅𑀚𑁆𑀚𑀢𑁆𑀢𑀺𑀓𑀸𑀦𑀺 𑀆𑀬𑀢𑀦𑀸𑀦𑀺’’.
7. ‘‘𑀲𑀢𑁆𑀢 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀲𑀢𑁆𑀢 𑀧𑁄𑀚𑁆𑀚𑀗𑁆𑀓𑀸’’.
8. ‘‘𑀅𑀝𑁆𑀝 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀅𑀭𑀺𑀬𑁄 𑀅𑀝𑁆𑀝𑀗𑁆𑀓𑀺𑀓𑁄 𑀫𑀓𑁆𑀓𑁄’’.
9. ‘‘𑀦𑀯 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀦𑀯 𑀲𑀢𑁆𑀢𑀸𑀯𑀸𑀲𑀸’’.
10. ‘‘𑀢𑀲 𑀦𑀸𑀫 𑀓𑀺𑀫𑁆’’? ‘‘𑀢𑀲𑀳𑀗𑁆𑀓𑁂𑀳𑀺 𑀲𑀫𑀦𑁆𑀦𑀸𑀓𑀢𑁄 ‘𑀅𑀭𑀳𑀸’𑀢𑀺 𑀯𑀼𑀘𑁆𑀘𑀢𑀻’’𑀢𑀺.
𑀓𑀼𑀫𑀸𑀭𑀧𑀜𑁆𑀳𑀸 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀸.
5. 𑀫𑀗𑁆𑀓𑀮𑀲𑀼𑀢𑁆𑀢𑀫𑁆
1. 𑀏𑀯𑀫𑁆 𑀫𑁂 𑀲𑀼𑀢𑀫𑁆 – 𑀏𑀓𑀫𑁆 𑀲𑀫𑀬𑀫𑁆 𑀧𑀓𑀯𑀸 𑀲𑀸𑀯𑀢𑁆𑀢𑀺𑀬𑀫𑁆 𑀯𑀺𑀳𑀭𑀢𑀺 𑀚𑁂𑀢𑀯𑀦𑁂 𑀅𑀦𑀸𑀢𑀧𑀺𑀡𑁆𑀝𑀺𑀓𑀲𑁆𑀲 𑀆𑀭𑀸𑀫𑁂. 𑀅𑀢 𑀓𑁄 𑀅𑀜𑁆𑀜𑀢𑀭𑀸 𑀢𑁂𑀯𑀢𑀸 𑀅𑀧𑀺𑀓𑁆𑀓𑀦𑁆𑀢𑀸𑀬 𑀭𑀢𑁆𑀢𑀺𑀬𑀸 𑀅𑀧𑀺𑀓𑁆𑀓𑀦𑁆𑀢𑀯𑀡𑁆𑀡𑀸 𑀓𑁂𑀯𑀮𑀓𑀧𑁆𑀧𑀫𑁆 𑀚𑁂𑀢𑀯𑀦𑀫𑁆 𑀑𑀧𑀸𑀲𑁂𑀢𑁆𑀯𑀸 𑀬𑁂𑀦 𑀧𑀓𑀯𑀸 𑀢𑁂𑀦𑀼𑀧𑀲𑀗𑁆𑀓𑀫𑀺; 𑀉𑀧𑀲𑀗𑁆𑀓𑀫𑀺𑀢𑁆𑀯𑀸 𑀧𑀓𑀯𑀦𑁆𑀢𑀫𑁆 𑀅𑀧𑀺𑀯𑀸𑀢𑁂𑀢𑁆𑀯𑀸 𑀏𑀓𑀫𑀦𑁆𑀢𑀫𑁆 𑀅𑀝𑁆𑀝𑀸𑀲𑀺. 𑀏𑀓𑀫𑀦𑁆𑀢𑀫𑁆 𑀝𑀺𑀢𑀸 𑀓𑁄 𑀲𑀸 𑀢𑁂𑀯𑀢𑀸 𑀧𑀓𑀯𑀦𑁆𑀢𑀫𑁆 𑀓𑀸𑀢𑀸𑀬 𑀅𑀚𑁆𑀚𑀧𑀸𑀲𑀺 –
2. ‘‘𑀧𑀳𑀽 𑀢𑁂𑀯𑀸 𑀫𑀦𑀼𑀲𑁆𑀲𑀸 𑀘, 𑀫𑀗𑁆𑀓𑀮𑀸𑀦𑀺 𑀅𑀘𑀺𑀦𑁆𑀢𑀬𑀼𑀫𑁆.
𑀆𑀓𑀗𑁆𑀓𑀫𑀸𑀦𑀸 𑀲𑁄𑀢𑁆𑀢𑀸𑀦𑀫𑁆, 𑀧𑁆𑀭𑀽𑀳𑀺 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆’’..
3. ‘‘𑀅𑀲𑁂𑀯𑀦𑀸 𑀘 𑀧𑀸𑀮𑀸𑀦𑀫𑁆, 𑀧𑀡𑁆𑀝𑀺𑀢𑀸𑀦𑀜𑁆𑀘 𑀲𑁂𑀯𑀦𑀸.
𑀧𑀽𑀚𑀸 𑀘 𑀧𑀽𑀚𑀦𑁂𑀬𑁆𑀬𑀸𑀦𑀫𑁆 , 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
4. ‘‘𑀧𑀢𑀺𑀭𑀽𑀧𑀢𑁂𑀲𑀯𑀸𑀲𑁄 𑀘, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀘 𑀓𑀢𑀧𑀼𑀜𑁆𑀜𑀢𑀸.
𑀅𑀢𑁆𑀢𑀲𑀫𑁆𑀫𑀸𑀧𑀡𑀺𑀥𑀺 𑀘, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
5. ‘‘𑀧𑀸𑀳𑀼𑀲𑀘𑁆𑀘𑀜𑁆𑀘 𑀲𑀺𑀧𑁆𑀧𑀜𑁆𑀘, 𑀯𑀺𑀦𑀬𑁄 𑀘 𑀲𑀼𑀲𑀺𑀓𑁆𑀓𑀺𑀢𑁄.
𑀲𑀼𑀧𑀸𑀲𑀺𑀢𑀸 𑀘 𑀬𑀸 𑀯𑀸𑀘𑀸, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
6. ‘‘𑀫𑀸𑀢𑀸𑀧𑀺𑀢𑀼 𑀉𑀧𑀝𑁆𑀝𑀸𑀦𑀫𑁆, 𑀧𑀼𑀢𑁆𑀢𑀢𑀸𑀭𑀲𑁆𑀲 𑀲𑀗𑁆𑀓𑀳𑁄.
𑀅𑀦𑀸𑀓𑀼𑀮𑀸 𑀘 𑀓𑀫𑁆𑀫𑀦𑁆𑀢𑀸, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
7. ‘‘𑀢𑀸𑀦𑀜𑁆𑀘 𑀥𑀫𑁆𑀫𑀘𑀭𑀺𑀬𑀸 𑀘, 𑀜𑀸𑀢𑀓𑀸𑀦𑀜𑁆𑀘 𑀲𑀗𑁆𑀓𑀳𑁄.
𑀅𑀦𑀯𑀚𑁆𑀚𑀸𑀦𑀺 𑀓𑀫𑁆𑀫𑀸𑀦𑀺, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
8. ‘‘𑀆𑀭𑀢𑀻 𑀯𑀺𑀭𑀢𑀻 𑀧𑀸𑀧𑀸, 𑀫𑀚𑁆𑀚𑀧𑀸𑀦𑀸 𑀘 𑀲𑀫𑁆𑀬𑀫𑁄.
𑀅𑀧𑁆𑀧𑀫𑀸𑀢𑁄 𑀘 𑀥𑀫𑁆𑀫𑁂𑀲𑀼, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
9. ‘‘𑀓𑀸𑀭𑀯𑁄 𑀘 𑀦𑀺𑀯𑀸𑀢𑁄 𑀘, 𑀲𑀦𑁆𑀢𑀼𑀝𑁆𑀝𑀺 𑀘 𑀓𑀢𑀜𑁆𑀜𑀼𑀢𑀸.
𑀓𑀸𑀮𑁂𑀦 𑀥𑀫𑁆𑀫𑀲𑁆𑀲𑀯𑀦𑀫𑁆, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
10. ‘‘𑀓𑀦𑁆𑀢𑀻 𑀘 𑀲𑁄𑀯𑀘𑀲𑁆𑀲𑀢𑀸, 𑀲𑀫𑀡𑀸𑀦𑀜𑁆𑀘 𑀢𑀲𑁆𑀲𑀦𑀫𑁆.
𑀓𑀸𑀮𑁂𑀦 𑀥𑀫𑁆𑀫𑀲𑀸𑀓𑀘𑁆𑀘𑀸, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
11. ‘‘𑀢𑀧𑁄 𑀘 𑀧𑁆𑀭𑀳𑁆𑀫𑀘𑀭𑀺𑀬𑀜𑁆𑀘, 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀸𑀦 𑀢𑀲𑁆𑀲𑀦𑀫𑁆.
𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀲𑀘𑁆𑀘𑀺𑀓𑀺𑀭𑀺𑀬𑀸 𑀘, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
12. ‘‘𑀧𑀼𑀝𑁆𑀝𑀲𑁆𑀲 𑀮𑁄𑀓𑀥𑀫𑁆𑀫𑁂𑀳𑀺, 𑀘𑀺𑀢𑁆𑀢𑀫𑁆 𑀬𑀲𑁆𑀲 𑀦 𑀓𑀫𑁆𑀧𑀢𑀺.
𑀅𑀲𑁄𑀓𑀫𑁆 𑀯𑀺𑀭𑀚𑀫𑁆 𑀓𑁂𑀫𑀫𑁆, 𑀏𑀢𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫𑀫𑁆..
13. ‘‘𑀏𑀢𑀸𑀢𑀺𑀲𑀸𑀦𑀺 𑀓𑀢𑁆𑀯𑀸𑀦, 𑀲𑀧𑁆𑀧𑀢𑁆𑀢𑀫𑀧𑀭𑀸𑀚𑀺𑀢𑀸.
𑀲𑀧𑁆𑀧𑀢𑁆𑀢 𑀲𑁄𑀢𑁆𑀢𑀺𑀫𑁆 𑀓𑀘𑁆𑀘𑀦𑁆𑀢𑀺, 𑀢𑀫𑁆 𑀢𑁂𑀲𑀫𑁆 𑀫𑀗𑁆𑀓𑀮𑀫𑀼𑀢𑁆𑀢𑀫’’𑀦𑁆𑀢𑀺..
𑀫𑀗𑁆𑀓𑀮𑀲𑀼𑀢𑁆𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
6. 𑀭𑀢𑀦𑀲𑀼𑀢𑁆𑀢𑀫𑁆
1. 𑀬𑀸𑀦𑀻𑀥 𑀧𑀽𑀢𑀸𑀦𑀺 𑀲𑀫𑀸𑀓𑀢𑀸𑀦𑀺, 𑀧𑀼𑀫𑁆𑀫𑀸𑀦𑀺 𑀯𑀸 𑀬𑀸𑀦𑀺 𑀯 𑀅𑀦𑁆𑀢𑀮𑀺𑀓𑁆𑀓𑁂.
𑀲𑀧𑁆𑀧𑁂𑀯 𑀧𑀽𑀢𑀸 𑀲𑀼𑀫𑀦𑀸 𑀧𑀯𑀦𑁆𑀢𑀼, 𑀅𑀢𑁄𑀧𑀺 𑀲𑀓𑁆𑀓𑀘𑁆𑀘 𑀲𑀼𑀡𑀦𑁆𑀢𑀼 𑀧𑀸𑀲𑀺𑀢𑀫𑁆..
2. 𑀢𑀲𑁆𑀫𑀸 𑀳𑀺 𑀧𑀽𑀢𑀸 𑀦𑀺𑀲𑀸𑀫𑁂𑀢 𑀲𑀧𑁆𑀧𑁂, 𑀫𑁂𑀢𑁆𑀢𑀫𑁆 𑀓𑀭𑁄𑀢 𑀫𑀸𑀦𑀼𑀲𑀺𑀬𑀸 𑀧𑀚𑀸𑀬.
𑀢𑀺𑀯𑀸 𑀘 𑀭𑀢𑁆𑀢𑁄 𑀘 𑀳𑀭𑀦𑁆𑀢𑀺 𑀬𑁂 𑀧𑀮𑀺𑀫𑁆, 𑀢𑀲𑁆𑀫𑀸 𑀳𑀺 𑀦𑁂 𑀭𑀓𑁆𑀓𑀢 𑀅𑀧𑁆𑀧𑀫𑀢𑁆𑀢𑀸..
3. 𑀬𑀫𑁆 𑀓𑀺𑀜𑁆𑀘𑀺 𑀯𑀺𑀢𑁆𑀢𑀫𑁆 𑀇𑀥 𑀯𑀸 𑀳𑀼𑀭𑀫𑁆 𑀯𑀸, 𑀲𑀓𑁆𑀓𑁂𑀲𑀼 𑀯𑀸 𑀬𑀫𑁆 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆.
𑀦 𑀦𑁄 𑀲𑀫𑀫𑁆 𑀅𑀢𑁆𑀢𑀺 𑀢𑀢𑀸𑀓𑀢𑁂𑀦, 𑀇𑀢𑀫𑁆𑀧𑀺 𑀧𑀼𑀢𑁆𑀥𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆.
𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
4. 𑀓𑀬𑀫𑁆 𑀯𑀺𑀭𑀸𑀓𑀫𑁆 𑀅𑀫𑀢𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀬𑀢𑀚𑁆𑀚𑀓𑀸 𑀲𑀓𑁆𑀬𑀫𑀼𑀦𑀻 𑀲𑀫𑀸𑀳𑀺𑀢𑁄.
𑀦 𑀢𑁂𑀦 𑀥𑀫𑁆𑀫𑁂𑀦 𑀲𑀫𑀢𑁆𑀢𑀺 𑀓𑀺𑀜𑁆𑀘𑀺, 𑀇𑀢𑀫𑁆𑀧𑀺 𑀥𑀫𑁆𑀫𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆.
𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
5. 𑀬𑀫𑁆 𑀧𑀼𑀢𑁆𑀥𑀲𑁂𑀝𑁆𑀝𑁄 𑀧𑀭𑀺𑀯𑀡𑁆𑀡𑀬𑀻 𑀲𑀼𑀘𑀺𑀫𑁆, 𑀲𑀫𑀸𑀥𑀺𑀫𑀸𑀦𑀦𑁆𑀢𑀭𑀺𑀓𑀜𑁆𑀜𑀫𑀸𑀳𑀼.
𑀲𑀫𑀸𑀥𑀺𑀦𑀸 𑀢𑁂𑀦 𑀲𑀫𑁄 𑀦 𑀯𑀺𑀚𑁆𑀚𑀢𑀺, 𑀇𑀢𑀫𑁆𑀧𑀺 𑀥𑀫𑁆𑀫𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆.
𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
6. 𑀬𑁂 𑀧𑀼𑀓𑁆𑀓𑀮𑀸 𑀅𑀝𑁆𑀝 𑀲𑀢𑀫𑁆 𑀧𑀲𑀢𑁆𑀢𑀸, 𑀘𑀢𑁆𑀢𑀸𑀭𑀺 𑀏𑀢𑀸𑀦𑀺 𑀬𑀼𑀓𑀸𑀦𑀺 𑀳𑁄𑀦𑁆𑀢𑀺.
𑀢𑁂 𑀢𑀓𑁆𑀓𑀺𑀡𑁂𑀬𑁆𑀬𑀸 𑀲𑀼𑀓𑀢𑀲𑁆𑀲 𑀲𑀸𑀯𑀓𑀸, 𑀏𑀢𑁂𑀲𑀼 𑀢𑀺𑀦𑁆𑀦𑀸𑀦𑀺 𑀫𑀳𑀧𑁆𑀧𑀮𑀸𑀦𑀺.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
7. 𑀬𑁂 𑀲𑀼𑀧𑁆𑀧𑀬𑀼𑀢𑁆𑀢𑀸 𑀫𑀦𑀲𑀸 𑀢𑀴𑁆𑀳𑁂𑀦, 𑀦𑀺𑀓𑁆𑀓𑀸𑀫𑀺𑀦𑁄 𑀓𑁄𑀢𑀫𑀲𑀸𑀲𑀦𑀫𑁆𑀳𑀺.
𑀢𑁂 𑀧𑀢𑁆𑀢𑀺𑀧𑀢𑁆𑀢𑀸 𑀅𑀫𑀢𑀫𑁆 𑀯𑀺𑀓𑀬𑁆𑀳, 𑀮𑀢𑁆𑀥𑀸 𑀫𑀼𑀥𑀸 𑀦𑀺𑀧𑁆𑀧𑀼𑀢𑀺𑀫𑁆 𑀧𑀼𑀜𑁆𑀚𑀫𑀸𑀦𑀸.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
8. 𑀬𑀢𑀺𑀦𑁆𑀢𑀓𑀻𑀮𑁄 𑀧𑀢𑀯𑀺𑀲𑁆𑀲𑀺𑀢𑁄 𑀲𑀺𑀬𑀸, 𑀘𑀢𑀼𑀧𑁆𑀧𑀺 𑀯𑀸𑀢𑁂𑀳𑀺 𑀅𑀲𑀫𑁆𑀧𑀓𑀫𑁆𑀧𑀺𑀬𑁄.
𑀢𑀢𑀽𑀧𑀫𑀫𑁆 𑀲𑀧𑁆𑀧𑀼𑀭𑀺𑀲𑀫𑁆 𑀯𑀢𑀸𑀫𑀺, 𑀬𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀸𑀦𑀺 𑀅𑀯𑁂𑀘𑁆𑀘 𑀧𑀲𑁆𑀲𑀢𑀺.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
9. 𑀬𑁂 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀸𑀦𑀺 𑀯𑀺𑀧𑀸𑀯𑀬𑀦𑁆𑀢𑀺, 𑀓𑀫𑁆𑀧𑀻𑀭𑀧𑀜𑁆𑀜𑁂𑀦 𑀲𑀼𑀢𑁂𑀲𑀺𑀢𑀸𑀦𑀺.
𑀓𑀺𑀜𑁆𑀘𑀸𑀧𑀺 𑀢𑁂 𑀳𑁄𑀦𑁆𑀢𑀺 𑀧𑀼𑀲𑀫𑁆 𑀧𑀫𑀢𑁆𑀢𑀸, 𑀦 𑀢𑁂 𑀧𑀯𑀫𑁆 𑀅𑀝𑁆𑀝𑀫𑀫𑀸𑀢𑀺𑀬𑀦𑁆𑀢𑀺.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
10. 𑀲𑀳𑀸𑀯𑀲𑁆𑀲 𑀢𑀲𑁆𑀲𑀦𑀲𑀫𑁆𑀧𑀢𑀸𑀬 , 𑀢𑀬𑀲𑁆𑀲𑀼 𑀥𑀫𑁆𑀫𑀸 𑀚𑀳𑀺𑀢𑀸 𑀧𑀯𑀦𑁆𑀢𑀺.
𑀲𑀓𑁆𑀓𑀸𑀬𑀢𑀺𑀝𑁆𑀝𑀻 𑀯𑀺𑀘𑀺𑀓𑀺𑀘𑁆𑀘𑀺𑀢𑀜𑁆𑀘, 𑀲𑀻𑀮𑀧𑁆𑀧𑀢𑀫𑁆 𑀯𑀸𑀧𑀺 𑀬𑀢𑀢𑁆𑀢𑀺 𑀓𑀺𑀜𑁆𑀘𑀺..
11. 𑀘𑀢𑀽𑀳𑀧𑀸𑀬𑁂𑀳𑀺 𑀘 𑀯𑀺𑀧𑁆𑀧𑀫𑀼𑀢𑁆𑀢𑁄, 𑀘𑀘𑁆𑀘𑀸𑀧𑀺𑀝𑀸𑀦𑀸𑀦𑀺.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
12. 𑀓𑀺𑀜𑁆𑀘𑀸𑀧𑀺 𑀲𑁄 𑀓𑀫𑁆𑀫 𑀓𑀭𑁄𑀢𑀺 𑀧𑀸𑀧𑀓𑀫𑁆, 𑀓𑀸𑀬𑁂𑀦 𑀯𑀸𑀘𑀸 𑀉𑀢 𑀘𑁂𑀢𑀲𑀸 𑀯𑀸.
𑀅𑀧𑀧𑁆𑀧 , 𑀅𑀧𑀧𑁆𑀧𑀢𑀸 𑀢𑀺𑀝𑁆𑀝𑀧𑀢𑀲𑁆𑀲 𑀯𑀼𑀢𑁆𑀢𑀸.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
13.𑀯𑀦𑀧𑁆𑀧𑀓𑀼𑀫𑁆𑀧𑁂 𑀬𑀢 𑀓𑀺𑀫𑁆𑀳𑁂.
𑀢𑀢𑀽𑀧𑀫𑀫𑁆 𑀥𑀫𑁆𑀫𑀯𑀭𑀫𑁆 𑀅𑀢𑁂𑀲𑀬𑀺 , 𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀓𑀸𑀫𑀺𑀫𑁆 𑀧𑀭𑀫𑀫𑁆 𑀳𑀺𑀢𑀸𑀬.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀧𑀼𑀢𑁆𑀥𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
14. 𑀯𑀭𑁄 𑀯𑀭𑀜𑁆𑀜𑀽 𑀯𑀭𑀢𑁄 𑀯𑀭𑀸𑀳𑀭𑁄, 𑀅𑀦𑀼𑀢𑁆𑀢𑀭𑁄 𑀥𑀫𑁆𑀫𑀯𑀭𑀫𑁆 𑀅𑀢𑁂𑀲𑀬𑀺.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀧𑀼𑀢𑁆𑀥𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
15. 𑀓𑀻𑀡𑀫𑁆 𑀧𑀼𑀭𑀸𑀡𑀫𑁆 𑀦𑀯 𑀦𑀢𑁆𑀢𑀺 𑀲𑀫𑁆𑀧𑀯𑀫𑁆, 𑀯𑀺𑀭𑀢𑁆𑀢𑀘𑀺𑀢𑁆𑀢𑀸𑀬𑀢𑀺𑀓𑁂 𑀧𑀯𑀲𑁆𑀫𑀺𑀫𑁆.
𑀢𑁂 𑀓𑀻𑀡𑀧𑀻𑀚𑀸 𑀅𑀯𑀺𑀭𑀽𑀴𑁆𑀳𑀺𑀘𑀦𑁆𑀢𑀸, 𑀦𑀺𑀧𑁆𑀧𑀦𑁆𑀢𑀺 𑀥𑀻𑀭𑀸 𑀬𑀢𑀸𑀬𑀫𑁆 𑀧𑀢𑀻𑀧𑁄.
𑀇𑀢𑀫𑁆𑀧𑀺 𑀲𑀗𑁆𑀓𑁂 𑀭𑀢𑀦𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆, 𑀏𑀢𑁂𑀦 𑀲𑀘𑁆𑀘𑁂𑀦 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
16. 𑀬𑀸𑀦𑀻𑀥 𑀧𑀽𑀢𑀸𑀦𑀺 𑀲𑀫𑀸𑀓𑀢𑀸𑀦𑀺, 𑀧𑀼𑀫𑁆𑀫𑀸𑀦𑀺 𑀯𑀸 𑀬𑀸𑀦𑀺 𑀯 𑀅𑀦𑁆𑀢𑀮𑀺𑀓𑁆𑀓𑁂.
𑀢𑀢𑀸𑀓𑀢𑀫𑁆 𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀧𑀽𑀚𑀺𑀢𑀫𑁆, 𑀧𑀼𑀢𑁆𑀥𑀫𑁆 𑀦𑀫𑀲𑁆𑀲𑀸𑀫 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
17. 𑀬𑀸𑀦𑀻𑀥 𑀧𑀽𑀢𑀸𑀦𑀺 𑀲𑀫𑀸𑀓𑀢𑀸𑀦𑀺, 𑀧𑀼𑀫𑁆𑀫𑀸𑀦𑀺 𑀯𑀸 𑀬𑀸𑀦𑀺 𑀯 𑀅𑀦𑁆𑀢𑀮𑀺𑀓𑁆𑀓𑁂.
𑀢𑀢𑀸𑀓𑀢𑀫𑁆 𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀧𑀽𑀚𑀺𑀢𑀫𑁆, 𑀥𑀫𑁆𑀫𑀫𑁆 𑀦𑀫𑀲𑁆𑀲𑀸𑀫 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀼..
18. 𑀬𑀸𑀦𑀻𑀥 𑀧𑀽𑀢𑀸𑀦𑀺 𑀲𑀫𑀸𑀓𑀢𑀸𑀦𑀺, 𑀧𑀼𑀫𑁆𑀫𑀸𑀦𑀺 𑀯𑀸 𑀬𑀸𑀦𑀺 𑀯 𑀅𑀦𑁆𑀢𑀮𑀺𑀓𑁆𑀓𑁂.
𑀢𑀢𑀸𑀓𑀢𑀫𑁆 𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀧𑀽𑀚𑀺𑀢𑀫𑁆, 𑀲𑀗𑁆𑀓𑀫𑁆 𑀦𑀫𑀲𑁆𑀲𑀸𑀫 𑀲𑀼𑀯𑀢𑁆𑀢𑀺 𑀳𑁄𑀢𑀽𑀢𑀺..
𑀭𑀢𑀦𑀲𑀼𑀢𑁆𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
7. 𑀢𑀺𑀭𑁄𑀓𑀼𑀝𑁆𑀝𑀲𑀼𑀢𑁆𑀢𑀫𑁆
1. 𑀢𑀺𑀭𑁄𑀓𑀼𑀝𑁆𑀝𑁂𑀲𑀼 𑀢𑀺𑀝𑁆𑀝𑀦𑁆𑀢𑀺, 𑀲𑀦𑁆𑀥𑀺𑀲𑀺𑀗𑁆𑀓𑀸𑀝𑀓𑁂𑀲𑀼 𑀘.
𑀢𑁆𑀯𑀸𑀭𑀧𑀸𑀳𑀸𑀲𑀼 𑀢𑀺𑀝𑁆𑀝𑀦𑁆𑀢𑀺, 𑀆𑀓𑀦𑁆𑀢𑁆𑀯𑀸𑀦 𑀲𑀓𑀫𑁆 𑀓𑀭𑀫𑁆..
2. 𑀧𑀳𑀽𑀢𑁂 𑀅𑀦𑁆𑀦𑀧𑀸𑀦𑀫𑁆𑀳𑀺, 𑀓𑀚𑁆𑀚𑀧𑁄𑀚𑁆𑀚𑁂 𑀉𑀧𑀝𑁆𑀝𑀺𑀢𑁂.
𑀦 𑀢𑁂𑀲𑀫𑁆 𑀓𑁄𑀘𑀺 𑀲𑀭𑀢𑀺, 𑀲𑀢𑁆𑀢𑀸𑀦𑀫𑁆 𑀓𑀫𑁆𑀫𑀧𑀘𑁆𑀘𑀬𑀸..
3. 𑀏𑀯𑀫𑁆 𑀢𑀢𑀦𑁆𑀢𑀺 𑀜𑀸𑀢𑀻𑀦𑀫𑁆, 𑀬𑁂 𑀳𑁄𑀦𑁆𑀢𑀺 𑀅𑀦𑀼𑀓𑀫𑁆𑀧𑀓𑀸.
𑀲𑀼𑀘𑀺𑀫𑁆 𑀧𑀡𑀻𑀢𑀫𑁆 𑀓𑀸𑀮𑁂𑀦, 𑀓𑀧𑁆𑀧𑀺𑀬𑀫𑁆 𑀧𑀸𑀦𑀧𑁄𑀚𑀦𑀫𑁆.
𑀇𑀢𑀫𑁆 𑀯𑁄 𑀜𑀸𑀢𑀻𑀦𑀫𑁆 𑀳𑁄𑀢𑀼, 𑀲𑀼𑀓𑀺𑀢𑀸 𑀳𑁄𑀦𑁆𑀢𑀼 𑀜𑀸𑀢𑀬𑁄..
4. 𑀢𑁂 𑀘 𑀢𑀢𑁆𑀢 𑀲𑀫𑀸𑀓𑀦𑁆𑀢𑁆𑀯𑀸, 𑀜𑀸𑀢𑀺𑀧𑁂𑀢𑀸 𑀲𑀫𑀸𑀓𑀢𑀸.
𑀧𑀳𑀽𑀢𑁂 𑀅𑀦𑁆𑀦𑀧𑀸𑀦𑀫𑁆𑀳𑀺, 𑀲𑀓𑁆𑀓𑀘𑁆𑀘𑀫𑁆 𑀅𑀦𑀼𑀫𑁄𑀢𑀭𑁂..
5. 𑀘𑀺𑀭𑀫𑁆 𑀚𑀻𑀯𑀦𑁆𑀢𑀼 𑀦𑁄 𑀜𑀸𑀢𑀻, 𑀬𑁂𑀲𑀫𑁆 𑀳𑁂𑀢𑀼 𑀮𑀧𑀸𑀫𑀲𑁂.
𑀅𑀫𑁆𑀳𑀸𑀓𑀜𑁆𑀘 𑀓𑀢𑀸 𑀧𑀽𑀚𑀸, 𑀢𑀸𑀬𑀓𑀸 𑀘 𑀅𑀦𑀺𑀧𑁆𑀧𑀮𑀸..
6. 𑀦 𑀳𑀺 𑀢𑀢𑁆𑀢 𑀓𑀲𑀺 𑀅𑀢𑁆𑀢𑀺, 𑀓𑁄𑀭𑀓𑁆𑀓𑁂𑀢𑁆𑀢 𑀦 𑀯𑀺𑀚𑁆𑀚𑀢𑀺.
𑀯𑀡𑀺𑀚𑁆𑀚𑀸 𑀢𑀸𑀢𑀺𑀲𑀻 𑀦𑀢𑁆𑀢𑀺, 𑀳𑀺𑀭𑀜𑁆𑀜𑁂𑀦 𑀓𑀬𑁄𑀓𑀬𑀫𑁆.
𑀇𑀢𑁄 𑀢𑀺𑀦𑁆𑀦𑁂𑀦 𑀬𑀸𑀧𑁂𑀦𑁆𑀢𑀺, 𑀧𑁂𑀢𑀸 𑀓𑀸𑀮𑀗𑁆𑀓𑀢𑀸 𑀢𑀳𑀺𑀫𑁆..
7. 𑀉𑀦𑁆𑀦𑀫𑁂 𑀉𑀢𑀓𑀫𑁆 𑀯𑀼𑀝𑁆𑀝𑀫𑁆, 𑀬𑀢𑀸 𑀦𑀺𑀦𑁆𑀦𑀫𑁆 𑀧𑀯𑀢𑁆𑀢𑀢𑀺.
𑀏𑀯𑀫𑁂𑀯 𑀇𑀢𑁄 𑀢𑀺𑀦𑁆𑀦𑀫𑁆, 𑀧𑁂𑀢𑀸𑀦𑀫𑁆 𑀉𑀧𑀓𑀧𑁆𑀧𑀢𑀺..
8. 𑀬𑀢𑀸 𑀯𑀸𑀭𑀺𑀯𑀳𑀸 𑀧𑀽𑀭𑀸, 𑀧𑀭𑀺𑀧𑀽𑀭𑁂𑀦𑁆𑀢𑀺 𑀲𑀸𑀓𑀭𑀫𑁆.
𑀏𑀯𑀫𑁂𑀯 𑀇𑀢𑁄 𑀢𑀺𑀦𑁆𑀦𑀫𑁆, 𑀧𑁂𑀢𑀸𑀦𑀫𑁆 𑀉𑀧𑀓𑀧𑁆𑀧𑀢𑀺..
9. 𑀅𑀢𑀸𑀲𑀺 𑀫𑁂 𑀅𑀓𑀸𑀲𑀺 𑀫𑁂, 𑀜𑀸𑀢𑀺𑀫𑀺𑀢𑁆𑀢𑀸 𑀲𑀓𑀸 𑀘 𑀫𑁂.
𑀧𑁂𑀢𑀸𑀦𑀫𑁆 𑀢𑀓𑁆𑀓𑀺𑀡𑀫𑁆 𑀢𑀚𑁆𑀚𑀸, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀓𑀢𑀫𑀦𑀼𑀲𑁆𑀲𑀭𑀫𑁆..
10. 𑀦 𑀳𑀺 𑀭𑀼𑀡𑁆𑀡𑀫𑁆 𑀯𑀸 𑀲𑁄𑀓𑁄 𑀯𑀸, 𑀬𑀸 𑀘𑀜𑁆𑀜𑀸 𑀧𑀭𑀺𑀢𑁂𑀯𑀦𑀸.
𑀦 𑀢𑀫𑁆 𑀧𑁂𑀢𑀸𑀦𑀫𑀢𑁆𑀢𑀸𑀬, 𑀏𑀯𑀫𑁆 𑀢𑀺𑀝𑁆𑀝𑀦𑁆𑀢𑀺 𑀜𑀸𑀢𑀬𑁄..
11. 𑀅𑀬𑀜𑁆𑀘 𑀓𑁄 𑀢𑀓𑁆𑀓𑀺𑀡𑀸 𑀢𑀺𑀦𑁆𑀦𑀸, 𑀲𑀗𑁆𑀓𑀫𑁆𑀳𑀺 𑀲𑀼𑀧𑁆𑀧𑀢𑀺𑀝𑁆𑀝𑀺𑀢𑀸.
𑀢𑀻𑀓𑀭𑀢𑁆𑀢𑀫𑁆 𑀳𑀺𑀢𑀸𑀬𑀲𑁆𑀲, 𑀝𑀸𑀦𑀲𑁄 𑀉𑀧𑀓𑀧𑁆𑀧𑀢𑀺..
12. 𑀲𑁄 𑀜𑀸𑀢𑀺𑀥𑀫𑁆𑀫𑁄 𑀘 𑀅𑀬𑀫𑁆 𑀦𑀺𑀢𑀲𑁆𑀲𑀺𑀢𑁄, 𑀧𑁂𑀢𑀸𑀦 𑀧𑀽𑀚𑀸 𑀘 𑀓𑀢𑀸 𑀉𑀴𑀸𑀭𑀸.
𑀧𑀮𑀜𑁆𑀘 𑀧𑀺𑀓𑁆𑀓𑀽𑀦𑀫𑀦𑀼𑀧𑁆𑀧𑀢𑀺𑀦𑁆𑀦𑀫𑁆 , 𑀢𑀼𑀫𑁆𑀳𑁂𑀳𑀺 𑀧𑀼𑀜𑁆𑀜𑀫𑁆 𑀧𑀲𑀼𑀢𑀫𑁆 𑀅𑀦𑀧𑁆𑀧𑀓𑀦𑁆𑀢𑀺..
𑀢𑀺𑀭𑁄𑀓𑀼𑀝𑁆𑀝𑀲𑀼𑀢𑁆𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
8. 𑀦𑀺𑀥𑀺𑀓𑀡𑁆𑀝𑀲𑀼𑀢𑁆𑀢𑀫𑁆
1. 𑀦𑀺𑀥𑀺𑀫𑁆 𑀦𑀺𑀥𑁂𑀢𑀺 𑀧𑀼𑀭𑀺𑀲𑁄, 𑀓𑀫𑁆𑀧𑀻𑀭𑁂 𑀑𑀢𑀓𑀦𑁆𑀢𑀺𑀓𑁂.
𑀅𑀢𑁆𑀢𑁂 𑀓𑀺𑀘𑁆𑀘𑁂 𑀲𑀫𑀼𑀧𑁆𑀧𑀦𑁆𑀦𑁂, 𑀅𑀢𑁆𑀢𑀸𑀬 𑀫𑁂 𑀧𑀯𑀺𑀲𑁆𑀲𑀢𑀺..
2. 𑀭𑀸𑀚𑀢𑁄 𑀯𑀸 𑀢𑀼𑀭𑀼𑀢𑁆𑀢𑀲𑁆𑀲, 𑀘𑁄𑀭𑀢𑁄 𑀧𑀻𑀴𑀺𑀢𑀲𑁆𑀲 𑀯𑀸.
𑀇𑀡𑀲𑁆𑀲 𑀯𑀸 𑀧𑀫𑁄𑀓𑁆𑀓𑀸𑀬, 𑀢𑀼𑀧𑁆𑀧𑀺𑀓𑁆𑀓𑁂 𑀆𑀧𑀢𑀸𑀲𑀼 𑀯𑀸.
𑀏𑀢𑀢𑀢𑁆𑀢𑀸𑀬 𑀮𑁄𑀓𑀲𑁆𑀫𑀺𑀫𑁆, 𑀦𑀺𑀥𑀺 𑀦𑀸𑀫 𑀦𑀺𑀥𑀻𑀬𑀢𑀺..
3. 𑀢𑀸𑀯𑀲𑁆𑀲𑀼𑀦𑀺𑀳𑀺𑀢𑁄 𑀲𑀦𑁆𑀢𑁄, 𑀓𑀫𑁆𑀧𑀻𑀭𑁂 𑀑𑀢𑀓𑀦𑁆𑀢𑀺𑀓𑁂.
𑀦 𑀲𑀧𑁆𑀧𑁄 𑀲𑀧𑁆𑀧𑀢𑀸 𑀏𑀯, 𑀢𑀲𑁆𑀲 𑀢𑀫𑁆 𑀉𑀧𑀓𑀧𑁆𑀧𑀢𑀺..
4. 𑀦𑀺𑀥𑀺 𑀯𑀸 𑀝𑀸𑀦𑀸 𑀘𑀯𑀢𑀺, 𑀲𑀜𑁆𑀜𑀸 𑀯𑀸𑀲𑁆𑀲 𑀯𑀺𑀫𑀼𑀬𑁆𑀳𑀢𑀺.
𑀦𑀸𑀓𑀸 𑀯𑀸 𑀅𑀧𑀦𑀸𑀫𑁂𑀦𑁆𑀢𑀺, 𑀬𑀓𑁆𑀓𑀸 𑀯𑀸𑀧𑀺 𑀳𑀭𑀦𑁆𑀢𑀺 𑀦𑀫𑁆..
5. 𑀅𑀧𑁆𑀧𑀺𑀬𑀸 𑀯𑀸𑀧𑀺 𑀢𑀸𑀬𑀸𑀢𑀸, 𑀉𑀢𑁆𑀥𑀭𑀦𑁆𑀢𑀺 𑀅𑀧𑀲𑁆𑀲𑀢𑁄.
𑀬𑀢𑀸 𑀧𑀼𑀜𑁆𑀜𑀓𑁆𑀓𑀬𑁄 𑀳𑁄𑀢𑀺, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑀫𑁆 𑀯𑀺𑀦𑀲𑁆𑀲𑀢𑀺..
6. 𑀬𑀲𑁆𑀲 𑀢𑀸𑀦𑁂𑀦 𑀲𑀻𑀮𑁂𑀦, 𑀲𑀫𑁆𑀬𑀫𑁂𑀦 𑀢𑀫𑁂𑀦 𑀘.
𑀦𑀺𑀥𑀻 𑀲𑀼𑀦𑀺𑀳𑀺𑀢𑁄 𑀳𑁄𑀢𑀺, 𑀇𑀢𑁆𑀢𑀺𑀬𑀸 𑀧𑀼𑀭𑀺𑀲𑀲𑁆𑀲 𑀯𑀸..
7. 𑀘𑁂𑀢𑀺𑀬𑀫𑁆𑀳𑀺 𑀘 𑀲𑀗𑁆𑀓𑁂 𑀯𑀸, 𑀧𑀼𑀓𑁆𑀓𑀮𑁂 𑀅𑀢𑀺𑀢𑀻𑀲𑀼 𑀯𑀸.
𑀫𑀸𑀢𑀭𑀺 𑀧𑀺𑀢𑀭𑀺 𑀘𑀸𑀧𑀺 , 𑀅𑀢𑁄 𑀚𑁂𑀝𑁆𑀝𑀫𑁆𑀳𑀺 𑀧𑀸𑀢𑀭𑀺..
8. 𑀏𑀲𑁄 𑀦𑀺𑀥𑀺 𑀲𑀼𑀦𑀺𑀳𑀺𑀢𑁄, 𑀅𑀚𑁂𑀬𑁆𑀬𑁄 𑀅𑀦𑀼𑀓𑀸𑀫𑀺𑀓𑁄.
𑀧𑀳𑀸𑀬 𑀓𑀫𑀦𑀻𑀬𑁂𑀲𑀼, 𑀏𑀢𑀫𑁆 𑀆𑀢𑀸𑀬 𑀓𑀘𑁆𑀘𑀢𑀺..
9. 𑀅𑀲𑀸𑀥𑀸𑀭𑀡𑀫𑀜𑁆𑀜𑁂𑀲𑀫𑁆, 𑀅𑀘𑁄𑀭𑀸𑀳𑀭𑀡𑁄 𑀦𑀺𑀥𑀺.
𑀓𑀬𑀺𑀭𑀸𑀢 𑀥𑀻𑀭𑁄 𑀧𑀼𑀜𑁆𑀜𑀸𑀦𑀺, 𑀬𑁄 𑀦𑀺𑀥𑀺 𑀅𑀦𑀼𑀓𑀸𑀫𑀺𑀓𑁄..
10. 𑀏𑀲 𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀸𑀦𑀫𑁆, 𑀲𑀧𑁆𑀧𑀓𑀸𑀫𑀢𑀢𑁄 𑀦𑀺𑀥𑀺.
𑀬𑀫𑁆 𑀬𑀢𑁂𑀯𑀸𑀧𑀺𑀧𑀢𑁆𑀢𑁂𑀦𑁆𑀢𑀺, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
11. 𑀲𑀼𑀯𑀡𑁆𑀡𑀢𑀸 𑀲𑀼𑀲𑀭𑀢𑀸, 𑀲𑀼𑀲𑀡𑁆𑀝𑀸𑀦𑀸 𑀲𑀼𑀭𑀽𑀧𑀢𑀸.
𑀆𑀥𑀺𑀧𑀘𑁆𑀘𑀧𑀭𑀺𑀯𑀸𑀭𑁄, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
12. 𑀧𑀢𑁂𑀲𑀭𑀚𑁆𑀚𑀫𑁆 𑀇𑀲𑁆𑀲𑀭𑀺𑀬𑀫𑁆, 𑀘𑀓𑁆𑀓𑀯𑀢𑁆𑀢𑀺𑀲𑀼𑀓𑀫𑁆 𑀧𑀺𑀬𑀫𑁆.
𑀢𑁂𑀯𑀭𑀚𑁆𑀚𑀫𑁆𑀧𑀺 𑀢𑀺𑀧𑁆𑀧𑁂𑀲𑀼, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
13. 𑀫𑀸𑀦𑀼𑀲𑁆𑀲𑀺𑀓𑀸 𑀘 𑀲𑀫𑁆𑀧𑀢𑁆𑀢𑀺, 𑀢𑁂𑀯𑀮𑁄𑀓𑁂 𑀘 𑀬𑀸 𑀭𑀢𑀺.
𑀬𑀸 𑀘 𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀲𑀫𑁆𑀧𑀢𑁆𑀢𑀺, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
14. 𑀫𑀺𑀢𑁆𑀢𑀲𑀫𑁆𑀧𑀢𑀫𑀸𑀓𑀫𑁆𑀫, 𑀬𑁄𑀦𑀺𑀲𑁄𑀯 𑀧𑀬𑀼𑀜𑁆𑀚𑀢𑁄.
𑀯𑀺𑀚𑁆𑀚𑀸 𑀯𑀺𑀫𑀼𑀢𑁆𑀢𑀺 𑀯𑀲𑀻𑀧𑀸𑀯𑁄, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
15. 𑀧𑀝𑀺𑀲𑀫𑁆𑀧𑀺𑀢𑀸 𑀯𑀺𑀫𑁄𑀓𑁆𑀓𑀸 𑀘, 𑀬𑀸 𑀘 𑀲𑀸𑀯𑀓𑀧𑀸𑀭𑀫𑀻.
𑀧𑀘𑁆𑀘𑁂𑀓𑀧𑁄𑀥𑀺 𑀧𑀼𑀢𑁆𑀥𑀧𑀽𑀫𑀺, 𑀲𑀧𑁆𑀧𑀫𑁂𑀢𑁂𑀦 𑀮𑀧𑁆𑀧𑀢𑀺..
16. 𑀏𑀯𑀫𑁆 𑀫𑀳𑀢𑁆𑀢𑀺𑀓𑀸 𑀏𑀲𑀸, 𑀬𑀢𑀺𑀢𑀫𑁆 𑀧𑀼𑀜𑁆𑀜𑀲𑀫𑁆𑀧𑀢𑀸.
𑀢𑀲𑁆𑀫𑀸 𑀥𑀻𑀭𑀸 𑀧𑀲𑀫𑁆𑀲𑀦𑁆𑀢𑀺, 𑀧𑀡𑁆𑀝𑀺𑀢𑀸 𑀓𑀢𑀧𑀼𑀜𑁆𑀜𑀢𑀦𑁆𑀢𑀺..
𑀦𑀺𑀥𑀺𑀓𑀡𑁆𑀝𑀲𑀼𑀢𑁆𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
9. 𑀫𑁂𑀢𑁆𑀢𑀲𑀼𑀢𑁆𑀢𑀫𑁆
1. 𑀓𑀭𑀡𑀻𑀬𑀫𑀢𑁆𑀢𑀓𑀼𑀲𑀮𑁂𑀦, 𑀬𑀦𑁆𑀢𑀲𑀦𑁆𑀢𑀫𑁆 𑀧𑀢𑀫𑁆 𑀅𑀧𑀺𑀲𑀫𑁂𑀘𑁆𑀘.
𑀲𑀓𑁆𑀓𑁄 𑀉𑀚𑀽 𑀘 𑀲𑀼𑀳𑀼𑀚𑀽 𑀘, 𑀲𑀼𑀯𑀘𑁄 𑀘𑀲𑁆𑀲 𑀫𑀼𑀢𑀼 𑀅𑀦𑀢𑀺𑀫𑀸𑀦𑀻..
2. 𑀲𑀦𑁆𑀢𑀼𑀲𑁆𑀲𑀓𑁄 𑀘 𑀲𑀼𑀧𑀭𑁄 𑀘, 𑀅𑀧𑁆𑀧𑀓𑀺𑀘𑁆𑀘𑁄 𑀘 𑀲𑀮𑁆𑀮𑀳𑀼𑀓𑀯𑀼𑀢𑁆𑀢𑀺.
𑀲𑀦𑁆𑀢𑀺𑀦𑁆𑀢𑁆𑀭𑀺𑀬𑁄 𑀘 𑀦𑀺𑀧𑀓𑁄 𑀘, 𑀅𑀧𑁆𑀧𑀓𑀧𑁆𑀧𑁄 𑀓𑀼𑀮𑁂𑀲𑁆𑀯𑀦𑀦𑀼𑀓𑀺𑀢𑁆𑀥𑁄..
3. 𑀦 𑀘 𑀓𑀼𑀢𑁆𑀢𑀫𑀸𑀘𑀭𑁂 𑀓𑀺𑀜𑁆𑀘𑀺, 𑀬𑁂𑀦 𑀯𑀺𑀜𑁆𑀜𑀽 𑀧𑀭𑁂 𑀉𑀧𑀯𑀢𑁂𑀬𑁆𑀬𑀼𑀫𑁆.
𑀲𑀼𑀓𑀺𑀦𑁄𑀯 𑀓𑁂𑀫𑀺𑀦𑁄 𑀳𑁄𑀦𑁆𑀢𑀼, 𑀲𑀧𑁆𑀧𑀲𑀢𑁆𑀢𑀸 𑀧𑀯𑀦𑁆𑀢𑀼 𑀲𑀼𑀓𑀺𑀢𑀢𑁆𑀢𑀸..
4. 𑀬𑁂 𑀓𑁂𑀘𑀺 𑀧𑀸𑀡𑀧𑀽𑀢𑀢𑁆𑀢𑀺, 𑀢𑀲𑀸 𑀯𑀸 𑀢𑀸𑀯𑀭𑀸 𑀯𑀦𑀯𑀲𑁂𑀲𑀸.
𑀢𑀻𑀓𑀸 𑀯𑀸 𑀬𑁂𑀯 𑀫𑀳𑀦𑁆𑀢𑀸 , 𑀫𑀚𑁆𑀚𑀺𑀫𑀸 𑀭𑀲𑁆𑀲𑀓𑀸 𑀅𑀡𑀼𑀓𑀢𑀽𑀮𑀸..
5. 𑀢𑀺𑀝𑁆𑀝𑀸 𑀯𑀸 𑀬𑁂𑀯 𑀅𑀢𑀺𑀝𑁆𑀝𑀸 𑀢𑀽𑀭𑁂 𑀯𑀲𑀦𑁆𑀢𑀺 𑀅𑀯𑀺𑀢𑀽𑀭𑁂.
𑀧𑀽𑀢𑀸 𑀯 , 𑀲𑀧𑁆𑀧𑀲𑀢𑁆𑀢𑀸 𑀧𑀯𑀦𑁆𑀢𑀼 𑀲𑀼𑀓𑀺𑀢𑀢𑁆𑀢𑀸..
6. 𑀦 𑀧𑀭𑁄 𑀧𑀭𑀫𑁆 𑀦𑀺𑀓𑀼𑀧𑁆𑀧𑁂𑀢, 𑀦𑀸𑀢𑀺𑀫𑀜𑁆𑀜𑁂𑀢 𑀓𑀢𑁆𑀢𑀘𑀺 𑀦 𑀓𑀜𑁆𑀘𑀺.
𑀧𑁆𑀬𑀸𑀭𑁄𑀲𑀦𑀸 𑀧𑀝𑀺𑀓𑀲𑀜𑁆𑀜𑀸, 𑀦𑀸𑀜𑁆𑀜𑀫𑀜𑁆𑀜𑀲𑁆𑀲 𑀢𑀼𑀓𑁆𑀓𑀫𑀺𑀘𑁆𑀘𑁂𑀬𑁆𑀬..
7. 𑀫𑀸𑀢𑀸 𑀬𑀢𑀸 𑀦𑀺𑀬𑀫𑁆 𑀧𑀼𑀢𑁆𑀢𑀫𑀸𑀬𑀼𑀲𑀸 𑀏𑀓𑀧𑀼𑀢𑁆𑀢𑀫𑀦𑀼𑀭𑀓𑁆𑀓𑁂.
𑀏𑀯𑀫𑁆𑀧𑀺 𑀲𑀧𑁆𑀧𑀧𑀽𑀢𑁂𑀲𑀼, 𑀫𑀸𑀦𑀲𑀫𑁆 𑀧𑀸𑀯𑀬𑁂 𑀅𑀧𑀭𑀺𑀫𑀸𑀡𑀫𑁆..
8. 𑀫𑁂𑀢𑁆𑀢𑀜𑁆𑀘 𑀲𑀧𑁆𑀧𑀮𑁄𑀓𑀲𑁆𑀫𑀺, 𑀫𑀸𑀦𑀲𑀫𑁆 𑀧𑀸𑀯𑀬𑁂 𑀅𑀧𑀭𑀺𑀫𑀸𑀡𑀫𑁆.
𑀉𑀢𑁆𑀥𑀫𑁆 𑀅𑀥𑁄 𑀘 𑀢𑀺𑀭𑀺𑀬𑀜𑁆𑀘, 𑀅𑀲𑀫𑁆𑀧𑀸𑀥𑀫𑁆 𑀅𑀯𑁂𑀭𑀫𑀲𑀧𑀢𑁆𑀢𑀫𑁆..
9. 𑀢𑀺𑀝𑁆𑀝𑀫𑁆 𑀘𑀭𑀫𑁆 𑀦𑀺𑀲𑀺𑀦𑁆𑀦𑁄 𑀯, 𑀲𑀬𑀸𑀦𑁄 𑀬𑀸𑀯𑀢𑀸𑀲𑁆𑀲 𑀯𑀺𑀢𑀫𑀺𑀢𑁆𑀥𑁄.
𑀏𑀢𑀫𑁆 𑀲𑀢𑀺𑀫𑁆 𑀅𑀥𑀺𑀝𑁆𑀝𑁂𑀬𑁆𑀬, 𑀧𑁆𑀭𑀳𑁆𑀫𑀫𑁂𑀢𑀫𑁆 𑀯𑀺𑀳𑀸𑀭𑀫𑀺𑀥𑀫𑀸𑀳𑀼..
10. 𑀢𑀺𑀝𑁆𑀝𑀺𑀜𑁆𑀘 𑀅𑀦𑀼𑀧𑀓𑁆𑀓𑀫𑁆𑀫, 𑀲𑀻𑀮𑀯𑀸 𑀢𑀲𑁆𑀲𑀦𑁂𑀦 𑀲𑀫𑁆𑀧𑀦𑁆𑀦𑁄.
𑀓𑀸𑀫𑁂𑀲𑀼 𑀯𑀺𑀦𑀬 𑀓𑁂𑀥𑀫𑁆, 𑀦 𑀳𑀺 𑀚𑀸𑀢𑀼𑀓𑁆𑀓𑀧𑁆𑀧𑀲𑁂𑀬𑁆𑀬 𑀧𑀼𑀦 𑀭𑁂𑀢𑀻𑀢𑀺..
𑀫𑁂𑀢𑁆𑀢𑀲𑀼𑀢𑁆𑀢𑀫𑁆 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀫𑁆.
𑀓𑀼𑀢𑁆𑀢𑀓𑀧𑀸𑀝𑀧𑀸𑀴𑀺 𑀦𑀺𑀝𑁆𑀝𑀺𑀢𑀸.
