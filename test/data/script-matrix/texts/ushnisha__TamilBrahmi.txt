𑀉𑀱𑁆𑀡𑀻𑀱 𑀯𑀺𑀚𑀬 𑀥𑀸𑀭𑀡𑀻
𑀑𑀫𑁆 𑀦𑀫𑁄 𑀧𑀓𑀯𑀢𑁂 𑀲𑀭𑁆𑀯𑀢𑁆𑀭𑁃𑀮𑁄𑀓𑁆𑀬 𑀧𑁆𑀭𑀢𑀺𑀯𑀺𑀰𑀺𑀱𑁆𑀝𑀸𑀬 𑀧𑀼𑀢𑁆𑀥𑀸𑀬 𑀢𑁂 𑀦𑀫𑀂 .
𑀢𑀢𑁆𑀬𑀢𑀸 .
𑀑𑀫𑁆 𑀧𑁆𑀭𑀽𑀫𑁆 𑀧𑁆𑀭𑀽𑀫𑁆 𑀧𑁆𑀭𑀽𑀫𑁆 .
𑀰𑁄𑀥𑀬 𑀰𑁄𑀥𑀬 . 𑀯𑀺𑀰𑁄𑀥𑀬 𑀯𑀺𑀰𑁄𑀥𑀬 .
𑀅𑀲𑀫 𑀲𑀫𑀦𑁆𑀢 𑀅𑀯𑀧𑀸𑀲𑀲𑁆𑀧𑀭𑀡 𑀓𑀢𑀺𑀓𑀓𑀦 𑀲𑁆𑀯𑀧𑀸𑀯 𑀯𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀅𑀧𑀺𑀱𑀺𑀜𑁆𑀘𑀦𑁆𑀢𑀼 𑀫𑀸𑀫𑁆 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢 𑀲𑀼𑀓𑀢 𑀯𑀭𑀯𑀘𑀦 𑀅𑀫𑁆𑀭𑀼𑀢 𑀅𑀧𑀺𑀱𑁂𑀓𑁃𑀂 .
𑀫𑀳𑀸𑀫𑀼𑀢𑁆𑀭 𑀫𑀦𑁆𑀢𑁆𑀭𑀧𑀢𑁃𑀂 .
𑀆𑀳𑀭 𑀆𑀳𑀭 .
𑀫𑀫 𑀆𑀬𑀼𑀂 𑀲𑀦𑁆𑀥𑀸𑀭𑀡𑀺 .
𑀰𑁄𑀥𑀬 𑀰𑁄𑀥𑀬 . 𑀯𑀺𑀰𑁄𑀥𑀬 𑀯𑀺𑀰𑁄𑀥𑀬 .
𑀓𑀓𑀦 𑀲𑁆𑀯𑀧𑀸𑀯 𑀯𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀉𑀱𑁆𑀡𑀻𑀱 𑀯𑀺𑀚𑀬 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀲𑀳𑀲𑁆𑀭 𑀭𑀰𑁆𑀫𑀺 𑀲𑀫𑁆𑀘𑁄𑀢𑀺𑀢𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢 𑀅𑀯𑀮𑁄𑀓𑀺𑀦𑀺 .
𑀱𑀝𑁆𑀧𑀸𑀭𑀫𑀺𑀢𑀸 𑀧𑀭𑀺𑀧𑀽𑀭𑀡𑀺 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢 𑀫𑀢𑀺 𑀢𑀰𑀧𑀽𑀫𑀺 𑀧𑁆𑀭𑀢𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢 𑀳𑁆𑀭𑀼𑀢𑀬 𑀅𑀥𑀺𑀱𑁆𑀝𑀸𑀦 𑀅𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀫𑀼𑀢𑁆𑀭𑁂 𑀫𑀼𑀢𑁆𑀭𑁂 . 𑀫𑀳𑀸𑀫𑀼𑀢𑁆𑀭𑁂 .
𑀯𑀚𑁆𑀭𑀓𑀸𑀬 𑀲𑀫𑁆𑀳𑀭𑀡 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀲𑀭𑁆𑀯𑀓𑀭𑁆𑀫 𑀆𑀯𑀭𑀡 𑀯𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀧𑁆𑀭𑀢𑀺𑀦𑀺𑀯𑀭𑁆𑀢𑀬 .
𑀫𑀫 𑀆𑀬𑀼𑀭𑁆𑀯𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢 𑀲𑀫𑀬 𑀅𑀥𑀺𑀱𑁆𑀝𑀸𑀦 𑀅𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀑𑀫𑁆 𑀫𑀼𑀦𑀺 𑀫𑀼𑀦𑀺. 𑀫𑀳𑀸𑀫𑀼𑀦𑀺 𑀯𑀺𑀫𑀼𑀦𑀺 𑀯𑀺𑀫𑀼𑀦𑀺 . 𑀫𑀳𑀸𑀯𑀺𑀫𑀼𑀦𑀺 .
𑀫𑀢𑀺 𑀫𑀢𑀺 . 𑀫𑀳𑀸𑀫𑀢𑀺 . 𑀫𑀫𑀢𑀺 . 𑀲𑀼𑀫𑀢𑀺.
𑀢𑀢𑀸𑀢𑀸 𑀧𑀽𑀢𑀓𑁄𑀝𑀺 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀯𑀺𑀲𑁆𑀧𑀼𑀝 𑀧𑀼𑀢𑁆𑀥𑁂 𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀳𑁂 𑀳𑁂 . 𑀚𑀬 𑀚𑀬 . 𑀯𑀺𑀚𑀬 𑀯𑀺𑀚𑀬 .
𑀲𑁆𑀫𑀭 𑀲𑁆𑀫𑀭 . 𑀲𑁆𑀧𑀭 𑀲𑁆𑀧𑀭 . 𑀲𑁆𑀧𑀸𑀭𑀬 𑀲𑁆𑀧𑀸𑀭𑀬 .
𑀲𑀭𑁆𑀯𑀧𑀼𑀢𑁆𑀥 𑀅𑀥𑀺𑀱𑁆𑀝𑀸𑀦 𑀅𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀰𑀼𑀢𑁆𑀥𑁂 𑀰𑀼𑀢𑁆𑀥𑁂 . 𑀧𑀼𑀢𑁆𑀥𑁂 𑀧𑀼𑀢𑁆𑀥𑁂 .
𑀯𑀚𑁆𑀭𑁂 𑀯𑀚𑁆𑀭𑁂 . 𑀫𑀳𑀸𑀯𑀚𑁆𑀭𑁂 𑀲𑀼𑀯𑀚𑁆𑀭𑁂 .
𑀯𑀚𑁆𑀭 𑀓𑀭𑁆𑀧𑁂 . 𑀚𑀬 𑀓𑀭𑁆𑀧𑁂 . 𑀯𑀺𑀚𑀬 𑀓𑀭𑁆𑀧𑁂 . 𑀯𑀚𑁆𑀭𑀚𑁆𑀯𑀸𑀮 𑀓𑀭𑁆𑀧𑁂 .
𑀯𑀚𑁆𑀭𑁂 𑀉𑀢𑁆𑀧𑀯𑁂 . 𑀯𑀚𑁆𑀭 𑀲𑀫𑁆𑀧𑀯𑁂 𑀯𑀚𑁆𑀭𑁂 𑀯𑀚𑁆𑀭𑀺𑀡𑀺 .
𑀯𑀚𑁆𑀭𑀫𑁆 𑀧𑀯𑀢𑀼 𑀫𑀫 𑀰𑀭𑀻𑀭𑀫𑁆 𑀲𑀭𑁆𑀯 𑀲𑀢𑁆𑀢𑁆𑀯𑀸𑀦𑀸𑀫𑁆 𑀘 𑀓𑀸𑀬 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑀺𑀭𑁆𑀧𑀯𑀢𑀼 .
𑀫𑁂 𑀲𑀢𑀸 𑀲𑀭𑁆𑀯𑀓𑀢𑀺 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑀺𑀰𑁆𑀘 . 𑀲𑀭𑁆𑀯𑀢𑀢𑀸𑀓𑀢𑀰𑁆𑀘 𑀫𑀸𑀫𑁆 𑀲𑀫𑀸𑀰𑁆𑀯𑀸𑀲𑀬𑀦𑁆𑀢𑀼 .
𑀧𑀼𑀢𑁆𑀥𑁆𑀬 𑀧𑀼𑀢𑁆𑀥𑁆𑀬 . 𑀲𑀺𑀢𑁆𑀥𑁆𑀬 𑀲𑀺𑀢𑁆𑀥𑁆𑀬 .
𑀧𑁄𑀥𑀬 𑀧𑁄𑀥𑀬 . 𑀯𑀺𑀧𑁄𑀥𑀬 𑀯𑀺𑀧𑁄𑀥𑀬 .
𑀫𑁄𑀘𑀬 𑀫𑁄𑀘𑀬 . 𑀯𑀺𑀫𑁄𑀘𑀬 𑀯𑀺𑀫𑁄𑀘𑀬 .
𑀰𑁄𑀥𑀬 𑀰𑁄𑀥𑀬 . 𑀯𑀺𑀰𑁄𑀥𑀬 𑀯𑀺𑀰𑁄𑀥𑀬 .
𑀲𑀫𑀦𑁆𑀢𑀸𑀦 𑀫𑁄𑀘𑀬 𑀫𑁄𑀘𑀬 .
𑀲𑀫𑀦𑁆𑀢 𑀭𑀰𑁆𑀫𑀺 𑀧𑀭𑀺𑀰𑀼𑀢𑁆𑀥𑁂 .
𑀲𑀭𑁆𑀯 𑀢𑀢𑀸𑀓𑀢 𑀳𑁆𑀭𑀼𑀢𑀬 𑀅𑀥𑀺𑀱𑁆𑀝𑀸𑀦 𑀅𑀥𑀺𑀱𑁆𑀝𑀺𑀢𑁂 .
𑀫𑀼𑀢𑁆𑀭𑁂 𑀫𑀼𑀢𑁆𑀭𑁂 . 𑀫𑀳𑀸𑀫𑀼𑀢𑁆𑀭𑁂 .
𑀫𑀳𑀸𑀫𑀼𑀢𑁆𑀭𑀸 𑀫𑀦𑁆𑀢𑁆𑀭 𑀧𑀢𑁃𑀂 𑀲𑁆𑀯𑀸𑀳𑀸 ..
𑀑𑀫𑁆 𑀧𑁆𑀭𑀽𑀫𑁆 𑀲𑁆𑀯𑀸𑀳𑀸 .
𑀑𑀫𑁆 𑀅𑀫𑁆𑀭𑀼𑀢 𑀆𑀬𑀼𑀭𑁆𑀢𑀢𑁂 𑀲𑁆𑀯𑀸𑀳𑀸 .
