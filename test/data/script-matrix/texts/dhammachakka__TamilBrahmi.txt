𑀥𑀫𑁆𑀫𑀘𑀓𑁆𑀓𑀧𑁆𑀧𑀯𑀢𑁆𑀢𑀦 𑀲𑀼𑀢𑁆𑀢𑀫𑁆
𑀦𑀫𑁄 𑀢𑀲𑁆𑀲 𑀧𑀓𑀯𑀢𑁄 𑀅𑀭𑀳𑀢𑁄 𑀲𑀫𑁆𑀫𑀸 𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑀲𑁆𑀲
𑀏𑀯𑀫𑁆 𑀫𑁂 𑀲𑀼𑀢𑀫𑁆, 𑀏𑀓𑀫𑁆 𑀲𑀫𑀬𑀫𑁆 𑀧𑀓𑀯𑀸 𑀧𑀸𑀭𑀸𑀡𑀲𑀺𑀬𑀫𑁆 𑀯𑀺𑀳𑀭𑀢𑀺 𑀇𑀲𑀺𑀧𑀢𑀦𑁂 𑀫𑀺𑀓𑀢𑀸𑀬𑁂 𑀢𑀢𑁆𑀭 𑀓𑁄 𑀧𑀓𑀯𑀸 𑀧𑀜𑁆𑀘𑀯𑀓𑁆𑀓𑀺𑀬𑁂 𑀧𑀺𑀓𑁆𑀓𑀽 𑀆𑀫𑀦𑁆𑀢𑁂𑀲𑀺: "𑀢𑁆𑀯𑁂 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀅𑀦𑁆𑀢𑀸 𑀧𑀧𑁆𑀧𑀚𑀺𑀢𑁂𑀦 𑀦 𑀲𑁂𑀯𑀺𑀢𑀧𑁆𑀧𑀸. 𑀬𑁄𑀘𑀬𑀸𑀫𑁆 𑀓𑀸𑀫𑁂𑀲𑀼 𑀓𑀸𑀫𑀲𑀼𑀓𑀮𑁆𑀮𑀺𑀓𑀸𑀦𑀼𑀬𑁄𑀓𑁄 𑀳𑀻𑀦𑁄 𑀓𑀫𑁆𑀫𑁄 𑀧𑁄𑀢𑀼𑀚𑁆𑀚𑀦𑀺𑀓𑁄 𑀅𑀦𑀭𑀺𑀬𑁄 𑀅𑀦𑀢𑁆𑀢𑀲𑀫𑁆𑀳𑀺𑀢𑁄, 𑀬𑁄 𑀘𑀸𑀬𑀫𑁆 𑀅𑀢𑁆𑀢𑀓𑀺𑀮𑀫𑀢𑀸𑀦𑀼𑀬𑁄𑀓𑁄 𑀢𑀼𑀓𑁆𑀓𑁄 𑀅𑀦𑀭𑀺𑀬𑁄 𑀅𑀦𑀢𑁆𑀢𑀲𑀫𑁆𑀳𑀺𑀢𑁄, 𑀏𑀢𑁂 𑀢𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀉𑀧𑁄 𑀅𑀦𑁆𑀢𑁂 𑀅𑀦𑀼𑀧𑀓𑀫𑁆𑀫 𑀫𑀚𑁆𑀚𑀺𑀫𑀸 𑀧𑀝𑀺𑀧𑀢𑀸 𑀢𑀢𑀸𑀓𑀢𑁂𑀦 𑀅𑀧𑀺𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑀸 𑀘𑀓𑁆𑀓𑀼𑀓𑀭𑀡𑀻 𑀜𑀸𑀡𑀓𑀭𑀡𑀻 𑀉𑀧𑀲𑀫𑀸𑀬 𑀅𑀧𑀺𑀜𑁆𑀜𑀸𑀬 𑀲𑀫𑁆𑀧𑁄𑀥𑀸𑀬 𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀸𑀬 𑀲𑀫𑁆𑀯𑀢𑁆𑀢𑀢𑀺. 𑀓𑀢𑀫𑀸 𑀘 𑀲𑀸 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀫𑀚𑁆𑀚𑀺𑀫𑀸 𑀧𑀝𑀺𑀧𑀢𑀸 𑀢𑀢𑀸𑀓𑀢𑁂𑀦 𑀅𑀧𑀺𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑀸 𑀘𑀓𑁆𑀓𑀼𑀓𑀭𑀡𑀻 𑀜𑀸𑀡𑀓𑀭𑀡𑀻 𑀉𑀧𑀲𑀫𑀸𑀬 𑀅𑀧𑀺𑀜𑁆𑀜𑀸𑀬 𑀲𑀫𑁆𑀧𑁄𑀥𑀸𑀬 𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀸𑀬 𑀲𑀫𑁆𑀯𑀢𑁆𑀢𑀢𑀺: 𑀅𑀬𑀫𑁂𑀯 𑀅𑀭𑀺𑀬𑁄 𑀅𑀝𑁆𑀝𑀗𑁆𑀓𑀺𑀓𑁄 𑀫𑀓𑁆𑀓𑁄 𑀲𑁂𑀬𑁆𑀬𑀢𑀻𑀢𑀫𑁆: 𑀲𑀫𑁆𑀫𑀸𑀢𑀺𑀝𑁆𑀝𑀺 𑀲𑀫𑁆𑀫𑀸𑀲𑀗𑁆𑀓𑀧𑁆𑀧𑁄 𑀲𑀫𑁆𑀫𑀸𑀯𑀸𑀘𑀸 𑀲𑀫𑁆𑀫𑀸𑀓𑀫𑁆𑀫𑀦𑁆𑀢𑁄 𑀲𑀫𑁆𑀫𑀸𑀆𑀚𑀻𑀯𑁄 𑀲𑀫𑁆𑀫𑀸𑀯𑀸𑀬𑀸𑀫𑁄 𑀲𑀫𑁆𑀫𑀸𑀲𑀢𑀺 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑀸𑀥𑀺. 𑀅𑀬𑀫𑁆 𑀓𑁄 𑀲𑀸 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀫𑀚𑁆𑀚𑀺𑀫𑀸 𑀧𑀝𑀺𑀧𑀢𑀸 𑀢𑀢𑀸𑀓𑀢𑁂𑀦 𑀅𑀧𑀺𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑀸 𑀘𑀓𑁆𑀓𑀼𑀓𑀭𑀡𑀻 𑀜𑀸𑀡𑀓𑀭𑀡𑀻 𑀉𑀧𑀲𑀫𑀸𑀬 𑀅𑀧𑀺𑀜𑁆𑀜𑀸𑀬 𑀲𑀫𑁆𑀧𑁄𑀥𑀸𑀬 𑀦𑀺𑀧𑁆𑀧𑀸𑀦𑀸𑀬 𑀲𑀫𑁆𑀯𑀢𑁆𑀢𑀢𑀺.
𑀇𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆: 𑀚𑀸𑀢𑀺𑀧𑀺 𑀢𑀼𑀓𑁆𑀓𑀸 𑀚𑀭𑀸𑀧𑀺 𑀢𑀼𑀓𑁆𑀓𑀸 𑀯𑁆𑀬𑀸𑀥𑀺𑀧𑀺 𑀢𑀼𑀓𑁆𑀓𑁄 𑀫𑀭𑀡𑀫𑁆𑀧𑀺 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀅𑀧𑁆𑀧𑀺𑀬𑁂𑀳𑀺 𑀲𑀫𑁆𑀧𑀬𑁄𑀓𑁄 𑀢𑀼𑀓𑁆𑀓𑁄 𑀧𑀺𑀬𑁂𑀳𑀺 𑀯𑀺𑀧𑁆𑀧𑀬𑁄𑀓𑁄 𑀢𑀼𑀓𑁆𑀓𑁄 𑀬𑀫𑁆𑀧𑀺𑀘𑁆𑀘𑀫𑁆 𑀦 𑀮𑀧𑀢𑀺 𑀢𑀫𑁆𑀧𑀺 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀲𑀗𑁆𑀓𑀺𑀢𑁆𑀢𑁂𑀦 𑀧𑀜𑁆𑀘𑀼𑀧𑀸𑀢𑀸𑀦𑀓𑁆𑀓𑀦𑁆𑀥𑀸 𑀢𑀼𑀓𑁆𑀓𑀸". 𑀇𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀢𑀼𑀓𑁆𑀓𑀲𑀫𑀼𑀢𑀬𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆: "𑀬𑀸𑀬𑀫𑁆 𑀢𑀡𑁆𑀳𑀸 𑀧𑁄𑀦𑁄𑀧𑀯𑀺𑀓𑀸 𑀦𑀦𑁆𑀢𑀺𑀭𑀸𑀓𑀲𑀳𑀓𑀢𑀸 𑀢𑀢𑁆𑀭 𑀢𑀢𑁆𑀭𑀸𑀧𑀺𑀦𑀦𑁆𑀢𑀺𑀦𑀻, 𑀲𑁂𑀬𑁆𑀬𑀢𑀻𑀢𑀫𑁆: 𑀓𑀸𑀫𑀢𑀡𑁆𑀳𑀸 𑀧𑀯𑀢𑀡𑁆𑀳𑀸 𑀯𑀺𑀧𑀯𑀢𑀡𑁆𑀳𑀸". 𑀇𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆: 𑀬𑁄 𑀢𑀲𑁆𑀲𑀸𑀬𑁂𑀯 𑀢𑀡𑁆𑀳𑀸𑀬 𑀅𑀲𑁂𑀲𑀯𑀺𑀭𑀸𑀓𑀦𑀺𑀭𑁄𑀥𑁄 𑀘𑀸𑀓𑁄 𑀧𑀝𑀺𑀦𑀺𑀲𑁆𑀲𑀓𑁆𑀓𑁄 𑀫𑀼𑀢𑁆𑀢𑀺 𑀅𑀦𑀸𑀮𑀬𑁄. 𑀇𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑀓𑀸𑀫𑀺𑀦𑀻 𑀧𑀝𑀺𑀧𑀢𑀸 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆: 𑀅𑀬𑀫𑁂𑀯 𑀅𑀭𑀺𑀬𑁄 𑀅𑀝𑁆𑀝𑀗𑁆𑀓𑀺𑀓𑁄 𑀫𑀓𑁆𑀓𑁄, 𑀲𑁂𑀬𑁆𑀬𑀢𑀻𑀢𑀫𑁆: 𑀲𑀫𑁆𑀫𑀸𑀢𑀺𑀝𑁆𑀝𑀺 𑀲𑀫𑁆𑀫𑀸𑀲𑀗𑁆𑀓𑀧𑁆𑀧𑁄 𑀲𑀫𑁆𑀫𑀸𑀯𑀸𑀘𑀸 𑀲𑀫𑁆𑀫𑀸𑀓𑀫𑁆𑀫𑀦𑁆𑀢𑁄 𑀲𑀫𑁆𑀫𑀸𑀆𑀚𑀻𑀯𑁄 𑀲𑀫𑁆𑀫𑀸𑀯𑀸𑀬𑀸𑀫𑁄 𑀲𑀫𑁆𑀫𑀸𑀲𑀢𑀺 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑀸𑀥𑀺.
"𑀇𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀦𑁆𑀢𑀺" 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀭𑀺𑀜𑁆𑀜𑁂𑀬𑁆𑀬𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀫𑁆 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀭𑀺𑀜𑁆𑀜𑀸𑀢𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺.
"𑀇𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀲𑀫𑀼𑀢𑀬𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀦𑁆𑀢𑀺" 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀲𑀫𑀼𑀢𑀬𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀳𑀸𑀢𑀧𑁆𑀧𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀲𑀫𑀼𑀢𑀬𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀳𑀻𑀦𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺.
"𑀇𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀦𑁆𑀢𑀺" 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀲𑀘𑁆𑀘𑀺𑀓𑀸𑀢𑀧𑁆𑀧𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑁄 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀲𑀘𑁆𑀘𑀺𑀓𑀢𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺.
"𑀇𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑀓𑀸𑀫𑀺𑀦𑀻 𑀧𑀝𑀺𑀧𑀢𑀸 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀦𑁆𑀢𑀺" 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑀓𑀸𑀫𑀺𑀦𑀻 𑀧𑀝𑀺𑀧𑀢𑀸 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀸𑀯𑁂𑀢𑀧𑁆𑀧𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺. 𑀢𑀫𑁆 𑀓𑁄 𑀧𑀦𑀺𑀢𑀫𑁆 𑀢𑀼𑀓𑁆𑀓𑀦𑀺𑀭𑁄𑀥𑀓𑀸𑀫𑀺𑀦𑀻 𑀧𑀝𑀺𑀧𑀢𑀸 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑀫𑁆 𑀧𑀸𑀯𑀺𑀢𑀦𑁆𑀢𑀺 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀧𑀼𑀧𑁆𑀧𑁂 𑀅𑀦𑀦𑀼𑀲𑁆𑀲𑀼𑀢𑁂𑀲𑀼 𑀥𑀫𑁆𑀫𑁂𑀲𑀼 𑀘𑀓𑁆𑀓𑀼𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀜𑀸𑀡𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀧𑀜𑁆𑀜𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀯𑀺𑀚𑁆𑀚𑀸 𑀉𑀢𑀧𑀸𑀢𑀺 𑀆𑀮𑁄𑀓𑁄 𑀉𑀢𑀧𑀸𑀢𑀺.
𑀬𑀸𑀯𑀓𑀻𑀯𑀜𑁆𑀘 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀇𑀫𑁂𑀲𑀼 𑀘𑀢𑀼𑀲𑀼 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑁂𑀲𑀼 𑀏𑀯𑀫𑁆 𑀢𑀺𑀧𑀭𑀺𑀯𑀝𑁆𑀝𑀫𑁆 𑀢𑁆𑀯𑀸𑀢𑀲𑀸𑀓𑀸𑀭𑀫𑁆 𑀬𑀢𑀸𑀧𑀽𑀢𑀫𑁆 𑀜𑀸𑀡𑀢𑀲𑁆𑀲𑀦𑀫𑁆 𑀦 𑀲𑀼𑀯𑀺𑀲𑀼𑀢𑁆𑀥𑀫𑁆 𑀅𑀳𑁄𑀲𑀺, 𑀦𑁂𑀯 𑀢𑀸𑀯𑀸𑀳𑀫𑁆 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀲𑀢𑁂𑀯𑀓𑁂 𑀮𑁄𑀓𑁂 𑀲𑀫𑀸𑀭𑀓𑁂 𑀲𑀧𑁆𑀭𑀳𑁆𑀫𑀓𑁂 𑀲𑀲𑁆𑀲𑀫𑀡𑀧𑁆𑀭𑀸𑀳𑁆𑀫𑀡𑀺𑀬𑀸 𑀧𑀚𑀸𑀬 𑀲𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀸𑀬 𑀅𑀦𑀼𑀢𑁆𑀢𑀭𑀫𑁆 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑁆𑀧𑁄𑀥𑀺𑀫𑁆 𑀅𑀧𑀺𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑁄 𑀧𑀘𑁆𑀘𑀜𑁆𑀜𑀸𑀲𑀺𑀫𑁆. 𑀬𑀢𑁄 𑀘 𑀓𑁄 𑀫𑁂 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀇𑀫𑁂𑀲𑀼 𑀘𑀢𑀼𑀲𑀼 𑀅𑀭𑀺𑀬𑀲𑀘𑁆𑀘𑁂𑀲𑀼 𑀏𑀯𑀫𑁆 𑀢𑀺𑀧𑀭𑀺𑀯𑀝𑁆𑀝𑀫𑁆 𑀢𑁆𑀯𑀸𑀢𑀲𑀸𑀓𑀸𑀭𑀫𑁆 𑀬𑀢𑀸𑀧𑀽𑀢𑀫𑁆 𑀜𑀸𑀡𑀢𑀲𑁆𑀲𑀦𑀫𑁆 𑀲𑀼𑀯𑀺𑀲𑀼𑀢𑁆𑀥𑀫𑁆 𑀅𑀳𑁄𑀲𑀺, 𑀅𑀢𑀸𑀳𑀫𑁆 𑀧𑀺𑀓𑁆𑀓𑀯𑁂, 𑀲𑀢𑁂𑀯𑀓𑁂 𑀮𑁄𑀓𑁂 𑀲𑀫𑀸𑀭𑀓𑁂 𑀲𑀧𑁆𑀭𑀳𑁆𑀫𑀓𑁂 𑀲𑀲𑁆𑀲𑀫𑀡𑀧𑁆𑀭𑀸𑀳𑁆𑀫𑀡𑀺𑀬𑀸 𑀧𑀚𑀸𑀬 𑀲𑀢𑁂𑀯𑀫𑀦𑀼𑀲𑁆𑀲𑀸𑀬 𑀅𑀦𑀼𑀢𑁆𑀢𑀭𑀫𑁆 𑀲𑀫𑁆𑀫𑀸𑀲𑀫𑁆𑀧𑁄𑀥𑀺𑀫𑁆 𑀅𑀧𑀺𑀲𑀫𑁆𑀧𑀼𑀢𑁆𑀥𑁄 𑀧𑀘𑁆𑀘𑀜𑁆𑀜𑀸𑀲𑀺𑀫𑁆. 𑀜𑀸𑀡𑀜𑁆𑀘 𑀧𑀦 𑀫𑁂 𑀢𑀲𑁆𑀲𑀦𑀫𑁆 𑀉𑀢𑀧𑀸𑀢𑀺 𑀅𑀓𑀼𑀧𑁆𑀧𑀸 𑀫𑁂 𑀘𑁂𑀢𑁄𑀯𑀺𑀫𑀼𑀢𑁆𑀢𑀺, 𑀅𑀬𑀫𑀦𑁆𑀢𑀺𑀫𑀸 𑀚𑀸𑀢𑀺 𑀦𑀢𑁆𑀢𑀺𑀢𑀸𑀦𑀺 𑀧𑀼𑀦𑀧𑁆𑀧𑀯𑁄𑀢𑀺. 𑀇𑀢𑀫𑀯𑁄𑀘 𑀧𑀓𑀯𑀸 𑀅𑀢𑁆𑀢𑀫𑀦𑀸 𑀧𑀜𑁆𑀘𑀯𑀓𑁆𑀓𑀺𑀬𑀸 𑀧𑀺𑀓𑁆𑀓𑀽 𑀧𑀓𑀯𑀢𑁄 𑀧𑀸𑀲𑀺𑀢𑀫𑁆 𑀅𑀧𑀺𑀦𑀦𑁆𑀢𑀼𑀦𑁆𑀢𑀺.
