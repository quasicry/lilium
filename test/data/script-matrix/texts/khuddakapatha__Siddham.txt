𑗃 𑖡𑖦𑖺 𑖝𑖭𑖿𑖭 𑖥𑖐𑖪𑖝𑖺 𑖀𑖨𑖮𑖝𑖺 𑖭𑖦𑖿𑖦𑖯𑖭𑖦𑖿𑖤𑖲𑖟𑖿𑖠𑖭𑖿𑖭 𑗃
𑖏𑖲𑖟𑖿𑖟𑖎𑖡𑖰𑖎𑖯𑖧𑖸
𑖏𑖲𑖟𑖿𑖟𑖎𑖢𑖯𑖙𑖢𑖯𑖩𑗀𑖰
1. 𑖭𑖨𑖜𑖝𑖿𑖝𑖧𑖽
𑖤𑖲𑖟𑖿𑖠𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖠𑖦𑖿𑖦𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖭𑖒𑖿𑖑𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗃
𑖟𑖲𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖤𑖲𑖟𑖿𑖠𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖟𑖲𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖠𑖦𑖿𑖦𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖟𑖲𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖭𑖒𑖿𑖑𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗃
𑖝𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖤𑖲𑖟𑖿𑖠𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖝𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖠𑖦𑖿𑖦𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗂
𑖝𑖝𑖰𑖧𑖦𑖿𑖢𑖰 𑖭𑖒𑖿𑖑𑖽 𑖭𑖨𑖜𑖽 𑖐𑖓𑖿𑖔𑖯𑖦𑖰𑗃
𑖭𑖨𑖜𑖝𑖿𑖝𑖧𑖽 𑖡𑖰𑖘𑖿𑖙𑖰𑖝𑖽𑗂
2. 𑖟𑖭𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽
1. 𑖢𑖯𑖜𑖯𑖝𑖰𑖢𑖯𑖝𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
2. 𑖀𑖟𑖰𑖡𑖿𑖡𑖯𑖟𑖯𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
3. 𑖀𑖤𑖿𑖨𑖮𑖿𑖦𑖓𑖨𑖰𑖧𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
4. 𑖦𑖲𑖭𑖯𑖪𑖯𑖟𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
5. 𑖭𑖲𑖨𑖯𑖦𑖸𑖨𑖧𑖦𑖕𑖿𑖕𑖢𑖦𑖯𑖟𑖘𑖿𑖙𑖯𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
6. 𑖪𑖰𑖎𑖯𑖩𑖥𑖺𑖕𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
7. 𑖡𑖓𑖿𑖓-𑖐𑖱𑖝-𑖪𑖯𑖟𑖰𑖝-𑖪𑖰𑖭𑖳𑖎𑖟𑖭𑖿𑖭𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
8. 𑖦𑖯𑖩𑖯-𑖐𑖡𑖿𑖠-𑖪𑖰𑖩𑖸𑖢𑖡-𑖠𑖯𑖨𑖜-𑖦𑖜𑖿𑖚𑖡-𑖪𑖰𑖥𑖳𑖭𑖡𑖘𑖿𑖙𑖯𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
9. 𑖄𑖓𑖿𑖓𑖯𑖭𑖧𑖡-𑖦𑖮𑖯𑖭𑖧𑖡𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
10. 𑖕𑖯𑖝𑖨𑖳𑖢-𑖨𑖕𑖝𑖢𑖘𑖰𑖐𑖿𑖐𑖮𑖜𑖯 𑖪𑖸𑖨𑖦𑖜𑖱-𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖭𑖦𑖯𑖟𑖰𑖧𑖯𑖦𑖰𑗂
𑖟𑖭𑖭𑖰𑖎𑖿𑖏𑖯𑖢𑖟𑖽 𑖡𑖰𑖘𑖿𑖙𑖰𑖝𑖽𑗂
3. 𑖟𑖿𑖪𑖝𑖿𑖝𑖰𑖽𑖭𑖯𑖎𑖯𑖨𑖺
𑖀𑖝𑖿𑖞𑖰 𑖂𑖦𑖭𑖿𑖦𑖰𑖽 𑖎𑖯𑖧𑖸 –
𑖎𑖸𑖭𑖯 𑖩𑖺𑖦𑖯 𑖡𑖏𑖯 𑖟𑖡𑖿𑖝𑖯 𑖝𑖓𑖺,
𑖦𑖽𑖭𑖽 𑖡𑖿𑖮𑖯𑖨𑖲 𑖀𑖘𑖿𑖙𑖰 𑖀𑖘𑖿𑖙𑖰𑖦𑖰𑖗𑖿𑖕𑖽 𑖪𑖎𑖿𑖎𑖽,
𑖮𑖟𑖧𑖽 𑖧𑖎𑖡𑖽 𑖎𑖰𑖩𑖺𑖦𑖎𑖽 𑖢𑖰𑖮𑖎𑖽 𑖢𑖢𑖿𑖣𑖯𑖭𑖽,
𑖀𑖡𑖿𑖝𑖽 𑖀𑖡𑖿𑖝𑖐𑖲𑖜𑖽 𑖄𑖟𑖨𑖰𑖧𑖽 𑖎𑖨𑖱𑖭𑖽 𑖦𑖝𑖿𑖞𑖩𑖲𑖒𑖿𑖐𑖽,
𑖢𑖰𑖝𑖿𑖝𑖽 𑖭𑖸𑖦𑖿𑖮𑖽 𑖢𑖲𑖤𑖿𑖤𑖺 𑖩𑖺𑖮𑖰𑖝𑖽 𑖭𑖸𑖟𑖺 𑖦𑖸𑖟𑖺,
𑖀𑖭𑖿𑖭𑖲 𑖪𑖭𑖯 𑖏𑖸𑖩𑗀𑖺 𑖭𑖰𑖒𑖿𑖑𑖯𑖜𑖰𑖎𑖯 𑖩𑖭𑖰𑖎𑖯 𑖦𑖲𑖝𑖿𑖝𑖡𑖿𑖝𑖰 𑗃
𑖟𑖿𑖪𑖝𑖿𑖝𑖰𑖽𑖭𑖯𑖎𑖯𑖨𑖺 𑖡𑖰𑖘𑖿𑖙𑖰𑖝𑖺𑗂
4. 𑖎𑖲𑖦𑖯𑖨𑖢𑖗𑖿𑖮𑖯
1. ‘‘𑖊𑖎𑖽 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖭𑖤𑖿𑖤𑖸 𑖭𑖝𑖿𑖝𑖯 𑖁𑖮𑖯𑖨𑖘𑖿𑖙𑖰𑖝𑖰𑖎𑖯’’𑗂
2. ‘‘𑖟𑖿𑖪𑖸 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖡𑖯𑖦𑖗𑖿𑖓 𑖨𑖳𑖢𑖗𑖿𑖓’’𑗂
3. ‘‘𑖝𑖱𑖜𑖰 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖝𑖰𑖭𑖿𑖭𑖺 𑖪𑖸𑖟𑖡𑖯’’𑗂
4. ‘‘𑖓𑖝𑖿𑖝𑖯𑖨𑖰 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖓𑖝𑖿𑖝𑖯𑖨𑖰 𑖀𑖨𑖰𑖧𑖭𑖓𑖿𑖓𑖯𑖡𑖰’’𑗂
5. ‘‘𑖢𑖗𑖿𑖓 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖢𑖗𑖿𑖓𑖲𑖢𑖯𑖟𑖯𑖡𑖎𑖿𑖏𑖡𑖿𑖠𑖯’’𑗂
6. ‘‘𑖔 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖔 𑖀𑖕𑖿𑖖𑖝𑖿𑖝𑖰𑖎𑖯𑖡𑖰 𑖁𑖧𑖝𑖡𑖯𑖡𑖰’’𑗂
7. ‘‘𑖭𑖝𑖿𑖝 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖭𑖝𑖿𑖝 𑖤𑖺𑖕𑖿𑖖𑖒𑖿𑖐𑖯’’𑗂
8. ‘‘𑖀𑖘𑖿𑖙 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖀𑖨𑖰𑖧𑖺 𑖀𑖘𑖿𑖙𑖒𑖿𑖐𑖰𑖎𑖺 𑖦𑖐𑖿𑖐𑖺’’𑗂
9. ‘‘𑖡𑖪 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖡𑖪 𑖭𑖝𑖿𑖝𑖯𑖪𑖯𑖭𑖯’’𑗂
10. ‘‘𑖟𑖭 𑖡𑖯𑖦 𑖎𑖰𑖽’’? ‘‘𑖟𑖭𑖮𑖒𑖿𑖐𑖸𑖮𑖰 𑖭𑖦𑖡𑖿𑖡𑖯𑖐𑖝𑖺 ‘𑖀𑖨𑖮𑖯’𑖝𑖰 𑖪𑖲𑖓𑖿𑖓𑖝𑖱’’𑖝𑖰𑗂
𑖎𑖲𑖦𑖯𑖨𑖢𑖗𑖿𑖮𑖯 𑖡𑖰𑖘𑖿𑖙𑖰𑖝𑖯𑗂
5. 𑖦𑖒𑖿𑖐𑖩𑖭𑖲𑖝𑖿𑖝𑖽
1. 𑖊𑖪𑖽 𑖦𑖸 𑖭𑖲𑖝𑖽 – 𑖊𑖎𑖽 𑖭𑖦𑖧𑖽 𑖥𑖐𑖪𑖯 𑖭𑖯𑖪𑖝𑖿𑖞𑖰𑖧𑖽 𑖪𑖰𑖮𑖨𑖝𑖰 𑖕𑖸𑖝𑖪𑖡𑖸 𑖀𑖡𑖯𑖞𑖢𑖰𑖜𑖿𑖚𑖰𑖎𑖭𑖿𑖭 𑖁𑖨𑖯𑖦𑖸𑗂 𑖀𑖞 𑖏𑖺 𑖀𑖗𑖿𑖗𑖝𑖨𑖯 𑖟𑖸𑖪𑖝𑖯 𑖀𑖥𑖰𑖎𑖿𑖎𑖡𑖿𑖝𑖯𑖧 𑖨𑖝𑖿𑖝𑖰𑖧𑖯 𑖀𑖥𑖰𑖎𑖿𑖎𑖡𑖿𑖝𑖪𑖜𑖿𑖜𑖯 𑖎𑖸𑖪𑖩𑖎𑖢𑖿𑖢𑖽 𑖕𑖸𑖝𑖪𑖡𑖽 𑖌𑖥𑖯𑖭𑖸𑖝𑖿𑖪𑖯 𑖧𑖸𑖡 𑖥𑖐𑖪𑖯 𑖝𑖸𑖡𑖲𑖢𑖭𑖒𑖿𑖎𑖦𑖰; 𑖄𑖢𑖭𑖒𑖿𑖎𑖦𑖰𑖝𑖿𑖪𑖯 𑖥𑖐𑖪𑖡𑖿𑖝𑖽 𑖀𑖥𑖰𑖪𑖯𑖟𑖸𑖝𑖿𑖪𑖯 𑖊𑖎𑖦𑖡𑖿𑖝𑖽 𑖀𑖘𑖿𑖙𑖯𑖭𑖰𑗂 𑖊𑖎𑖦𑖡𑖿𑖝𑖽 𑖙𑖰𑖝𑖯 𑖏𑖺 𑖭𑖯 𑖟𑖸𑖪𑖝𑖯 𑖥𑖐𑖪𑖡𑖿𑖝𑖽 𑖐𑖯𑖞𑖯𑖧 𑖀𑖕𑖿𑖖𑖥𑖯𑖭𑖰 –
2. ‘‘𑖤𑖮𑖳 𑖟𑖸𑖪𑖯 𑖦𑖡𑖲𑖭𑖿𑖭𑖯 𑖓, 𑖦𑖒𑖿𑖐𑖩𑖯𑖡𑖰 𑖀𑖓𑖰𑖡𑖿𑖝𑖧𑖲𑖽𑗂
𑖁𑖎𑖒𑖿𑖏𑖦𑖯𑖡𑖯 𑖭𑖺𑖝𑖿𑖞𑖯𑖡𑖽, 𑖤𑖿𑖨𑖳𑖮𑖰 𑖦𑖒𑖿𑖐𑖩𑖦𑖲𑖝𑖿𑖝𑖦𑖽’’𑗃
3. ‘‘𑖀𑖭𑖸𑖪𑖡𑖯 𑖓 𑖤𑖯𑖩𑖯𑖡𑖽, 𑖢𑖜𑖿𑖚𑖰𑖝𑖯𑖡𑖗𑖿𑖓 𑖭𑖸𑖪𑖡𑖯𑗂
𑖢𑖳𑖕𑖯 𑖓 𑖢𑖳𑖕𑖡𑖸𑖧𑖿𑖧𑖯𑖡𑖽 , 𑖊𑖝𑖽 𑖦𑖒𑖿𑖐𑖩𑖦𑖲𑖝𑖿𑖝𑖦𑖽𑗃
4. ‘‘𑖢𑖝𑖰𑖨𑖳𑖢𑖟𑖸𑖭𑖪𑖯𑖭𑖺 𑖓, 𑖢𑖲𑖤𑖿𑖤𑖸 𑖓 𑖎𑖝𑖢𑖲𑖗𑖿𑖗𑖝𑖯𑗂
𑖀𑖝𑖿𑖝𑖭𑖦𑖿𑖦𑖯𑖢𑖜𑖰𑖠𑖰 𑖓, 𑖊𑖝𑖽 𑖦𑖒𑖿𑖐𑖩𑖦𑖲𑖝𑖿𑖝𑖦𑖽𑗃
5. ‘‘𑖤𑖯𑖮𑖲𑖭𑖓𑖿𑖓𑖗𑖿𑖓 𑖭𑖰𑖢𑖿𑖢𑖗𑖿𑖓, 𑖪𑖰𑖡𑖧𑖺 𑖓 𑖭𑖲𑖭𑖰𑖎𑖿𑖏𑖰𑖝𑖺𑗂
𑖭𑖲𑖥𑖯𑖭𑖰𑖝𑖯 𑖓 𑖧𑖯 𑖪𑖯𑖓𑖯, 𑖊𑖝𑖽 𑖦𑖒𑖿𑖐𑖩𑖦𑖲𑖝𑖿𑖝𑖦𑖽𑗃
6. ‘‘𑖦𑖯𑖝𑖯𑖢𑖰𑖝𑖲 𑖄𑖢𑖘𑖿𑖙𑖯𑖡𑖽, 𑖢𑖲𑖝𑖿𑖝𑖟𑖯𑖨𑖭𑖿𑖭 𑖭𑖒𑖿𑖐𑖮𑖺𑗂
𑖀𑖡𑖯𑖎𑖲𑖩𑖯 𑖓 𑖎𑖦𑖿𑖦𑖡𑖿𑖝𑖯, 𑖊𑖝𑖽 𑖦𑖒𑖿𑖐𑖩𑖦𑖲𑖝𑖿𑖝𑖦𑖽𑗃
7. ‘‘𑖟𑖯𑖡𑖗𑖿𑖓 𑖠𑖦𑖿𑖦𑖓𑖨𑖰𑖧𑖯 𑖓, 𑖗𑖯𑖝𑖎𑖯𑖡𑖗𑖿𑖓 𑖭𑖒𑖿𑖐𑖮𑖺𑗂

𑖦𑖒𑖿𑖐𑖩𑖭𑖲𑖝𑖿𑖝𑖽 𑖡𑖰𑖘𑖿𑖙𑖰𑖝𑖽𑗂
6. 𑖨𑖝𑖡𑖭𑖲𑖝𑖿𑖝𑖽
1. 𑖧𑖯𑖡𑖱𑖠 𑖥𑖳𑖝𑖯𑖡𑖰 𑖭𑖦𑖯𑖐𑖝𑖯𑖡𑖰, 𑖥𑖲𑖦𑖿𑖦𑖯𑖡𑖰 𑖪𑖯 𑖧𑖯𑖡𑖰 𑖪 𑖀𑖡𑖿𑖝𑖩𑖰𑖎𑖿𑖏𑖸𑗂
𑖭𑖤𑖿𑖤𑖸𑖪 𑖥𑖳𑖝𑖯 𑖭𑖲𑖦𑖡𑖯 𑖥𑖪𑖡𑖿𑖝𑖲, 𑖀𑖞𑖺𑖢𑖰 𑖭𑖎𑖿𑖎𑖓𑖿𑖓 𑖭𑖲𑖜𑖡𑖿𑖝𑖲 𑖥𑖯𑖭𑖰𑖝𑖽𑗃
2. 𑖝𑖭𑖿𑖦𑖯 𑖮𑖰 𑖥𑖳𑖝𑖯 𑖡𑖰𑖭𑖯𑖦𑖸𑖞 𑖭𑖤𑖿𑖤𑖸, 𑖦𑖸𑖝𑖿𑖝𑖽 𑖎𑖨𑖺𑖞 𑖦𑖯𑖡𑖲𑖭𑖰𑖧𑖯 𑖢𑖕𑖯𑖧𑗂
𑖟𑖰𑖪𑖯 𑖓 𑖨𑖝𑖿𑖝𑖺 𑖓 𑖮𑖨𑖡𑖿𑖝𑖰 𑖧𑖸 𑖤𑖩𑖰𑖽, 𑖝𑖭𑖿𑖦𑖯 𑖮𑖰 𑖡𑖸 𑖨𑖎𑖿𑖏𑖞 𑖀𑖢𑖿𑖢𑖦𑖝𑖿𑖝𑖯𑗃
3. 𑖧𑖽 𑖎𑖰𑖗𑖿𑖓𑖰 𑖪𑖰𑖝𑖿𑖝𑖽 𑖂𑖠 𑖪𑖯 𑖮𑖲𑖨𑖽 𑖪𑖯, 𑖭𑖐𑖿𑖐𑖸𑖭𑖲 𑖪𑖯 𑖧𑖽 𑖨𑖝𑖡𑖽 𑖢𑖜𑖱𑖝𑖽𑗂
𑖡 𑖡𑖺 𑖭𑖦𑖽 𑖀𑖝𑖿𑖞𑖰 𑖝𑖞𑖯𑖐𑖝𑖸𑖡, 𑖂𑖟𑖦𑖿𑖢𑖰 𑖤𑖲𑖟𑖿𑖠𑖸 𑖨𑖝𑖡𑖽 𑖢𑖜𑖱𑖝𑖽𑗂
𑖊𑖝𑖸𑖡 𑖭𑖓𑖿𑖓𑖸𑖡 𑖭𑖲𑖪𑖝𑖿𑖞𑖰 𑖮𑖺𑖝𑖲𑗃
4. 𑖏𑖧𑖽 𑖪𑖰𑖨𑖯𑖐𑖽 𑖀𑖦𑖝𑖽 𑖢𑖜𑖱𑖝𑖽, 𑖧𑖟𑖕𑖿𑖖𑖐𑖯 𑖭𑖎𑖿𑖧𑖦𑖲𑖡𑖱 𑖭𑖦𑖯𑖮𑖰𑖝𑖺𑗂
