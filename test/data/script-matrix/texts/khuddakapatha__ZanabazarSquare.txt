𑩃 𑨝𑨢𑨆 𑨙𑨰𑩇𑨰 𑨡𑨍𑨭𑨙𑨆 𑨀𑨫𑨱𑨙𑨆 𑨰𑨢𑩇𑨢𑨊𑨰𑨢𑩇𑨠𑨂𑨛𑩇𑨜𑨰𑩇𑨰 𑩃
𑨌𑨂𑨛𑩇𑨛𑨋𑨝𑨁𑨋𑨊𑨪𑨄
𑨌𑨂𑨛𑩇𑨛𑨋𑨞𑨊𑨕𑨞𑨊𑨬𑨁
1. 𑨰𑨫𑨘𑨙𑩇𑨙𑨪𑨸
𑨠𑨂𑨛𑩇𑨜𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨜𑨢𑩇𑨢𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨰𑨏𑩇𑨎𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩃
𑨛𑨂𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨠𑨂𑨛𑩇𑨜𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨛𑨂𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨜𑨢𑩇𑨢𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨛𑨂𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩃
𑨙𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨠𑨂𑨛𑩇𑨜𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨙𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨜𑨢𑩇𑨢𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩂
𑨙𑨙𑨁𑨪𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨸 𑨰𑨫𑨘𑨸 𑨍𑨣𑩇𑨤𑨊𑨢𑨁𑩃
𑨰𑨫𑨘𑨙𑩇𑨙𑨪𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
2. 𑨛𑨰𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸
1. 𑨞𑨊𑨘𑨊𑨙𑨁𑨞𑨊𑨙𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
2. 𑨀𑨛𑨁𑨝𑩇𑨝𑨊𑨛𑨊𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
3. 𑨀𑨠𑩇𑨫𑨱𑩇𑨢𑨣𑨫𑨁𑨪𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
4. 𑨢𑨂𑨰𑨊𑨭𑨊𑨛𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
5. 𑨰𑨂𑨫𑨊𑨢𑨄𑨫𑨪𑨢𑨥𑩇𑨥𑨞𑨢𑨊𑨛𑨔𑩇𑨕𑨊𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
6. 𑨭𑨁𑨋𑨊𑨬𑨡𑨆𑨥𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
7. 𑨝𑨣𑩇𑨣-𑨍𑨁𑨊𑨙-𑨭𑨊𑨛𑨁𑨙-𑨭𑨁𑨰𑨂𑨊𑨋𑨛𑨰𑩇𑨰𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
8. 𑨢𑨊𑨬𑨊-𑨍𑨝𑩇𑨜-𑨭𑨁𑨬𑨄𑨞𑨝-𑨜𑨊𑨫𑨘-𑨢𑨘𑩇𑨖𑨝-𑨭𑨁𑨡𑨂𑨊𑨰𑨝𑨔𑩇𑨕𑨊𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
9. 𑨀𑨂𑨣𑩇𑨣𑨊𑨰𑨪𑨝-𑨢𑨱𑨊𑨰𑨪𑨝𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
10. 𑨥𑨊𑨙𑨫𑨂𑨊𑨞-𑨫𑨥𑨙𑨞𑨔𑨁𑨍𑩇𑨍𑨱𑨘𑨊 𑨭𑨄𑨫𑨢𑨘𑨁𑨊-𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨰𑨢𑨊𑨛𑨁𑨪𑨊𑨢𑨁𑩂
𑨛𑨰𑨰𑨁𑨋𑩇𑨌𑨊𑨞𑨛𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
3. 𑨛𑩇𑨭𑨙𑩇𑨙𑨁𑨸𑨰𑨊𑨋𑨊𑨫𑨆
𑨀𑨙𑩇𑨚𑨁 𑨀𑨁𑨢𑨰𑩇𑨢𑨁𑨸 𑨋𑨊𑨪𑨄 –
𑨋𑨄𑨰𑨊 𑨬𑨆𑨢𑨊 𑨝𑨌𑨊 𑨛𑨝𑩇𑨙𑨊 𑨙𑨣𑨆,
𑨢𑨸𑨰𑨸 𑨝𑩇𑨱𑨊𑨫𑨂 𑨀𑨔𑩇𑨕𑨁 𑨀𑨔𑩇𑨕𑨁𑨢𑨁𑨓𑩇𑨥𑨸 𑨭𑨋𑩇𑨋𑨸,
𑨱𑨛𑨪𑨸 𑨪𑨋𑨝𑨸 𑨋𑨁𑨬𑨆𑨢𑨋𑨸 𑨞𑨁𑨱𑨋𑨸 𑨞𑨞𑩇𑨟𑨊𑨰𑨸,
𑨀𑨝𑩇𑨙𑨸 𑨀𑨝𑩇𑨙𑨍𑨂𑨘𑨸 𑨀𑨂𑨛𑨫𑨁𑨪𑨸 𑨋𑨫𑨁𑨊𑨰𑨸 𑨢𑨙𑩇𑨚𑨬𑨂𑨏𑩇𑨍𑨸,
𑨞𑨁𑨙𑩇𑨙𑨸 𑨰𑨄𑨢𑩇𑨱𑨸 𑨞𑨂𑨠𑩇𑨠𑨆 𑨬𑨆𑨱𑨁𑨙𑨸 𑨰𑨄𑨛𑨆 𑨢𑨄𑨛𑨆,
𑨀𑨰𑩇𑨰𑨂 𑨭𑨰𑨊 𑨌𑨄𑨬𑨆 𑨰𑨁𑨏𑩇𑨎𑨊𑨘𑨁𑨋𑨊 𑨬𑨰𑨁𑨋𑨊 𑨢𑨂𑨙𑩇𑨙𑨝𑩇𑨙𑨁 𑩃
𑨛𑩇𑨭𑨙𑩇𑨙𑨁𑨸𑨰𑨊𑨋𑨊𑨫𑨆 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨆𑩂
4. 𑨋𑨂𑨢𑨊𑨫𑨞𑨓𑩇𑨱𑨊
1. ‘‘𑨀𑨄𑨋𑨸 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨰𑨠𑩇𑨠𑨄 𑨰𑨙𑩇𑨙𑨊 𑨀𑨊𑨱𑨊𑨫𑨔𑩇𑨕𑨁𑨙𑨁𑨋𑨊’’𑩂
2. ‘‘𑨛𑩇𑨭𑨄 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨝𑨊𑨢𑨓𑩇𑨣 𑨫𑨂𑨊𑨞𑨓𑩇𑨣’’𑩂
3. ‘‘𑨙𑨁𑨊𑨘𑨁 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨙𑨁𑨰𑩇𑨰𑨆 𑨭𑨄𑨛𑨝𑨊’’𑩂
4. ‘‘𑨣𑨙𑩇𑨙𑨊𑨫𑨁 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨣𑨙𑩇𑨙𑨊𑨫𑨁 𑨀𑨫𑨁𑨪𑨰𑨣𑩇𑨣𑨊𑨝𑨁’’𑩂
5. ‘‘𑨞𑨓𑩇𑨣 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨞𑨓𑩇𑨣𑨂𑨞𑨊𑨛𑨊𑨝𑨋𑩇𑨌𑨝𑩇𑨜𑨊’’𑩂
6. ‘‘𑨤 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨤 𑨀𑨥𑩇𑨦𑨙𑩇𑨙𑨁𑨋𑨊𑨝𑨁 𑨀𑨊𑨪𑨙𑨝𑨊𑨝𑨁’’𑩂
7. ‘‘𑨰𑨙𑩇𑨙 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨰𑨙𑩇𑨙 𑨠𑨆𑨥𑩇𑨦𑨏𑩇𑨍𑨊’’𑩂
8. ‘‘𑨀𑨔𑩇𑨕 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨀𑨫𑨁𑨪𑨆 𑨀𑨔𑩇𑨕𑨏𑩇𑨍𑨁𑨋𑨆 𑨢𑨍𑩇𑨍𑨆’’𑩂
9. ‘‘𑨝𑨭 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨝𑨭 𑨰𑨙𑩇𑨙𑨊𑨭𑨊𑨰𑨊’’𑩂
10. ‘‘𑨛𑨰 𑨝𑨊𑨢 𑨋𑨁𑨸’’? ‘‘𑨛𑨰𑨱𑨏𑩇𑨍𑨄𑨱𑨁 𑨰𑨢𑨝𑩇𑨝𑨊𑨍𑨙𑨆 ‘𑨀𑨫𑨱𑨊’𑨙𑨁 𑨭𑨂𑨣𑩇𑨣𑨙𑨁𑨊’’𑨙𑨁𑩂
𑨋𑨂𑨢𑨊𑨫𑨞𑨓𑩇𑨱𑨊 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨊𑩂
5. 𑨢𑨏𑩇𑨍𑨬𑨰𑨂𑨙𑩇𑨙𑨸
1. 𑨀𑨄𑨭𑨸 𑨢𑨄 𑨰𑨂𑨙𑨸 – 𑨀𑨄𑨋𑨸 𑨰𑨢𑨪𑨸 𑨡𑨍𑨭𑨊 𑨰𑨊𑨭𑨙𑩇𑨚𑨁𑨪𑨸 𑨭𑨁𑨱𑨫𑨙𑨁 𑨥𑨄𑨙𑨭𑨝𑨄 𑨀𑨝𑨊𑨚𑨞𑨁𑨘𑩇𑨖𑨁𑨋𑨰𑩇𑨰 𑨀𑨊𑨫𑨊𑨢𑨄𑩂 𑨀𑨚 𑨌𑨆 𑨀𑨓𑩇𑨓𑨙𑨫𑨊 𑨛𑨄𑨭𑨙𑨊 𑨀𑨡𑨁𑨋𑩇𑨋𑨝𑩇𑨙𑨊𑨪 𑨫𑨙𑩇𑨙𑨁𑨪𑨊 𑨀𑨡𑨁𑨋𑩇𑨋𑨝𑩇𑨙𑨭𑨘𑩇𑨘𑨊 𑨋𑨄𑨭𑨬𑨋𑨞𑩇𑨞𑨸 𑨥𑨄𑨙𑨭𑨝𑨸 𑨀𑨆𑨡𑨊𑨰𑨄𑨙𑩇𑨭𑨊 𑨪𑨄𑨝 𑨡𑨍𑨭𑨊 𑨙𑨄𑨝𑨂𑨞𑨰𑨏𑩇𑨋𑨢𑨁; 𑨀𑨂𑨞𑨰𑨏𑩇𑨋𑨢𑨁𑨙𑩇𑨭𑨊 𑨡𑨍𑨭𑨝𑩇𑨙𑨸 𑨀𑨡𑨁𑨭𑨊𑨛𑨄𑨙𑩇𑨭𑨊 𑨀𑨄𑨋𑨢𑨝𑩇𑨙𑨸 𑨀𑨔𑩇𑨕𑨊𑨰𑨁𑩂 𑨀𑨄𑨋𑨢𑨝𑩇𑨙𑨸 𑨕𑨁𑨙𑨊 𑨌𑨆 𑨰𑨊 𑨛𑨄𑨭𑨙𑨊 𑨡𑨍𑨭𑨝𑩇𑨙𑨸 𑨍𑨊𑨚𑨊𑨪 𑨀𑨥𑩇𑨦𑨡𑨊𑨰𑨁 –
2. ‘‘𑨠𑨱𑨂𑨊 𑨛𑨄𑨭𑨊 𑨢𑨝𑨂𑨰𑩇𑨰𑨊 𑨣, 𑨢𑨏𑩇𑨍𑨬𑨊𑨝𑨁 𑨀𑨣𑨁𑨝𑩇𑨙𑨪𑨂𑨸𑩂
𑨀𑨊𑨋𑨏𑩇𑨌𑨢𑨊𑨝𑨊 𑨰𑨆𑨙𑩇𑨚𑨊𑨝𑨸, 𑨠𑩇𑨫𑨂𑨊𑨱𑨁 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸’’𑩃
3. ‘‘𑨀𑨰𑨄𑨭𑨝𑨊 𑨣 𑨠𑨊𑨬𑨊𑨝𑨸, 𑨞𑨘𑩇𑨖𑨁𑨙𑨊𑨝𑨓𑩇𑨣 𑨰𑨄𑨭𑨝𑨊𑩂
𑨞𑨂𑨊𑨥𑨊 𑨣 𑨞𑨂𑨊𑨥𑨝𑨄𑨪𑩇𑨪𑨊𑨝𑨸 , 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
4. ‘‘𑨞𑨙𑨁𑨫𑨂𑨊𑨞𑨛𑨄𑨰𑨭𑨊𑨰𑨆 𑨣, 𑨞𑨂𑨠𑩇𑨠𑨄 𑨣 𑨋𑨙𑨞𑨂𑨓𑩇𑨓𑨙𑨊𑩂
𑨀𑨙𑩇𑨙𑨰𑨢𑩇𑨢𑨊𑨞𑨘𑨁𑨜𑨁 𑨣, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
5. ‘‘𑨠𑨊𑨱𑨂𑨰𑨣𑩇𑨣𑨓𑩇𑨣 𑨰𑨁𑨞𑩇𑨞𑨓𑩇𑨣, 𑨭𑨁𑨝𑨪𑨆 𑨣 𑨰𑨂𑨰𑨁𑨋𑩇𑨌𑨁𑨙𑨆𑩂
𑨰𑨂𑨡𑨊𑨰𑨁𑨙𑨊 𑨣 𑨪𑨊 𑨭𑨊𑨣𑨊, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
6. ‘‘𑨢𑨊𑨙𑨊𑨞𑨁𑨙𑨂 𑨀𑨂𑨞𑨔𑩇𑨕𑨊𑨝𑨸, 𑨞𑨂𑨙𑩇𑨙𑨛𑨊𑨫𑨰𑩇𑨰 𑨰𑨏𑩇𑨍𑨱𑨆𑩂
𑨀𑨝𑨊𑨋𑨂𑨬𑨊 𑨣 𑨋𑨢𑩇𑨢𑨝𑩇𑨙𑨊, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
7. ‘‘𑨛𑨊𑨝𑨓𑩇𑨣 𑨜𑨢𑩇𑨢𑨣𑨫𑨁𑨪𑨊 𑨣, 𑨓𑨊𑨙𑨋𑨊𑨝𑨓𑩇𑨣 𑨰𑨏𑩇𑨍𑨱𑨆𑩂
𑨀𑨝𑨭𑨥𑩇𑨥𑨊𑨝𑨁 𑨋𑨢𑩇𑨢𑨊𑨝𑨁, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
8. ‘‘𑨀𑨊𑨫𑨙𑨁𑨊 𑨭𑨁𑨫𑨙𑨁𑨊 𑨞𑨊𑨞𑨊, 𑨢𑨥𑩇𑨥𑨞𑨊𑨝𑨊 𑨣 𑨰𑨸𑨪𑨢𑨆𑩂
𑨀𑨞𑩇𑨞𑨢𑨊𑨛𑨆 𑨣 𑨜𑨢𑩇𑨢𑨄𑨰𑨂, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
9. ‘‘𑨍𑨊𑨫𑨭𑨆 𑨣 𑨝𑨁𑨭𑨊𑨙𑨆 𑨣, 𑨰𑨝𑩇𑨙𑨂𑨔𑩇𑨕𑨁 𑨣 𑨋𑨙𑨓𑩇𑨓𑨂𑨙𑨊𑩂
𑨋𑨊𑨬𑨄𑨝 𑨜𑨢𑩇𑨢𑨰𑩇𑨰𑨭𑨝𑨸, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
10. ‘‘𑨌𑨝𑩇𑨙𑨁𑨊 𑨣 𑨰𑨆𑨭𑨣𑨰𑩇𑨰𑨙𑨊, 𑨰𑨢𑨘𑨊𑨝𑨓𑩇𑨣 𑨛𑨰𑩇𑨰𑨝𑨸𑩂
𑨋𑨊𑨬𑨄𑨝 𑨜𑨢𑩇𑨢𑨰𑨊𑨋𑨣𑩇𑨤𑨊, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
11. ‘‘𑨙𑨞𑨆 𑨣 𑨠𑩇𑨫𑨱𑩇𑨢𑨣𑨫𑨁𑨪𑨓𑩇𑨣, 𑨀𑨫𑨁𑨪𑨰𑨣𑩇𑨣𑨊𑨝 𑨛𑨰𑩇𑨰𑨝𑨸𑩂
𑨝𑨁𑨠𑩇𑨠𑨊𑨝𑨰𑨣𑩇𑨤𑨁𑨋𑨁𑨫𑨁𑨪𑨊 𑨣, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
12. ‘‘𑨟𑨂𑨔𑩇𑨕𑨰𑩇𑨰 𑨬𑨆𑨋𑨜𑨢𑩇𑨢𑨄𑨱𑨁, 𑨣𑨁𑨙𑩇𑨙𑨸 𑨪𑨰𑩇𑨰 𑨝 𑨋𑨢𑩇𑨞𑨙𑨁𑩂
𑨀𑨰𑨆𑨋𑨸 𑨭𑨁𑨫𑨥𑨸 𑨌𑨄𑨢𑨸, 𑨀𑨄𑨙𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢𑨸𑩃
13. ‘‘𑨀𑨄𑨙𑨊𑨛𑨁𑨰𑨊𑨝𑨁 𑨋𑨙𑩇𑨭𑨊𑨝, 𑨰𑨠𑩇𑨠𑨙𑩇𑨚𑨢𑨞𑨫𑨊𑨥𑨁𑨙𑨊𑩂
𑨰𑨠𑩇𑨠𑨙𑩇𑨚 𑨰𑨆𑨙𑩇𑨚𑨁𑨸 𑨍𑨣𑩇𑨤𑨝𑩇𑨙𑨁, 𑨙𑨸 𑨙𑨄𑨰𑨸 𑨢𑨏𑩇𑨍𑨬𑨢𑨂𑨙𑩇𑨙𑨢’’𑨝𑩇𑨙𑨁𑩃
𑨢𑨏𑩇𑨍𑨬𑨰𑨂𑨙𑩇𑨙𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
6. 𑨫𑨙𑨝𑨰𑨂𑨙𑩇𑨙𑨸
1. 𑨪𑨊𑨝𑨁𑨊𑨜 𑨡𑨂𑨊𑨙𑨊𑨝𑨁 𑨰𑨢𑨊𑨍𑨙𑨊𑨝𑨁, 𑨡𑨂𑨢𑩇𑨢𑨊𑨝𑨁 𑨭𑨊 𑨪𑨊𑨝𑨁 𑨭 𑨀𑨝𑩇𑨙𑨬𑨁𑨋𑩇𑨌𑨄𑩂
𑨰𑨠𑩇𑨠𑨄𑨭 𑨡𑨂𑨊𑨙𑨊 𑨰𑨂𑨢𑨝𑨊 𑨡𑨭𑨝𑩇𑨙𑨂, 𑨀𑨚𑨆𑨞𑨁 𑨰𑨋𑩇𑨋𑨣𑩇𑨣 𑨰𑨂𑨘𑨝𑩇𑨙𑨂 𑨡𑨊𑨰𑨁𑨙𑨸𑩃
2. 𑨙𑨰𑩇𑨢𑨊 𑨱𑨁 𑨡𑨂𑨊𑨙𑨊 𑨝𑨁𑨰𑨊𑨢𑨄𑨚 𑨰𑨠𑩇𑨠𑨄, 𑨢𑨄𑨙𑩇𑨙𑨸 𑨋𑨫𑨆𑨚 𑨢𑨊𑨝𑨂𑨰𑨁𑨪𑨊 𑨞𑨥𑨊𑨪𑩂
𑨛𑨁𑨭𑨊 𑨣 𑨫𑨙𑩇𑨙𑨆 𑨣 𑨱𑨫𑨝𑩇𑨙𑨁 𑨪𑨄 𑨠𑨬𑨁𑨸, 𑨙𑨰𑩇𑨢𑨊 𑨱𑨁 𑨝𑨄 𑨫𑨋𑩇𑨌𑨚 𑨀𑨞𑩇𑨞𑨢𑨙𑩇𑨙𑨊𑩃
3. 𑨪𑨸 𑨋𑨁𑨓𑩇𑨣𑨁 𑨭𑨁𑨙𑩇𑨙𑨸 𑨀𑨁𑨜 𑨭𑨊 𑨱𑨂𑨫𑨸 𑨭𑨊, 𑨰𑨍𑩇𑨍𑨄𑨰𑨂 𑨭𑨊 𑨪𑨸 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸𑩂
𑨝 𑨝𑨆 𑨰𑨢𑨸 𑨀𑨙𑩇𑨚𑨁 𑨙𑨚𑨊𑨍𑨙𑨄𑨝, 𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨠𑨂𑨛𑩇𑨜𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸𑩂
𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
4. 𑨌𑨪𑨸 𑨭𑨁𑨫𑨊𑨍𑨸 𑨀𑨢𑨙𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨪𑨛𑨥𑩇𑨦𑨍𑨊 𑨰𑨋𑩇𑨪𑨢𑨂𑨝𑨁𑨊 𑨰𑨢𑨊𑨱𑨁𑨙𑨆𑩂
𑨝 𑨙𑨄𑨝 𑨜𑨢𑩇𑨢𑨄𑨝 𑨰𑨢𑨙𑩇𑨚𑨁 𑨋𑨁𑨓𑩇𑨣𑨁, 𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨜𑨢𑩇𑨢𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸𑩂
𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
5. 𑨪𑨸 𑨠𑨂𑨛𑩇𑨜𑨰𑨄𑨔𑩇𑨕𑨆 𑨞𑨫𑨁𑨭𑨘𑩇𑨘𑨪𑨁𑨊 𑨰𑨂𑨣𑨁𑨸, 𑨰𑨢𑨊𑨜𑨁𑨢𑨊𑨝𑨝𑩇𑨙𑨫𑨁𑨋𑨓𑩇𑨓𑨢𑨊𑨱𑨂𑩂
𑨰𑨢𑨊𑨜𑨁𑨝𑨊 𑨙𑨄𑨝 𑨰𑨢𑨆 𑨝 𑨭𑨁𑨥𑩇𑨥𑨙𑨁, 𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨜𑨢𑩇𑨢𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸𑩂
𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
6. 𑨪𑨄 𑨞𑨂𑨍𑩇𑨍𑨬𑨊 𑨀𑨔𑩇𑨕 𑨰𑨙𑨸 𑨞𑨰𑨙𑩇𑨚𑨊, 𑨣𑨙𑩇𑨙𑨊𑨫𑨁 𑨀𑨄𑨙𑨊𑨝𑨁 𑨪𑨂𑨍𑨊𑨝𑨁 𑨱𑨆𑨝𑩇𑨙𑨁𑩂
𑨙𑨄 𑨛𑨋𑩇𑨌𑨁𑨘𑨄𑨪𑩇𑨪𑨊 𑨰𑨂𑨍𑨙𑨰𑩇𑨰 𑨰𑨊𑨭𑨋𑨊, 𑨀𑨄𑨙𑨄𑨰𑨂 𑨛𑨁𑨝𑩇𑨝𑨊𑨝𑨁 𑨢𑨱𑨞𑩇𑨟𑨬𑨊𑨝𑨁𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
7. 𑨪𑨄 𑨰𑨂𑨞𑩇𑨞𑨪𑨂𑨙𑩇𑨙𑨊 𑨢𑨝𑨰𑨊 𑨛𑨬𑩇𑨱𑨄𑨝, 𑨝𑨁𑨋𑩇𑨋𑨊𑨢𑨁𑨝𑨆 𑨍𑨆𑨙𑨢𑨰𑨊𑨰𑨝𑨢𑩇𑨱𑨁𑩂
𑨙𑨄 𑨞𑨙𑩇𑨙𑨁𑨞𑨙𑩇𑨙𑨊 𑨀𑨢𑨙𑨸 𑨭𑨁𑨍𑨪𑩇𑨱, 𑨬𑨛𑩇𑨜𑨊 𑨢𑨂𑨜𑨊 𑨝𑨁𑨠𑩇𑨠𑨂𑨙𑨁𑨸 𑨡𑨂𑨓𑩇𑨥𑨢𑨊𑨝𑨊𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
8. 𑨪𑨚𑨁𑨝𑩇𑨛𑨌𑨁𑨊𑨬𑨆 𑨞𑨚𑨭𑨁𑨰𑩇𑨰𑨁𑨙𑨆 𑨰𑨁𑨪𑨊, 𑨣𑨙𑨂𑨠𑩇𑨡𑨁 𑨭𑨊𑨙𑨄𑨱𑨁 𑨀𑨰𑨢𑩇𑨞𑨋𑨢𑩇𑨞𑨁𑨪𑨆𑩂
𑨙𑨚𑨂𑨊𑨞𑨢𑨸 𑨰𑨞𑩇𑨞𑨂𑨫𑨁𑨰𑨸 𑨭𑨛𑨊𑨢𑨁, 𑨪𑨆 𑨀𑨫𑨁𑨪𑨰𑨣𑩇𑨣𑨊𑨝𑨁 𑨀𑨭𑨄𑨣𑩇𑨣 𑨞𑨰𑩇𑨰𑨙𑨁𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
9. 𑨪𑨄 𑨀𑨫𑨁𑨪𑨰𑨣𑩇𑨣𑨊𑨝𑨁 𑨭𑨁𑨡𑨊𑨭𑨪𑨝𑩇𑨙𑨁, 𑨍𑨢𑩇𑨡𑨁𑨊𑨫𑨞𑨓𑩇𑨓𑨄𑨝 𑨰𑨂𑨛𑨄𑨰𑨁𑨙𑨊𑨝𑨁𑩂
𑨋𑨁𑨓𑩇𑨣𑨊𑨞𑨁 𑨙𑨄 𑨱𑨆𑨝𑩇𑨙𑨁 𑨡𑨂𑨰𑨸 𑨞𑨢𑨙𑩇𑨙𑨊, 𑨝 𑨙𑨄 𑨡𑨭𑨸 𑨀𑨔𑩇𑨕𑨢𑨢𑨊𑨛𑨁𑨪𑨝𑩇𑨙𑨁𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
10. 𑨰𑨱𑨊𑨭𑨰𑩇𑨰 𑨛𑨰𑩇𑨰𑨝𑨰𑨢𑩇𑨞𑨛𑨊𑨪 , 𑨙𑨪𑨰𑩇𑨰𑨂 𑨜𑨢𑩇𑨢𑨊 𑨥𑨱𑨁𑨙𑨊 𑨡𑨭𑨝𑩇𑨙𑨁𑩂
𑨰𑨋𑩇𑨋𑨊𑨪𑨛𑨁𑨔𑩇𑨕𑨁𑨊 𑨭𑨁𑨣𑨁𑨋𑨁𑨣𑩇𑨤𑨁𑨙𑨓𑩇𑨣, 𑨰𑨁𑨊𑨬𑨠𑩇𑨠𑨙𑨸 𑨭𑨊𑨞𑨁 𑨪𑨛𑨙𑩇𑨚𑨁 𑨋𑨁𑨓𑩇𑨣𑨁𑩃
11. 𑨣𑨙𑨂𑨊𑨱𑨞𑨊𑨪𑨄𑨱𑨁 𑨣 𑨭𑨁𑨞𑩇𑨞𑨢𑨂𑨙𑩇𑨙𑨆, 𑨤𑨣𑩇𑨣𑨊𑨡𑨁𑨕𑨊𑨝𑨊𑨝𑨁𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
12. 𑨋𑨁𑨓𑩇𑨣𑨊𑨞𑨁 𑨰𑨆 𑨋𑨢𑩇𑨢 𑨋𑨫𑨆𑨙𑨁 𑨞𑨊𑨞𑨋𑨸, 𑨋𑨊𑨪𑨄𑨝 𑨭𑨊𑨣𑨊 𑨀𑨂𑨛 𑨣𑨄𑨙𑨰𑨊 𑨭𑨊𑩂
𑨀𑨡𑨠𑩇𑨠 , 𑨀𑨡𑨠𑩇𑨠𑨙𑨊 𑨛𑨁𑨔𑩇𑨕𑨞𑨛𑨰𑩇𑨰 𑨭𑨂𑨙𑩇𑨙𑨊𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
13.𑨭𑨝𑨞𑩇𑨞𑨍𑨂𑨢𑩇𑨠𑨄 𑨪𑨚 𑨍𑨁𑨢𑩇𑨱𑨄𑩂
𑨙𑨚𑨂𑨊𑨞𑨢𑨸 𑨜𑨢𑩇𑨢𑨭𑨫𑨸 𑨀𑨛𑨄𑨰𑨪𑨁 , 𑨝𑨁𑨠𑩇𑨠𑨊𑨝𑨍𑨊𑨢𑨁𑨸 𑨞𑨫𑨢𑨸 𑨱𑨁𑨙𑨊𑨪𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨠𑨂𑨛𑩇𑨜𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
14. 𑨭𑨫𑨆 𑨭𑨫𑨓𑩇𑨓𑨂𑨊 𑨭𑨫𑨛𑨆 𑨭𑨫𑨊𑨱𑨫𑨆, 𑨀𑨝𑨂𑨙𑩇𑨙𑨫𑨆 𑨜𑨢𑩇𑨢𑨭𑨫𑨸 𑨀𑨛𑨄𑨰𑨪𑨁𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨠𑨂𑨛𑩇𑨜𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
15. 𑨌𑨁𑨊𑨘𑨸 𑨞𑨂𑨫𑨊𑨘𑨸 𑨝𑨭 𑨝𑨙𑩇𑨚𑨁 𑨰𑨢𑩇𑨡𑨭𑨸, 𑨭𑨁𑨫𑨙𑩇𑨙𑨣𑨁𑨙𑩇𑨙𑨊𑨪𑨙𑨁𑨋𑨄 𑨡𑨭𑨰𑩇𑨢𑨁𑨸𑩂
𑨙𑨄 𑨌𑨁𑨊𑨘𑨠𑨁𑨊𑨥𑨊 𑨀𑨭𑨁𑨫𑨂𑨊𑨬𑩇𑨱𑨁𑨤𑨝𑩇𑨛𑨊, 𑨝𑨁𑨠𑩇𑨠𑨝𑩇𑨙𑨁 𑨜𑨁𑨊𑨫𑨊 𑨪𑨚𑨊𑨪𑨸 𑨞𑨛𑨁𑨊𑨞𑨆𑩂
𑨀𑨁𑨛𑨢𑩇𑨞𑨁 𑨰𑨏𑩇𑨎𑨄 𑨫𑨙𑨝𑨸 𑨞𑨘𑨁𑨊𑨙𑨸, 𑨀𑨄𑨙𑨄𑨝 𑨰𑨣𑩇𑨣𑨄𑨝 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
16. 𑨪𑨊𑨝𑨁𑨊𑨜 𑨡𑨂𑨊𑨙𑨊𑨝𑨁 𑨰𑨢𑨊𑨍𑨙𑨊𑨝𑨁, 𑨡𑨂𑨢𑩇𑨢𑨊𑨝𑨁 𑨭𑨊 𑨪𑨊𑨝𑨁 𑨭 𑨀𑨝𑩇𑨙𑨬𑨁𑨋𑩇𑨌𑨄𑩂
𑨙𑨚𑨊𑨍𑨙𑨸 𑨛𑨄𑨭𑨢𑨝𑨂𑨰𑩇𑨰𑨞𑨂𑨊𑨥𑨁𑨙𑨸, 𑨠𑨂𑨛𑩇𑨜𑨸 𑨝𑨢𑨰𑩇𑨰𑨊𑨢 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
17. 𑨪𑨊𑨝𑨁𑨊𑨜 𑨡𑨂𑨊𑨙𑨊𑨝𑨁 𑨰𑨢𑨊𑨍𑨙𑨊𑨝𑨁, 𑨡𑨂𑨢𑩇𑨢𑨊𑨝𑨁 𑨭𑨊 𑨪𑨊𑨝𑨁 𑨭 𑨀𑨝𑩇𑨙𑨬𑨁𑨋𑩇𑨌𑨄𑩂
𑨙𑨚𑨊𑨍𑨙𑨸 𑨛𑨄𑨭𑨢𑨝𑨂𑨰𑩇𑨰𑨞𑨂𑨊𑨥𑨁𑨙𑨸, 𑨜𑨢𑩇𑨢𑨸 𑨝𑨢𑨰𑩇𑨰𑨊𑨢 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑩃
18. 𑨪𑨊𑨝𑨁𑨊𑨜 𑨡𑨂𑨊𑨙𑨊𑨝𑨁 𑨰𑨢𑨊𑨍𑨙𑨊𑨝𑨁, 𑨡𑨂𑨢𑩇𑨢𑨊𑨝𑨁 𑨭𑨊 𑨪𑨊𑨝𑨁 𑨭 𑨀𑨝𑩇𑨙𑨬𑨁𑨋𑩇𑨌𑨄𑩂
𑨙𑨚𑨊𑨍𑨙𑨸 𑨛𑨄𑨭𑨢𑨝𑨂𑨰𑩇𑨰𑨞𑨂𑨊𑨥𑨁𑨙𑨸, 𑨰𑨏𑩇𑨎𑨸 𑨝𑨢𑨰𑩇𑨰𑨊𑨢 𑨰𑨂𑨭𑨙𑩇𑨚𑨁 𑨱𑨆𑨙𑨂𑨊𑨙𑨁𑩃
𑨫𑨙𑨝𑨰𑨂𑨙𑩇𑨙𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
7. 𑨙𑨁𑨫𑨆𑨋𑨂𑨔𑩇𑨔𑨰𑨂𑨙𑩇𑨙𑨸
1. 𑨙𑨁𑨫𑨆𑨋𑨂𑨔𑩇𑨔𑨄𑨰𑨂 𑨙𑨁𑨔𑩇𑨕𑨝𑩇𑨙𑨁, 𑨰𑨝𑩇𑨜𑨁𑨰𑨁𑨏𑩇𑨎𑨊𑨔𑨋𑨄𑨰𑨂 𑨣𑩂
𑨛𑩇𑨭𑨊𑨫𑨠𑨊𑨱𑨊𑨰𑨂 𑨙𑨁𑨔𑩇𑨕𑨝𑩇𑨙𑨁, 𑨀𑨊𑨍𑨝𑩇𑨙𑩇𑨭𑨊𑨝 𑨰𑨋𑨸 𑨎𑨫𑨸𑩃
2. 𑨞𑨱𑨂𑨊𑨙𑨄 𑨀𑨝𑩇𑨝𑨞𑨊𑨝𑨢𑩇𑨱𑨁, 𑨌𑨥𑩇𑨥𑨡𑨆𑨥𑩇𑨥𑨄 𑨀𑨂𑨞𑨔𑩇𑨕𑨁𑨙𑨄𑩂
𑨝 𑨙𑨄𑨰𑨸 𑨋𑨆𑨣𑨁 𑨰𑨫𑨙𑨁, 𑨰𑨙𑩇𑨙𑨊𑨝𑨸 𑨋𑨢𑩇𑨢𑨞𑨣𑩇𑨣𑨪𑨊𑩃
3. 𑨀𑨄𑨭𑨸 𑨛𑨛𑨝𑩇𑨙𑨁 𑨓𑨊𑨙𑨁𑨊𑨝𑨸, 𑨪𑨄 𑨱𑨆𑨝𑩇𑨙𑨁 𑨀𑨝𑨂𑨋𑨢𑩇𑨞𑨋𑨊𑩂
𑨰𑨂𑨣𑨁𑨸 𑨞𑨘𑨁𑨊𑨙𑨸 𑨋𑨊𑨬𑨄𑨝, 𑨋𑨞𑩇𑨞𑨁𑨪𑨸 𑨞𑨊𑨝𑨡𑨆𑨥𑨝𑨸𑩂
𑨀𑨁𑨛𑨸 𑨭𑨆 𑨓𑨊𑨙𑨁𑨊𑨝𑨸 𑨱𑨆𑨙𑨂, 𑨰𑨂𑨌𑨁𑨙𑨊 𑨱𑨆𑨝𑩇𑨙𑨂 𑨓𑨊𑨙𑨪𑨆𑩃
4. 𑨙𑨄 𑨣 𑨙𑨙𑩇𑨚 𑨰𑨢𑨊𑨍𑨝𑩇𑨙𑩇𑨭𑨊, 𑨓𑨊𑨙𑨁𑨞𑨄𑨙𑨊 𑨰𑨢𑨊𑨍𑨙𑨊𑩂
𑨞𑨱𑨂𑨊𑨙𑨄 𑨀𑨝𑩇𑨝𑨞𑨊𑨝𑨢𑩇𑨱𑨁, 𑨰𑨋𑩇𑨋𑨣𑩇𑨣𑨸 𑨀𑨝𑨂𑨢𑨆𑨛𑨫𑨄𑩃
5. 𑨣𑨁𑨫𑨸 𑨥𑨁𑨊𑨭𑨝𑩇𑨙𑨂 𑨝𑨆 𑨓𑨊𑨙𑨁𑨊, 𑨪𑨄𑨰𑨸 𑨱𑨄𑨙𑨂 𑨬𑨡𑨊𑨢𑨰𑨄𑩂
𑨀𑨢𑩇𑨱𑨊𑨋𑨓𑩇𑨣 𑨋𑨙𑨊 𑨞𑨂𑨊𑨥𑨊, 𑨛𑨊𑨪𑨋𑨊 𑨣 𑨀𑨝𑨁𑨞𑩇𑨟𑨬𑨊𑩃
6. 𑨝 𑨱𑨁 𑨙𑨙𑩇𑨚 𑨋𑨰𑨁 𑨀𑨙𑩇𑨚𑨁, 𑨍𑨆𑨫𑨋𑩇𑨌𑨄𑨙𑩇𑨚 𑨝 𑨭𑨁𑨥𑩇𑨥𑨙𑨁𑩂
𑨭𑨘𑨁𑨥𑩇𑨥𑨊 𑨙𑨊𑨛𑨁𑨰𑨁𑨊 𑨝𑨙𑩇𑨚𑨁, 𑨱𑨁𑨫𑨓𑩇𑨓𑨄𑨝 𑨋𑨪𑨆𑨋𑨪𑨸𑩂
𑨀𑨁𑨙𑨆 𑨛𑨁𑨝𑩇𑨝𑨄𑨝 𑨪𑨊𑨞𑨄𑨝𑩇𑨙𑨁, 𑨞𑨄𑨙𑨊 𑨋𑨊𑨬𑨏𑩇𑨋𑨙𑨊 𑨙𑨱𑨁𑨸𑩃
7. 𑨀𑨂𑨝𑩇𑨝𑨢𑨄 𑨀𑨂𑨛𑨋𑨸 𑨭𑨂𑨔𑩇𑨕𑨸, 𑨪𑨚𑨊 𑨝𑨁𑨝𑩇𑨝𑨸 𑨞𑨭𑨙𑩇𑨙𑨙𑨁𑩂
𑨀𑨄𑨭𑨢𑨄𑨭 𑨀𑨁𑨙𑨆 𑨛𑨁𑨝𑩇𑨝𑨸, 𑨞𑨄𑨙𑨊𑨝𑨸 𑨀𑨂𑨞𑨋𑨞𑩇𑨞𑨙𑨁𑩃
8. 𑨪𑨚𑨊 𑨭𑨊𑨫𑨁𑨭𑨱𑨊 𑨞𑨂𑨊𑨫𑨊, 𑨞𑨫𑨁𑨞𑨂𑨊𑨫𑨄𑨝𑩇𑨙𑨁 𑨰𑨊𑨍𑨫𑨸𑩂
𑨀𑨄𑨭𑨢𑨄𑨭 𑨀𑨁𑨙𑨆 𑨛𑨁𑨝𑩇𑨝𑨸, 𑨞𑨄𑨙𑨊𑨝𑨸 𑨀𑨂𑨞𑨋𑨞𑩇𑨞𑨙𑨁𑩃
9. 𑨀𑨛𑨊𑨰𑨁 𑨢𑨄 𑨀𑨋𑨊𑨰𑨁 𑨢𑨄, 𑨓𑨊𑨙𑨁𑨢𑨁𑨙𑩇𑨙𑨊 𑨰𑨌𑨊 𑨣 𑨢𑨄𑩂
𑨞𑨄𑨙𑨊𑨝𑨸 𑨛𑨋𑩇𑨌𑨁𑨘𑨸 𑨛𑨥𑩇𑨥𑨊, 𑨞𑨂𑨠𑩇𑨠𑨄 𑨋𑨙𑨢𑨝𑨂𑨰𑩇𑨰𑨫𑨸𑩃
10. 𑨝 𑨱𑨁 𑨫𑨂𑨘𑩇𑨘𑨸 𑨭𑨊 𑨰𑨆𑨋𑨆 𑨭𑨊, 𑨪𑨊 𑨣𑨓𑩇𑨓𑨊 𑨞𑨫𑨁𑨛𑨄𑨭𑨝𑨊𑩂
𑨝 𑨙𑨸 𑨞𑨄𑨙𑨊𑨝𑨢𑨙𑩇𑨚𑨊𑨪, 𑨀𑨄𑨭𑨸 𑨙𑨁𑨔𑩇𑨕𑨝𑩇𑨙𑨁 𑨓𑨊𑨙𑨪𑨆𑩃
11. 𑨀𑨪𑨓𑩇𑨣 𑨌𑨆 𑨛𑨋𑩇𑨌𑨁𑨘𑨊 𑨛𑨁𑨝𑩇𑨝𑨊, 𑨰𑨏𑩇𑨎𑨢𑩇𑨱𑨁 𑨰𑨂𑨞𑩇𑨞𑨙𑨁𑨔𑩇𑨕𑨁𑨙𑨊𑩂
𑨛𑨁𑨊𑨎𑨫𑨙𑩇𑨙𑨸 𑨱𑨁𑨙𑨊𑨪𑨰𑩇𑨰, 𑨕𑨊𑨝𑨰𑨆 𑨀𑨂𑨞𑨋𑨞𑩇𑨞𑨙𑨁𑩃
12. 𑨰𑨆 𑨓𑨊𑨙𑨁𑨜𑨢𑩇𑨢𑨆 𑨣 𑨀𑨪𑨸 𑨝𑨁𑨛𑨰𑩇𑨰𑨁𑨙𑨆, 𑨞𑨄𑨙𑨊𑨝 𑨞𑨂𑨊𑨥𑨊 𑨣 𑨋𑨙𑨊 𑨀𑨂𑨬𑨊𑨫𑨊𑩂
𑨠𑨬𑨓𑩇𑨣 𑨡𑨁𑨋𑩇𑨌𑨂𑨊𑨝𑨢𑨝𑨂𑨞𑩇𑨞𑨛𑨁𑨝𑩇𑨝𑨸 , 𑨙𑨂𑨢𑩇𑨱𑨄𑨱𑨁 𑨞𑨂𑨓𑩇𑨓𑨸 𑨞𑨰𑨂𑨙𑨸 𑨀𑨝𑨞𑩇𑨞𑨋𑨝𑩇𑨙𑨁𑩃
𑨙𑨁𑨫𑨆𑨋𑨂𑨔𑩇𑨔𑨰𑨂𑨙𑩇𑨙𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
8. 𑨝𑨁𑨜𑨁𑨋𑨘𑩇𑨖𑨰𑨂𑨙𑩇𑨙𑨸
1. 𑨝𑨁𑨜𑨁𑨸 𑨝𑨁𑨜𑨄𑨙𑨁 𑨞𑨂𑨫𑨁𑨰𑨆, 𑨍𑨢𑩇𑨡𑨁𑨊𑨫𑨄 𑨀𑨆𑨛𑨋𑨝𑩇𑨙𑨁𑨋𑨄𑩂
𑨀𑨙𑩇𑨚𑨄 𑨋𑨁𑨣𑩇𑨣𑨄 𑨰𑨢𑨂𑨞𑩇𑨞𑨝𑩇𑨝𑨄, 𑨀𑨙𑩇𑨚𑨊𑨪 𑨢𑨄 𑨡𑨭𑨁𑨰𑩇𑨰𑨙𑨁𑩃
2. 𑨫𑨊𑨥𑨙𑨆 𑨭𑨊 𑨛𑨂𑨫𑨂𑨙𑩇𑨙𑨰𑩇𑨰, 𑨣𑨆𑨫𑨙𑨆 𑨞𑨁𑨊𑨬𑨁𑨙𑨰𑩇𑨰 𑨭𑨊𑩂
𑨀𑨁𑨘𑨰𑩇𑨰 𑨭𑨊 𑨞𑨢𑨆𑨋𑩇𑨌𑨊𑨪, 𑨛𑨂𑨠𑩇𑨡𑨁𑨋𑩇𑨌𑨄 𑨀𑨊𑨞𑨛𑨊𑨰𑨂 𑨭𑨊𑩂
𑨀𑨄𑨙𑨛𑨙𑩇𑨚𑨊𑨪 𑨬𑨆𑨋𑨰𑩇𑨢𑨁𑨸, 𑨝𑨁𑨜𑨁 𑨝𑨊𑨢 𑨝𑨁𑨜𑨁𑨊𑨪𑨙𑨁𑩃

9. 𑨀𑨰𑨊𑨜𑨊𑨫𑨘𑨢𑨓𑩇𑨓𑨄𑨰𑨸, 𑨀𑨣𑨆𑨫𑨊𑨱𑨫𑨘𑨆 𑨝𑨁𑨜𑨁𑩂
𑨋𑨪𑨁𑨫𑨊𑨚 𑨜𑨁𑨊𑨫𑨆 𑨞𑨂𑨓𑩇𑨓𑨊𑨝𑨁, 𑨪𑨆 𑨝𑨁𑨜𑨁 𑨀𑨝𑨂𑨍𑨊𑨢𑨁𑨋𑨆𑩃
10. 𑨀𑨄𑨰 𑨛𑨄𑨭𑨢𑨝𑨂𑨰𑩇𑨰𑨊𑨝𑨸, 𑨰𑨠𑩇𑨠𑨋𑨊𑨢𑨛𑨛𑨆 𑨝𑨁𑨜𑨁𑩂
𑨪𑨸 𑨪𑨛𑨄𑨭𑨊𑨡𑨁𑨞𑨙𑩇𑨚𑨄𑨝𑩇𑨙𑨁, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
11. 𑨰𑨂𑨭𑨘𑩇𑨘𑨙𑨊 𑨰𑨂𑨰𑨫𑨙𑨊, 𑨰𑨂𑨰𑨘𑩇𑨕𑨊𑨝𑨊 𑨰𑨂𑨫𑨂𑨊𑨞𑨙𑨊𑩂
𑨀𑨊𑨜𑨁𑨞𑨣𑩇𑨣𑨞𑨫𑨁𑨭𑨊𑨫𑨆, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
12. 𑨞𑨛𑨄𑨰𑨫𑨥𑩇𑨥𑨸 𑨀𑨁𑨰𑩇𑨰𑨫𑨁𑨪𑨸, 𑨣𑨋𑩇𑨋𑨭𑨙𑩇𑨙𑨁𑨰𑨂𑨌𑨸 𑨞𑨁𑨪𑨸𑩂
𑨛𑨄𑨭𑨫𑨥𑩇𑨥𑨢𑩇𑨞𑨁 𑨛𑨁𑨠𑩇𑨠𑨄𑨰𑨂, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
13. 𑨢𑨊𑨝𑨂𑨰𑩇𑨰𑨁𑨋𑨊 𑨣 𑨰𑨢𑩇𑨞𑨙𑩇𑨙𑨁, 𑨛𑨄𑨭𑨬𑨆𑨋𑨄 𑨣 𑨪𑨊 𑨫𑨙𑨁𑩂
𑨪𑨊 𑨣 𑨝𑨁𑨠𑩇𑨠𑨊𑨝𑨰𑨢𑩇𑨞𑨙𑩇𑨙𑨁, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
14. 𑨢𑨁𑨙𑩇𑨙𑨰𑨢𑩇𑨞𑨛𑨢𑨊𑨍𑨢𑩇𑨢, 𑨪𑨆𑨝𑨁𑨰𑨆𑨭 𑨞𑨪𑨂𑨓𑩇𑨥𑨙𑨆𑩂
𑨭𑨁𑨥𑩇𑨥𑨊 𑨭𑨁𑨢𑨂𑨙𑩇𑨙𑨁 𑨭𑨰𑨁𑨊𑨡𑨊𑨭𑨆, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
15. 𑨞𑨔𑨁𑨰𑨢𑩇𑨡𑨁𑨛𑨊 𑨭𑨁𑨢𑨆𑨋𑩇𑨌𑨊 𑨣, 𑨪𑨊 𑨣 𑨰𑨊𑨭𑨋𑨞𑨊𑨫𑨢𑨁𑨊𑩂
𑨞𑨣𑩇𑨣𑨄𑨋𑨠𑨆𑨜𑨁 𑨠𑨂𑨛𑩇𑨜𑨡𑨂𑨊𑨢𑨁, 𑨰𑨠𑩇𑨠𑨢𑨄𑨙𑨄𑨝 𑨬𑨠𑩇𑨡𑨙𑨁𑩃
16. 𑨀𑨄𑨭𑨸 𑨢𑨱𑨙𑩇𑨚𑨁𑨋𑨊 𑨀𑨄𑨰𑨊, 𑨪𑨛𑨁𑨛𑨸 𑨞𑨂𑨓𑩇𑨓𑨰𑨢𑩇𑨞𑨛𑨊𑩂
𑨙𑨰𑩇𑨢𑨊 𑨜𑨁𑨊𑨫𑨊 𑨞𑨰𑨸𑨰𑨝𑩇𑨙𑨁, 𑨞𑨘𑩇𑨖𑨁𑨙𑨊 𑨋𑨙𑨞𑨂𑨓𑩇𑨓𑨙𑨝𑩇𑨙𑨁𑩃
𑨝𑨁𑨜𑨁𑨋𑨘𑩇𑨖𑨰𑨂𑨙𑩇𑨙𑨸 𑨝𑨁𑨔𑩇𑨕𑨁𑨙𑨸𑩂
9. 𑨢𑨄𑨙𑩇𑨙𑨰𑨂𑨙𑩇𑨙𑨸
1. 𑨋𑨫𑨘𑨁𑨊𑨪𑨢𑨙𑩇𑨚𑨋𑨂𑨰𑨬𑨄𑨝, 𑨪𑨝𑩇𑨙𑨰𑨝𑩇𑨙𑨸 𑨞𑨛𑨸 𑨀𑨡𑨁𑨰𑨢𑨄𑨣𑩇𑨣𑩂
𑨰𑨋𑩇𑨋𑨆 𑨀𑨂𑨥𑨂𑨊 𑨣 𑨰𑨂𑨱𑨂𑨥𑨂𑨊 𑨣, 𑨰𑨂𑨭𑨣𑨆 𑨣𑨰𑩇𑨰 𑨢𑨂𑨛𑨂 𑨀𑨝𑨙𑨁𑨢𑨊𑨝𑨁𑨊𑩃
2. 𑨰𑨝𑩇𑨙𑨂𑨰𑩇𑨰𑨋𑨆 𑨣 𑨰𑨂𑨡𑨫𑨆 𑨣, 𑨀𑨞𑩇𑨞𑨋𑨁𑨣𑩇𑨣𑨆 𑨣 𑨰𑨬𑩇𑨬𑨱𑨂𑨋𑨭𑨂𑨙𑩇𑨙𑨁𑩂
𑨰𑨝𑩇𑨙𑨁𑨝𑩇𑨛𑩇𑨫𑨁𑨪𑨆 𑨣 𑨝𑨁𑨞𑨋𑨆 𑨣, 𑨀𑨞𑩇𑨞𑨍𑨠𑩇𑨡𑨆 𑨋𑨂𑨬𑨄𑨰𑩇𑨭𑨝𑨝𑨂𑨍𑨁𑨛𑩇𑨜𑨆𑩃
3. 𑨝 𑨣 𑨌𑨂𑨛𑩇𑨛𑨢𑨊𑨣𑨫𑨄 𑨋𑨁𑨓𑩇𑨣𑨁, 𑨪𑨄𑨝 𑨭𑨁𑨓𑩇𑨓𑨂𑨊 𑨞𑨫𑨄 𑨀𑨂𑨞𑨭𑨛𑨄𑨪𑩇𑨪𑨂𑨸𑩂
𑨰𑨂𑨌𑨁𑨝𑨆𑨭 𑨌𑨄𑨢𑨁𑨝𑨆 𑨱𑨆𑨝𑩇𑨙𑨂, 𑨰𑨠𑩇𑨠𑨰𑨙𑩇𑨙𑨊 𑨡𑨭𑨝𑩇𑨙𑨂 𑨰𑨂𑨌𑨁𑨙𑨙𑩇𑨙𑨊𑩃
4. 𑨪𑨄 𑨋𑨄𑨣𑨁 𑨞𑨊𑨘𑨡𑨂𑨊𑨙𑨙𑩇𑨚𑨁, 𑨙𑨰𑨊 𑨭𑨊 𑨚𑨊𑨭𑨫𑨊 𑨭𑨝𑨭𑨰𑨄𑨰𑨊𑩂
𑨛𑨁𑨊𑨎𑨊 𑨭𑨊 𑨪𑨄𑨭 𑨢𑨱𑨝𑩇𑨙𑨊 , 𑨢𑨥𑩇𑨦𑨁𑨢𑨊 𑨫𑨰𑩇𑨰𑨋𑨊 𑨀𑨘𑨂𑨋𑨚𑨂𑨊𑨬𑨊𑩃
5. 𑨛𑨁𑨔𑩇𑨕𑨊 𑨭𑨊 𑨪𑨄𑨭 𑨀𑨛𑨁𑨔𑩇𑨕𑨊 𑨛𑨂𑨊𑨫𑨄 𑨭𑨰𑨝𑩇𑨙𑨁 𑨀𑨭𑨁𑨛𑨂𑨊𑨫𑨄𑩂
𑨡𑨂𑨊𑨙𑨊 𑨭 , 𑨰𑨠𑩇𑨠𑨰𑨙𑩇𑨙𑨊 𑨡𑨭𑨝𑩇𑨙𑨂 𑨰𑨂𑨌𑨁𑨙𑨙𑩇𑨙𑨊𑩃

7. 𑨢𑨊𑨙𑨊 𑨪𑨚𑨊 𑨝𑨁𑨪𑨸 𑨞𑨂𑨙𑩇𑨙𑨢𑨊𑨪𑨂𑨰𑨊 𑨀𑨄𑨋𑨞𑨂𑨙𑩇𑨙𑨢𑨝𑨂𑨫𑨋𑩇𑨌𑨄𑩂

8. 𑨢𑨄𑨙𑩇𑨙𑨓𑩇𑨣 𑨰𑨠𑩇𑨠𑨬𑨆𑨋𑨰𑩇𑨢𑨁, 𑨢𑨊𑨝𑨰𑨸 𑨡𑨊𑨭𑨪𑨄 𑨀𑨞𑨫𑨁𑨢𑨊𑨘𑨸𑩂
𑨀𑨂𑨛𑩇𑨜𑨸 𑨀𑨜𑨆 𑨣 𑨙𑨁𑨫𑨁𑨪𑨓𑩇𑨣, 𑨀𑨰𑨢𑩇𑨠𑨊𑨜𑨸 𑨀𑨭𑨄𑨫𑨢𑨰𑨞𑨙𑩇𑨙𑨸𑩃

10. 𑨛𑨁𑨔𑩇𑨕𑨁𑨓𑩇𑨣 𑨀𑨝𑨂𑨞𑨍𑩇𑨍𑨢𑩇𑨢, 𑨰𑨁𑨊𑨬𑨭𑨊 𑨛𑨰𑩇𑨰𑨝𑨄𑨝 𑨰𑨢𑩇𑨞𑨝𑩇𑨝𑨆𑩂
