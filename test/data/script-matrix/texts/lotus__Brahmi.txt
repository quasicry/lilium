𑀉𑀧𑀧𑀸𑀤𑀼𑀓 𑀢𑁂 𑀚𑀺𑀦𑁄𑀭𑀲𑀸𑀂

𑀲𑁄 𑀘𑁃𑀯 𑀅𑀫𑀺𑀢𑀸𑀪𑀦𑀸𑀬𑀓𑀂
𑀧𑀤𑁆𑀫𑀕𑀭𑁆𑀪𑁂 𑀯𑀺𑀭𑀚𑁂 𑀫𑀦𑁄𑀭𑀫𑁂𑁇
𑀲𑀺𑀁𑀳𑀸𑀲𑀦𑀺 𑀲𑀁𑀦𑀺𑀱𑀡𑁆𑀡𑀓𑁄
𑀰𑀸𑀮𑀭𑀚𑁄 𑀯 𑀬𑀣𑀸 𑀯𑀺𑀭𑀸𑀚𑀢𑁂𑁈𑁩𑁨𑁈

𑀬𑀦𑁆𑀫𑁂 𑀧𑀼𑀡𑁆𑀬 𑀲𑁆𑀢𑀯𑀺𑀢𑁆𑀯 𑀲𑀁𑀘𑀺𑀢𑀁
𑀓𑁆𑀱𑀺𑀧𑁆𑀭 𑀪𑁄𑀫𑀺 𑀬𑀣 𑀢𑁆𑀯𑀁 𑀦𑀭𑁄𑀢𑁆𑀢𑀫𑁈𑁩𑁩𑁈 𑀇𑀢𑀺𑁈
