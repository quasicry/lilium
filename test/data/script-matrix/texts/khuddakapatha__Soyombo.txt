𑪜 𑩯𑩴𑩖 𑩫𑪁𑪘 𑩳𑩞𑩾𑩫𑩖 𑩐𑩼𑪂𑩫𑩖 𑪁𑩴𑪘𑩛𑪁𑩴𑪙𑩲𑩒𑩭𑪙𑩮𑪁𑪘 𑪜
𑩝𑩒𑩭𑪘𑩜𑩯𑩑𑩜𑩛𑩻𑩔
𑩝𑩒𑩭𑪘𑩜𑩰𑩛𑩧𑩰𑩛𑩽𑩑
1. 𑪁𑩼𑩪𑩫𑪘𑩻𑪖
𑩲𑩒𑩭𑪙𑩮𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑩮𑩴𑪘𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑪁𑩠𑪙𑩟𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪜
𑩭𑩒𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑩲𑩒𑩭𑪙𑩮𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑩭𑩒𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑩮𑩴𑪘𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑩭𑩒𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪜
𑩫𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑩲𑩒𑩭𑪙𑩮𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑩫𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑩮𑩴𑪘𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪛
𑩫𑩫𑩑𑩻𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑪖 𑪁𑩼𑩪𑪖 𑩞𑩵𑪙𑩶𑩛𑩴𑩑𑪜
𑪁𑩼𑩪𑩫𑪘𑩻𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
2. 𑩭𑪁𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖
1. 𑩰𑩛𑩪𑩛𑩫𑩑𑩰𑩛𑩫𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
2. 𑩐𑩭𑩑𑩯𑪘𑩛𑩭𑩛𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
3. 𑩐𑩲𑪙𑩼𑪂𑪙𑩴𑩵𑩼𑩑𑩻𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
4. 𑩴𑩒𑪁𑩛𑩾𑩛𑩭𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
5. 𑪁𑩒𑩼𑩛𑩴𑩔𑩼𑩻𑩴𑩷𑪘𑩰𑩴𑩛𑩭𑩦𑪙𑩧𑩛𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
6. 𑩾𑩑𑩜𑩛𑩽𑩳𑩖𑩷𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
7. 𑩯𑩵𑪘-𑩞𑩑𑩛𑩫-𑩾𑩛𑩭𑩑𑩫-𑩾𑩑𑪁𑩒𑩛𑩜𑩭𑪁𑪘𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
8. 𑩴𑩛𑩽𑩛-𑩞𑩯𑪙𑩮-𑩾𑩑𑩽𑩔𑩰𑩯-𑩮𑩛𑩼𑩪-𑩴𑩪𑪙𑩨𑩯-𑩾𑩑𑩳𑩒𑩛𑪁𑩯𑩦𑪙𑩧𑩛𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
9. 𑩐𑩒𑩵𑪘𑩛𑪁𑩻𑩯-𑩴𑪂𑩛𑪁𑩻𑩯𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
10. 𑩷𑩛𑩫𑩼𑩒𑩛𑩰-𑩼𑩷𑩫𑩰𑩦𑩑𑩞𑪘𑪂𑩪𑩛 𑩾𑩔𑩼𑩴𑩪𑩑𑩛-𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑪁𑩴𑩛𑩭𑩑𑩻𑩛𑩴𑩑𑪛
𑩭𑪁𑪁𑩑𑩜𑪙𑩝𑩛𑩰𑩭𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
3. 𑩭𑪙𑩾𑩫𑪘𑩑𑪖𑪁𑩛𑩜𑩛𑩼𑩖
𑩐𑩫𑪙𑩬𑩑 𑩐𑩑𑩴𑪁𑪙𑩴𑩑𑪖 𑩜𑩛𑩻𑩔 –
𑩜𑩔𑪁𑩛 𑩽𑩖𑩴𑩛 𑩯𑩝𑩛 𑩭𑩯𑪙𑩫𑩛 𑩫𑩵𑩖,
𑩴𑪖𑪁𑪖 𑩯𑪙𑪂𑩛𑩼𑩒 𑩐𑩦𑪙𑩧𑩑 𑩐𑩦𑪙𑩧𑩑𑩴𑩑𑩥𑪙𑩷𑪖 𑩾𑩜𑪘𑪖,
𑪂𑩭𑩻𑪖 𑩻𑩜𑩯𑪖 𑩜𑩑𑩽𑩖𑩴𑩜𑪖 𑩰𑩑𑪂𑩜𑪖 𑩰𑩰𑪙𑩱𑩛𑪁𑪖,
𑩐𑩯𑪙𑩫𑪖 𑩐𑩯𑪙𑩫𑩞𑩒𑩪𑪖 𑩐𑩒𑩭𑩼𑩑𑩻𑪖 𑩜𑩼𑩑𑩛𑪁𑪖 𑩴𑩫𑪙𑩬𑩽𑩒𑩠𑪙𑩞𑪖,
𑩰𑩑𑩫𑪘𑪖 𑪁𑩔𑩴𑪙𑪂𑪖 𑩰𑩒𑩲𑪘𑩖 𑩽𑩖𑪂𑩑𑩫𑪖 𑪁𑩔𑩭𑩖 𑩴𑩔𑩭𑩖,
𑩐𑪁𑪘𑩒 𑩾𑪁𑩛 𑩝𑩔𑩽𑩖 𑪁𑩑𑩠𑪙𑩟𑩛𑩪𑩑𑩜𑩛 𑩽𑪁𑩑𑩜𑩛 𑩴𑩒𑩫𑪘𑩯𑪙𑩫𑩑 𑪜
𑩭𑪙𑩾𑩫𑪘𑩑𑪖𑪁𑩛𑩜𑩛𑩼𑩖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑩖𑪛
4. 𑩜𑩒𑩴𑩛𑩼𑩰𑩥𑪙𑪂𑩛
1. ‘‘𑩐𑩔𑩜𑪖 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑪁𑩲𑪘𑩔 𑪁𑩫𑪘𑩛 𑩐𑩛𑪂𑩛𑩼𑩦𑪙𑩧𑩑𑩫𑩑𑩜𑩛’’𑪛
2. ‘‘𑩭𑪙𑩾𑩔 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩯𑩛𑩴𑩥𑪙𑩵 𑩼𑩒𑩛𑩰𑩥𑪙𑩵’’𑪛
3. ‘‘𑩫𑩑𑩛𑩪𑩑 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩫𑩑𑪁𑪘𑩖 𑩾𑩔𑩭𑩯𑩛’’𑪛
4. ‘‘𑩵𑩫𑪘𑩛𑩼𑩑 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩵𑩫𑪘𑩛𑩼𑩑 𑩐𑩼𑩑𑩻𑪁𑩵𑪘𑩛𑩯𑩑’’𑪛
5. ‘‘𑩰𑩥𑪙𑩵 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩰𑩥𑪙𑩵𑩒𑩰𑩛𑩭𑩛𑩯𑩜𑪙𑩝𑩯𑪙𑩮𑩛’’𑪛
6. ‘‘𑩶 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩶 𑩐𑩷𑪙𑩤𑩫𑪘𑩑𑩜𑩛𑩯𑩑 𑩐𑩛𑩻𑩫𑩯𑩛𑩯𑩑’’𑪛
7. ‘‘𑪁𑩫𑪘 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑪁𑩫𑪘 𑩲𑩖𑩷𑪙𑩤𑩠𑪙𑩞𑩛’’𑪛
8. ‘‘𑩐𑩦𑪙𑩧 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩐𑩼𑩑𑩻𑩖 𑩐𑩦𑪙𑩧𑩠𑪙𑩞𑩑𑩜𑩖 𑩴𑩞𑪘𑩖’’𑪛
9. ‘‘𑩯𑩾 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩯𑩾 𑪁𑩫𑪘𑩛𑩾𑩛𑪁𑩛’’𑪛
10. ‘‘𑩭𑪁 𑩯𑩛𑩴 𑩜𑩑𑪖’’? ‘‘𑩭𑪁𑪂𑩠𑪙𑩞𑩔𑪂𑩑 𑪁𑩴𑩯𑪘𑩛𑩞𑩫𑩖 ‘𑩐𑩼𑪂𑩛’𑩫𑩑 𑩾𑩒𑩵𑪘𑩫𑩑𑩛’’𑩫𑩑𑪛
𑩜𑩒𑩴𑩛𑩼𑩰𑩥𑪙𑪂𑩛 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑩛𑪛
5. 𑩴𑩠𑪙𑩞𑩽𑪁𑩒𑩫𑪘𑪖
1. 𑩐𑩔𑩾𑪖 𑩴𑩔 𑪁𑩒𑩫𑪖 – 𑩐𑩔𑩜𑪖 𑪁𑩴𑩻𑪖 𑩳𑩞𑩾𑩛 𑪁𑩛𑩾𑩫𑪙𑩬𑩑𑩻𑪖 𑩾𑩑𑪂𑩼𑩫𑩑 𑩷𑩔𑩫𑩾𑩯𑩔 𑩐𑩯𑩛𑩬𑩰𑩑𑩪𑪙𑩨𑩑𑩜𑪁𑪘 𑩐𑩛𑩼𑩛𑩴𑩔𑪛 𑩐𑩬 𑩝𑩖 𑩐𑩥𑪘𑩫𑩼𑩛 𑩭𑩔𑩾𑩫𑩛 𑩐𑩳𑩑𑩜𑪘𑩯𑪙𑩫𑩛𑩻 𑩼𑩫𑪘𑩑𑩻𑩛 𑩐𑩳𑩑𑩜𑪘𑩯𑪙𑩫𑩾𑩪𑪘𑩛 𑩜𑩔𑩾𑩽𑩜𑩰𑪘𑪖 𑩷𑩔𑩫𑩾𑩯𑪖 𑩐𑩖𑩳𑩛𑪁𑩔𑩫𑪙𑩾𑩛 𑩻𑩔𑩯 𑩳𑩞𑩾𑩛 𑩫𑩔𑩯𑩒𑩰𑪁𑩠𑪙𑩜𑩴𑩑; 𑩐𑩒𑩰𑪁𑩠𑪙𑩜𑩴𑩑𑩫𑪙𑩾𑩛 𑩳𑩞𑩾𑩯𑪙𑩫𑪖 𑩐𑩳𑩑𑩾𑩛𑩭𑩔𑩫𑪙𑩾𑩛 𑩐𑩔𑩜𑩴𑩯𑪙𑩫𑪖 𑩐𑩦𑪙𑩧𑩛𑪁𑩑𑪛 𑩐𑩔𑩜𑩴𑩯𑪙𑩫𑪖 𑩧𑩑𑩫𑩛 𑩝𑩖 𑪁𑩛 𑩭𑩔𑩾𑩫𑩛 𑩳𑩞𑩾𑩯𑪙𑩫𑪖 𑩞𑩛𑩬𑩛𑩻 𑩐𑩷𑪙𑩤𑩳𑩛𑪁𑩑 –
2. ‘‘𑩲𑪂𑩒𑩛 𑩭𑩔𑩾𑩛 𑩴𑩯𑩒𑪁𑪘𑩛 𑩵, 𑩴𑩠𑪙𑩞𑩽𑩛𑩯𑩑 𑩐𑩵𑩑𑩯𑪙𑩫𑩻𑩒𑪖𑪛
𑩐𑩛𑩜𑩠𑪙𑩝𑩴𑩛𑩯𑩛 𑪁𑩖𑩫𑪙𑩬𑩛𑩯𑪖, 𑩲𑪙𑩼𑩒𑩛𑪂𑩑 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖’’𑪜
3. ‘‘𑩐𑪁𑩔𑩾𑩯𑩛 𑩵 𑩲𑩛𑩽𑩛𑩯𑪖, 𑩰𑩪𑪙𑩨𑩑𑩫𑩛𑩯𑩥𑪙𑩵 𑪁𑩔𑩾𑩯𑩛𑪛
𑩰𑩒𑩛𑩷𑩛 𑩵 𑩰𑩒𑩛𑩷𑩯𑩔𑩻𑪘𑩛𑩯𑪖 , 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
4. ‘‘𑩰𑩫𑩑𑩼𑩒𑩛𑩰𑩭𑩔𑪁𑩾𑩛𑪁𑩖 𑩵, 𑩰𑩒𑩲𑪘𑩔 𑩵 𑩜𑩫𑩰𑩒𑩥𑪘𑩫𑩛𑪛
𑩐𑩫𑪘𑪁𑩴𑪘𑩛𑩰𑩪𑩑𑩮𑩑 𑩵, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
5. ‘‘𑩲𑩛𑪂𑩒𑪁𑩵𑪘𑩥𑪙𑩵 𑪁𑩑𑩰𑪘𑩥𑪙𑩵, 𑩾𑩑𑩯𑩻𑩖 𑩵 𑪁𑩒𑪁𑩑𑩜𑪙𑩝𑩑𑩫𑩖𑪛
𑪁𑩒𑩳𑩛𑪁𑩑𑩫𑩛 𑩵 𑩻𑩛 𑩾𑩛𑩵𑩛, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
6. ‘‘𑩴𑩛𑩫𑩛𑩰𑩑𑩫𑩒 𑩐𑩒𑩰𑩦𑪙𑩧𑩛𑩯𑪖, 𑩰𑩒𑩫𑪘𑩭𑩛𑩼𑪁𑪘 𑪁𑩠𑪙𑩞𑪂𑩖𑪛
𑩐𑩯𑩛𑩜𑩒𑩽𑩛 𑩵 𑩜𑩴𑪘𑩯𑪙𑩫𑩛, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
7. ‘‘𑩭𑩛𑩯𑩥𑪙𑩵 𑩮𑩴𑪘𑩵𑩼𑩑𑩻𑩛 𑩵, 𑩥𑩛𑩫𑩜𑩛𑩯𑩥𑪙𑩵 𑪁𑩠𑪙𑩞𑪂𑩖𑪛
𑩐𑩯𑩾𑩷𑪘𑩛𑩯𑩑 𑩜𑩴𑪘𑩛𑩯𑩑, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
8. ‘‘𑩐𑩛𑩼𑩫𑩑𑩛 𑩾𑩑𑩼𑩫𑩑𑩛 𑩰𑩛𑩰𑩛, 𑩴𑩷𑪘𑩰𑩛𑩯𑩛 𑩵 𑪁𑪖𑩻𑩴𑩖𑪛
𑩐𑩰𑪘𑩴𑩛𑩭𑩖 𑩵 𑩮𑩴𑪘𑩔𑪁𑩒, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
9. ‘‘𑩞𑩛𑩼𑩾𑩖 𑩵 𑩯𑩑𑩾𑩛𑩫𑩖 𑩵, 𑪁𑩯𑪙𑩫𑩒𑩦𑪙𑩧𑩑 𑩵 𑩜𑩫𑩥𑪘𑩒𑩫𑩛𑪛
𑩜𑩛𑩽𑩔𑩯 𑩮𑩴𑪘𑪁𑪘𑩾𑩯𑪖, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
10. ‘‘𑩝𑩯𑪙𑩫𑩑𑩛 𑩵 𑪁𑩖𑩾𑩵𑪁𑪘𑩫𑩛, 𑪁𑩴𑩪𑩛𑩯𑩥𑪙𑩵 𑩭𑪁𑪘𑩯𑪖𑪛
𑩜𑩛𑩽𑩔𑩯 𑩮𑩴𑪘𑪁𑩛𑩜𑩵𑪙𑩶𑩛, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
11. ‘‘𑩫𑩰𑩖 𑩵 𑩲𑪙𑩼𑪂𑪙𑩴𑩵𑩼𑩑𑩻𑩥𑪙𑩵, 𑩐𑩼𑩑𑩻𑪁𑩵𑪘𑩛𑩯 𑩭𑪁𑪘𑩯𑪖𑪛
𑩯𑩑𑩲𑪘𑩛𑩯𑪁𑩵𑪙𑩶𑩑𑩜𑩑𑩼𑩑𑩻𑩛 𑩵, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
12. ‘‘𑩱𑩒𑩦𑪙𑩧𑪁𑪘 𑩽𑩖𑩜𑩮𑩴𑪘𑩔𑪂𑩑, 𑩵𑩑𑩫𑪘𑪖 𑩻𑪁𑪘 𑩯 𑩜𑩴𑪙𑩰𑩫𑩑𑪛
𑩐𑪁𑩖𑩜𑪖 𑩾𑩑𑩼𑩷𑪖 𑩝𑩔𑩴𑪖, 𑩐𑩔𑩫𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴𑪖𑪜
13. ‘‘𑩐𑩔𑩫𑩛𑩭𑩑𑪁𑩛𑩯𑩑 𑩜𑩫𑪙𑩾𑩛𑩯, 𑪁𑩲𑪘𑩫𑪙𑩬𑩴𑩰𑩼𑩛𑩷𑩑𑩫𑩛𑪛
𑪁𑩲𑪘𑩫𑪙𑩬 𑪁𑩖𑩫𑪙𑩬𑩑𑪖 𑩞𑩵𑪙𑩶𑩯𑪙𑩫𑩑, 𑩫𑪖 𑩫𑩔𑪁𑪖 𑩴𑩠𑪙𑩞𑩽𑩴𑩒𑩫𑪘𑩴’’𑩯𑪙𑩫𑩑𑪜
𑩴𑩠𑪙𑩞𑩽𑪁𑩒𑩫𑪘𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
6. 𑩼𑩫𑩯𑪁𑩒𑩫𑪘𑪖
1. 𑩻𑩛𑩯𑩑𑩛𑩮 𑩳𑩒𑩛𑩫𑩛𑩯𑩑 𑪁𑩴𑩛𑩞𑩫𑩛𑩯𑩑, 𑩳𑩒𑩴𑪘𑩛𑩯𑩑 𑩾𑩛 𑩻𑩛𑩯𑩑 𑩾 𑩐𑩯𑪙𑩫𑩽𑩑𑩜𑪙𑩝𑩔𑪛
𑪁𑩲𑪘𑩔𑩾 𑩳𑩒𑩛𑩫𑩛 𑪁𑩒𑩴𑩯𑩛 𑩳𑩾𑩯𑪙𑩫𑩒, 𑩐𑩬𑩖𑩰𑩑 𑪁𑩜𑪘𑩵𑪘 𑪁𑩒𑩪𑩯𑪙𑩫𑩒 𑩳𑩛𑪁𑩑𑩫𑪖𑪜
2. 𑩫𑪁𑪙𑩴𑩛 𑪂𑩑 𑩳𑩒𑩛𑩫𑩛 𑩯𑩑𑪁𑩛𑩴𑩔𑩬 𑪁𑩲𑪘𑩔, 𑩴𑩔𑩫𑪘𑪖 𑩜𑩼𑩖𑩬 𑩴𑩛𑩯𑩒𑪁𑩑𑩻𑩛 𑩰𑩷𑩛𑩻𑪛
𑩭𑩑𑩾𑩛 𑩵 𑩼𑩫𑪘𑩖 𑩵 𑪂𑩼𑩯𑪙𑩫𑩑 𑩻𑩔 𑩲𑩽𑩑𑪖, 𑩫𑪁𑪙𑩴𑩛 𑪂𑩑 𑩯𑩔 𑩼𑩜𑪙𑩝𑩬 𑩐𑩰𑪘𑩴𑩫𑪘𑩛𑪜
3. 𑩻𑪖 𑩜𑩑𑩥𑪙𑩵𑩑 𑩾𑩑𑩫𑪘𑪖 𑩐𑩑𑩮 𑩾𑩛 𑪂𑩒𑩼𑪖 𑩾𑩛, 𑪁𑩞𑪘𑩔𑪁𑩒 𑩾𑩛 𑩻𑪖 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖𑪛
𑩯 𑩯𑩖 𑪁𑩴𑪖 𑩐𑩫𑪙𑩬𑩑 𑩫𑩬𑩛𑩞𑩫𑩔𑩯, 𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑩲𑩒𑩭𑪙𑩮𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖𑪛
𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
4. 𑩝𑩻𑪖 𑩾𑩑𑩼𑩛𑩞𑪖 𑩐𑩴𑩫𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩻𑩭𑩷𑪙𑩤𑩞𑩛 𑪁𑩜𑪙𑩻𑩴𑩒𑩯𑩑𑩛 𑪁𑩴𑩛𑪂𑩑𑩫𑩖𑪛
𑩯 𑩫𑩔𑩯 𑩮𑩴𑪘𑩔𑩯 𑪁𑩴𑩫𑪙𑩬𑩑 𑩜𑩑𑩥𑪙𑩵𑩑, 𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑩮𑩴𑪘𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖𑪛
𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
5. 𑩻𑪖 𑩲𑩒𑩭𑪙𑩮𑪁𑩔𑩦𑪙𑩧𑩖 𑩰𑩼𑩑𑩾𑩪𑪘𑩻𑩑𑩛 𑪁𑩒𑩵𑩑𑪖, 𑪁𑩴𑩛𑩮𑩑𑩴𑩛𑩯𑩯𑪙𑩫𑩼𑩑𑩜𑩥𑪘𑩴𑩛𑪂𑩒𑪛
𑪁𑩴𑩛𑩮𑩑𑩯𑩛 𑩫𑩔𑩯 𑪁𑩴𑩖 𑩯 𑩾𑩑𑩷𑪘𑩫𑩑, 𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑩮𑩴𑪘𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖𑪛
𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
6. 𑩻𑩔 𑩰𑩒𑩞𑪘𑩽𑩛 𑩐𑩦𑪙𑩧 𑪁𑩫𑪖 𑩰𑪁𑩫𑪙𑩬𑩛, 𑩵𑩫𑪘𑩛𑩼𑩑 𑩐𑩔𑩫𑩛𑩯𑩑 𑩻𑩒𑩞𑩛𑩯𑩑 𑪂𑩖𑩯𑪙𑩫𑩑𑪛
𑩫𑩔 𑩭𑩜𑪙𑩝𑩑𑩪𑩔𑩻𑪘𑩛 𑪁𑩒𑩞𑩫𑪁𑪘 𑪁𑩛𑩾𑩜𑩛, 𑩐𑩔𑩫𑩔𑪁𑩒 𑩭𑩑𑩯𑪘𑩛𑩯𑩑 𑩴𑪂𑩰𑪙𑩱𑩽𑩛𑩯𑩑𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
7. 𑩻𑩔 𑪁𑩒𑩰𑪘𑩻𑩒𑩫𑪘𑩛 𑩴𑩯𑪁𑩛 𑩭𑩽𑪙𑪂𑩔𑩯, 𑩯𑩑𑩜𑪘𑩛𑩴𑩑𑩯𑩖 𑩞𑩖𑩫𑩴𑪁𑩛𑪁𑩯𑩴𑪙𑪂𑩑𑪛
𑩫𑩔 𑩰𑩫𑪘𑩑𑩰𑩫𑪘𑩛 𑩐𑩴𑩫𑪖 𑩾𑩑𑩞𑩻𑪙𑪂, 𑩽𑩭𑪙𑩮𑩛 𑩴𑩒𑩮𑩛 𑩯𑩑𑩲𑪘𑩒𑩫𑩑𑪖 𑩳𑩒𑩥𑪙𑩷𑩴𑩛𑩯𑩛𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
8. 𑩻𑩬𑩑𑩯𑪙𑩭𑩝𑩑𑩛𑩽𑩖 𑩰𑩬𑩾𑩑𑪁𑪘𑩑𑩫𑩖 𑪁𑩑𑩻𑩛, 𑩵𑩫𑩒𑩲𑪙𑩳𑩑 𑩾𑩛𑩫𑩔𑪂𑩑 𑩐𑪁𑩴𑪙𑩰𑩜𑩴𑪙𑩰𑩑𑩻𑩖𑪛
𑩫𑩬𑩒𑩛𑩰𑩴𑪖 𑪁𑩰𑪘𑩒𑩼𑩑𑪁𑪖 𑩾𑩭𑩛𑩴𑩑, 𑩻𑩖 𑩐𑩼𑩑𑩻𑪁𑩵𑪘𑩛𑩯𑩑 𑩐𑩾𑩔𑩵𑪘 𑩰𑪁𑪘𑩫𑩑𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
9. 𑩻𑩔 𑩐𑩼𑩑𑩻𑪁𑩵𑪘𑩛𑩯𑩑 𑩾𑩑𑩳𑩛𑩾𑩻𑩯𑪙𑩫𑩑, 𑩞𑩴𑪙𑩳𑩑𑩛𑩼𑩰𑩥𑪘𑩔𑩯 𑪁𑩒𑩭𑩔𑪁𑩑𑩫𑩛𑩯𑩑𑪛
𑩜𑩑𑩥𑪙𑩵𑩛𑩰𑩑 𑩫𑩔 𑪂𑩖𑩯𑪙𑩫𑩑 𑩳𑩒𑪁𑪖 𑩰𑩴𑩫𑪘𑩛, 𑩯 𑩫𑩔 𑩳𑩾𑪖 𑩐𑩦𑪙𑩧𑩴𑩴𑩛𑩭𑩑𑩻𑩯𑪙𑩫𑩑𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
10. 𑪁𑪂𑩛𑩾𑪁𑪘 𑩭𑪁𑪘𑩯𑪁𑩴𑪙𑩰𑩭𑩛𑩻 , 𑩫𑩻𑪁𑪘𑩒 𑩮𑩴𑪘𑩛 𑩷𑪂𑩑𑩫𑩛 𑩳𑩾𑩯𑪙𑩫𑩑𑪛
𑪁𑩜𑪘𑩛𑩻𑩭𑩑𑩦𑪙𑩧𑩑𑩛 𑩾𑩑𑩵𑩑𑩜𑩑𑩵𑪙𑩶𑩑𑩫𑩥𑪙𑩵, 𑪁𑩑𑩛𑩽𑩲𑪘𑩫𑪖 𑩾𑩛𑩰𑩑 𑩻𑩭𑩫𑪙𑩬𑩑 𑩜𑩑𑩥𑪙𑩵𑩑𑪜
11. 𑩵𑩫𑩒𑩛𑪂𑩰𑩛𑩻𑩔𑪂𑩑 𑩵 𑩾𑩑𑩰𑪘𑩴𑩒𑩫𑪘𑩖, 𑩶𑩵𑪘𑩛𑩳𑩑𑩧𑩛𑩯𑩛𑩯𑩑𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
12. 𑩜𑩑𑩥𑪙𑩵𑩛𑩰𑩑 𑪁𑩖 𑩜𑩴𑪘 𑩜𑩼𑩖𑩫𑩑 𑩰𑩛𑩰𑩜𑪖, 𑩜𑩛𑩻𑩔𑩯 𑩾𑩛𑩵𑩛 𑩐𑩒𑩭 𑩵𑩔𑩫𑪁𑩛 𑩾𑩛𑪛
𑩐𑩳𑩲𑪘 , 𑩐𑩳𑩲𑪘𑩫𑩛 𑩭𑩑𑩦𑪙𑩧𑩰𑩭𑪁𑪘 𑩾𑩒𑩫𑪘𑩛𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
13.𑩾𑩯𑩰𑪘𑩞𑩒𑩴𑪙𑩲𑩔 𑩻𑩬 𑩞𑩑𑩴𑪙𑪂𑩔𑪛
𑩫𑩬𑩒𑩛𑩰𑩴𑪖 𑩮𑩴𑪘𑩾𑩼𑪖 𑩐𑩭𑩔𑪁𑩻𑩑 , 𑩯𑩑𑩲𑪘𑩛𑩯𑩞𑩛𑩴𑩑𑪖 𑩰𑩼𑩴𑪖 𑪂𑩑𑩫𑩛𑩻𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑩲𑩒𑩭𑪙𑩮𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
14. 𑩾𑩼𑩖 𑩾𑩼𑩥𑪘𑩒𑩛 𑩾𑩼𑩭𑩖 𑩾𑩼𑩛𑪂𑩼𑩖, 𑩐𑩯𑩒𑩫𑪘𑩼𑩖 𑩮𑩴𑪘𑩾𑩼𑪖 𑩐𑩭𑩔𑪁𑩻𑩑𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑩲𑩒𑩭𑪙𑩮𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
15. 𑩝𑩑𑩛𑩪𑪖 𑩰𑩒𑩼𑩛𑩪𑪖 𑩯𑩾 𑩯𑩫𑪙𑩬𑩑 𑪁𑩴𑪙𑩳𑩾𑪖, 𑩾𑩑𑩼𑩫𑪘𑩵𑩑𑩫𑪘𑩛𑩻𑩫𑩑𑩜𑩔 𑩳𑩾𑪁𑪙𑩴𑩑𑪖𑪛
𑩫𑩔 𑩝𑩑𑩛𑩪𑩲𑩑𑩛𑩷𑩛 𑩐𑩾𑩑𑩼𑩒𑩛𑩽𑪙𑪂𑩑𑩶𑩯𑪙𑩭𑩛, 𑩯𑩑𑩲𑪘𑩯𑪙𑩫𑩑 𑩮𑩑𑩛𑩼𑩛 𑩻𑩬𑩛𑩻𑪖 𑩰𑩭𑩑𑩛𑩰𑩖𑪛
𑩐𑩑𑩭𑩴𑪙𑩰𑩑 𑪁𑩠𑪙𑩟𑩔 𑩼𑩫𑩯𑪖 𑩰𑩪𑩑𑩛𑩫𑪖, 𑩐𑩔𑩫𑩔𑩯 𑪁𑩵𑪘𑩔𑩯 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
16. 𑩻𑩛𑩯𑩑𑩛𑩮 𑩳𑩒𑩛𑩫𑩛𑩯𑩑 𑪁𑩴𑩛𑩞𑩫𑩛𑩯𑩑, 𑩳𑩒𑩴𑪘𑩛𑩯𑩑 𑩾𑩛 𑩻𑩛𑩯𑩑 𑩾 𑩐𑩯𑪙𑩫𑩽𑩑𑩜𑪙𑩝𑩔𑪛
𑩫𑩬𑩛𑩞𑩫𑪖 𑩭𑩔𑩾𑩴𑩯𑩒𑪁𑪘𑩰𑩒𑩛𑩷𑩑𑩫𑪖, 𑩲𑩒𑩭𑪙𑩮𑪖 𑩯𑩴𑪁𑪘𑩛𑩴 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
17. 𑩻𑩛𑩯𑩑𑩛𑩮 𑩳𑩒𑩛𑩫𑩛𑩯𑩑 𑪁𑩴𑩛𑩞𑩫𑩛𑩯𑩑, 𑩳𑩒𑩴𑪘𑩛𑩯𑩑 𑩾𑩛 𑩻𑩛𑩯𑩑 𑩾 𑩐𑩯𑪙𑩫𑩽𑩑𑩜𑪙𑩝𑩔𑪛
𑩫𑩬𑩛𑩞𑩫𑪖 𑩭𑩔𑩾𑩴𑩯𑩒𑪁𑪘𑩰𑩒𑩛𑩷𑩑𑩫𑪖, 𑩮𑩴𑪘𑪖 𑩯𑩴𑪁𑪘𑩛𑩴 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑪜
18. 𑩻𑩛𑩯𑩑𑩛𑩮 𑩳𑩒𑩛𑩫𑩛𑩯𑩑 𑪁𑩴𑩛𑩞𑩫𑩛𑩯𑩑, 𑩳𑩒𑩴𑪘𑩛𑩯𑩑 𑩾𑩛 𑩻𑩛𑩯𑩑 𑩾 𑩐𑩯𑪙𑩫𑩽𑩑𑩜𑪙𑩝𑩔𑪛
𑩫𑩬𑩛𑩞𑩫𑪖 𑩭𑩔𑩾𑩴𑩯𑩒𑪁𑪘𑩰𑩒𑩛𑩷𑩑𑩫𑪖, 𑪁𑩠𑪙𑩟𑪖 𑩯𑩴𑪁𑪘𑩛𑩴 𑪁𑩒𑩾𑩫𑪙𑩬𑩑 𑪂𑩖𑩫𑩒𑩛𑩫𑩑𑪜
𑩼𑩫𑩯𑪁𑩒𑩫𑪘𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
7. 𑩫𑩑𑩼𑩖𑩜𑩒𑩦𑪘𑪁𑩒𑩫𑪘𑪖
1. 𑩫𑩑𑩼𑩖𑩜𑩒𑩦𑪘𑩔𑪁𑩒 𑩫𑩑𑩦𑪙𑩧𑩯𑪙𑩫𑩑, 𑪁𑩯𑪙𑩮𑩑𑪁𑩑𑩠𑪙𑩟𑩛𑩦𑩜𑩔𑪁𑩒 𑩵𑪛
𑩭𑪙𑩾𑩛𑩼𑩲𑩛𑪂𑩛𑪁𑩒 𑩫𑩑𑩦𑪙𑩧𑩯𑪙𑩫𑩑, 𑩐𑩛𑩞𑩯𑪙𑩫𑪙𑩾𑩛𑩯 𑪁𑩜𑪖 𑩟𑩼𑪖𑪜
2. 𑩰𑪂𑩒𑩛𑩫𑩔 𑩐𑩯𑪘𑩰𑩛𑩯𑩴𑪙𑪂𑩑, 𑩝𑩷𑪘𑩳𑩖𑩷𑪘𑩔 𑩐𑩒𑩰𑩦𑪙𑩧𑩑𑩫𑩔𑪛
𑩯 𑩫𑩔𑪁𑪖 𑩜𑩖𑩵𑩑 𑪁𑩼𑩫𑩑, 𑪁𑩫𑪘𑩛𑩯𑪖 𑩜𑩴𑪘𑩰𑩵𑪘𑩻𑩛𑪜
3. 𑩐𑩔𑩾𑪖 𑩭𑩭𑩯𑪙𑩫𑩑 𑩥𑩛𑩫𑩑𑩛𑩯𑪖, 𑩻𑩔 𑪂𑩖𑩯𑪙𑩫𑩑 𑩐𑩯𑩒𑩜𑩴𑪙𑩰𑩜𑩛𑪛
𑪁𑩒𑩵𑩑𑪖 𑩰𑩪𑩑𑩛𑩫𑪖 𑩜𑩛𑩽𑩔𑩯, 𑩜𑩰𑪘𑩑𑩻𑪖 𑩰𑩛𑩯𑩳𑩖𑩷𑩯𑪖𑪛
𑩐𑩑𑩭𑪖 𑩾𑩖 𑩥𑩛𑩫𑩑𑩛𑩯𑪖 𑪂𑩖𑩫𑩒, 𑪁𑩒𑩝𑩑𑩫𑩛 𑪂𑩖𑩯𑪙𑩫𑩒 𑩥𑩛𑩫𑩻𑩖𑪜
4. 𑩫𑩔 𑩵 𑩫𑩫𑪙𑩬 𑪁𑩴𑩛𑩞𑩯𑪙𑩫𑪙𑩾𑩛, 𑩥𑩛𑩫𑩑𑩰𑩔𑩫𑩛 𑪁𑩴𑩛𑩞𑩫𑩛𑪛
𑩰𑪂𑩒𑩛𑩫𑩔 𑩐𑩯𑪘𑩰𑩛𑩯𑩴𑪙𑪂𑩑, 𑪁𑩜𑪘𑩵𑪘𑪖 𑩐𑩯𑩒𑩴𑩖𑩭𑩼𑩔𑪜
5. 𑩵𑩑𑩼𑪖 𑩷𑩑𑩛𑩾𑩯𑪙𑩫𑩒 𑩯𑩖 𑩥𑩛𑩫𑩑𑩛, 𑩻𑩔𑪁𑪖 𑪂𑩔𑩫𑩒 𑩽𑩳𑩛𑩴𑪁𑩔𑪛
𑩐𑩴𑪙𑪂𑩛𑩜𑩥𑪙𑩵 𑩜𑩫𑩛 𑩰𑩒𑩛𑩷𑩛, 𑩭𑩛𑩻𑩜𑩛 𑩵 𑩐𑩯𑩑𑩰𑪙𑩱𑩽𑩛𑪜
6. 𑩯 𑪂𑩑 𑩫𑩫𑪙𑩬 𑩜𑪁𑩑 𑩐𑩫𑪙𑩬𑩑, 𑩞𑩖𑩼𑩜𑪙𑩝𑩔𑩫𑪙𑩬 𑩯 𑩾𑩑𑩷𑪘𑩫𑩑𑪛
𑩾𑩪𑩑𑩷𑪘𑩛 𑩫𑩛𑩭𑩑𑪁𑩑𑩛 𑩯𑩫𑪙𑩬𑩑, 𑪂𑩑𑩼𑩥𑪘𑩔𑩯 𑩜𑩻𑩖𑩜𑩻𑪖𑪛
𑩐𑩑𑩫𑩖 𑩭𑩑𑩯𑪘𑩔𑩯 𑩻𑩛𑩰𑩔𑩯𑪙𑩫𑩑, 𑩰𑩔𑩫𑩛 𑩜𑩛𑩽𑩠𑪙𑩜𑩫𑩛 𑩫𑪂𑩑𑪖𑪜
7. 𑩐𑩒𑩯𑪘𑩴𑩔 𑩐𑩒𑩭𑩜𑪖 𑩾𑩒𑩦𑪙𑩧𑪖, 𑩻𑩬𑩛 𑩯𑩑𑩯𑪘𑪖 𑩰𑩾𑩫𑪘𑩫𑩑𑪛
𑩐𑩔𑩾𑩴𑩔𑩾 𑩐𑩑𑩫𑩖 𑩭𑩑𑩯𑪘𑪖, 𑩰𑩔𑩫𑩛𑩯𑪖 𑩐𑩒𑩰𑩜𑩰𑪘𑩫𑩑𑪜
8. 𑩻𑩬𑩛 𑩾𑩛𑩼𑩑𑩾𑪂𑩛 𑩰𑩒𑩛𑩼𑩛, 𑩰𑩼𑩑𑩰𑩒𑩛𑩼𑩔𑩯𑪙𑩫𑩑 𑪁𑩛𑩞𑩼𑪖𑪛
𑩐𑩔𑩾𑩴𑩔𑩾 𑩐𑩑𑩫𑩖 𑩭𑩑𑩯𑪘𑪖, 𑩰𑩔𑩫𑩛𑩯𑪖 𑩐𑩒𑩰𑩜𑩰𑪘𑩫𑩑𑪜
9. 𑩐𑩭𑩛𑪁𑩑 𑩴𑩔 𑩐𑩜𑩛𑪁𑩑 𑩴𑩔, 𑩥𑩛𑩫𑩑𑩴𑩑𑩫𑪘𑩛 𑪁𑩝𑩛 𑩵 𑩴𑩔𑪛
𑩰𑩔𑩫𑩛𑩯𑪖 𑩭𑩜𑪙𑩝𑩑𑩪𑪖 𑩭𑩷𑪘𑩛, 𑩰𑩒𑩲𑪘𑩔 𑩜𑩫𑩴𑩯𑩒𑪁𑪘𑩼𑪖𑪜
10. 𑩯 𑪂𑩑 𑩼𑩒𑩪𑪘𑪖 𑩾𑩛 𑪁𑩖𑩜𑩖 𑩾𑩛, 𑩻𑩛 𑩵𑩥𑪘𑩛 𑩰𑩼𑩑𑩭𑩔𑩾𑩯𑩛𑪛
𑩯 𑩫𑪖 𑩰𑩔𑩫𑩛𑩯𑩴𑩫𑪙𑩬𑩛𑩻, 𑩐𑩔𑩾𑪖 𑩫𑩑𑩦𑪙𑩧𑩯𑪙𑩫𑩑 𑩥𑩛𑩫𑩻𑩖𑪜
11. 𑩐𑩻𑩥𑪙𑩵 𑩝𑩖 𑩭𑩜𑪙𑩝𑩑𑩪𑩛 𑩭𑩑𑩯𑪘𑩛, 𑪁𑩠𑪙𑩟𑩴𑪙𑪂𑩑 𑪁𑩒𑩰𑪘𑩫𑩑𑩦𑪙𑩧𑩑𑩫𑩛𑪛
𑩭𑩑𑩛𑩟𑩼𑩫𑪘𑪖 𑪂𑩑𑩫𑩛𑩻𑪁𑪘, 𑩧𑩛𑩯𑪁𑩖 𑩐𑩒𑩰𑩜𑩰𑪘𑩫𑩑𑪜
12. 𑪁𑩖 𑩥𑩛𑩫𑩑𑩮𑩴𑪘𑩖 𑩵 𑩐𑩻𑪖 𑩯𑩑𑩭𑪁𑪘𑩑𑩫𑩖, 𑩰𑩔𑩫𑩛𑩯 𑩰𑩒𑩛𑩷𑩛 𑩵 𑩜𑩫𑩛 𑩐𑩒𑩽𑩛𑩼𑩛𑪛
𑩲𑩽𑩥𑪙𑩵 𑩳𑩑𑩜𑪙𑩝𑩒𑩛𑩯𑩴𑩯𑩒𑩰𑪘𑩭𑩑𑩯𑪘𑪖 , 𑩫𑩒𑩴𑪙𑪂𑩔𑪂𑩑 𑩰𑩒𑩥𑪘𑪖 𑩰𑪁𑩒𑩫𑪖 𑩐𑩯𑩰𑪘𑩜𑩯𑪙𑩫𑩑𑪜
𑩫𑩑𑩼𑩖𑩜𑩒𑩦𑪘𑪁𑩒𑩫𑪘𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
8. 𑩯𑩑𑩮𑩑𑩜𑩪𑪙𑩨𑪁𑩒𑩫𑪘𑪖
1. 𑩯𑩑𑩮𑩑𑪖 𑩯𑩑𑩮𑩔𑩫𑩑 𑩰𑩒𑩼𑩑𑪁𑩖, 𑩞𑩴𑪙𑩳𑩑𑩛𑩼𑩔 𑩐𑩖𑩭𑩜𑩯𑪙𑩫𑩑𑩜𑩔𑪛
𑩐𑩫𑪙𑩬𑩔 𑩜𑩑𑩵𑪘𑩔 𑪁𑩴𑩒𑩰𑪘𑩯𑪘𑩔, 𑩐𑩫𑪙𑩬𑩛𑩻 𑩴𑩔 𑩳𑩾𑩑𑪁𑪘𑩫𑩑𑪜
2. 𑩼𑩛𑩷𑩫𑩖 𑩾𑩛 𑩭𑩒𑩼𑩒𑩫𑪘𑪁𑪘, 𑩵𑩖𑩼𑩫𑩖 𑩰𑩑𑩛𑩽𑩑𑩫𑪁𑪘 𑩾𑩛𑪛
𑩐𑩑𑩪𑪁𑪘 𑩾𑩛 𑩰𑩴𑩖𑩜𑪙𑩝𑩛𑩻, 𑩭𑩒𑩲𑪙𑩳𑩑𑩜𑪙𑩝𑩔 𑩐𑩛𑩰𑩭𑩛𑪁𑩒 𑩾𑩛𑪛
𑩐𑩔𑩫𑩭𑩫𑪙𑩬𑩛𑩻 𑩽𑩖𑩜𑪁𑪙𑩴𑩑𑪖, 𑩯𑩑𑩮𑩑 𑩯𑩛𑩴 𑩯𑩑𑩮𑩑𑩛𑩻𑩫𑩑𑪜
3. 𑩫𑩛𑩾𑪁𑪘𑩒𑩯𑩑𑪂𑩑𑩫𑩖 𑪁𑩯𑪙𑩫𑩖, 𑩞𑩴𑪙𑩳𑩑𑩛𑩼𑩔 𑩐𑩖𑩭𑩜𑩯𑪙𑩫𑩑𑩜𑩔𑪛
𑩯 𑪁𑩲𑪘𑩖 𑪁𑩲𑪘𑩭𑩛 𑩐𑩔𑩾, 𑩫𑪁𑪘 𑩫𑪖 𑩐𑩒𑩰𑩜𑩰𑪘𑩫𑩑𑪜
4. 𑩯𑩑𑩮𑩑 𑩾𑩛 𑩧𑩛𑩯𑩛 𑩵𑩾𑩫𑩑, 𑪁𑩥𑪘𑩛 𑩾𑩛𑪁𑪘 𑩾𑩑𑩴𑩒𑩻𑪙𑪂𑩫𑩑𑪛
𑩯𑩛𑩞𑩛 𑩾𑩛 𑩐𑩰𑩯𑩛𑩴𑩔𑩯𑪙𑩫𑩑, 𑩻𑩜𑪙𑩝𑩛 𑩾𑩛𑩰𑩑 𑪂𑩼𑩯𑪙𑩫𑩑 𑩯𑪖𑪜
5. 𑩐𑩰𑪘𑩑𑩻𑩛 𑩾𑩛𑩰𑩑 𑩭𑩛𑩻𑩛𑩭𑩛, 𑩐𑩒𑩭𑪙𑩮𑩼𑩯𑪙𑩫𑩑 𑩐𑩰𑪁𑪘𑩫𑩖𑪛
𑩻𑩭𑩛 𑩰𑩒𑩥𑪘𑩜𑪙𑩝𑩻𑩖 𑪂𑩖𑩫𑩑, 𑪁𑩲𑪘𑩴𑩔𑩫𑪖 𑩾𑩑𑩯𑪁𑪘𑩫𑩑𑪜
6. 𑩻𑪁𑪘 𑩭𑩛𑩯𑩔𑩯 𑪁𑩑𑩛𑩽𑩔𑩯, 𑪁𑪖𑩻𑩴𑩔𑩯 𑩭𑩴𑩔𑩯 𑩵𑪛
𑩯𑩑𑩮𑩑𑩛 𑪁𑩒𑩯𑩑𑪂𑩑𑩫𑩖 𑪂𑩖𑩫𑩑, 𑩐𑩑𑩫𑪙𑩬𑩑𑩻𑩛 𑩰𑩒𑩼𑩑𑪁𑪁𑪘 𑩾𑩛𑪜
7. 𑩵𑩔𑩫𑩑𑩻𑩴𑪙𑪂𑩑 𑩵 𑪁𑩠𑪙𑩟𑩔 𑩾𑩛, 𑩰𑩒𑩞𑪘𑩽𑩔 𑩐𑩫𑩑𑩬𑩑𑩛𑪁𑩒 𑩾𑩛𑪛
𑩴𑩛𑩫𑩼𑩑 𑩰𑩑𑩫𑩼𑩑 𑩵𑩛𑩰𑩑 , 𑩐𑩬𑩖 𑩷𑩔𑩦𑪙𑩧𑩴𑪙𑪂𑩑 𑩳𑩛𑩫𑩼𑩑𑪜
8. 𑩐𑩔𑪁𑩖 𑩯𑩑𑩮𑩑 𑪁𑩒𑩯𑩑𑪂𑩑𑩫𑩖, 𑩐𑩷𑩔𑩻𑪘𑩖 𑩐𑩯𑩒𑩞𑩛𑩴𑩑𑩜𑩖𑪛
𑩰𑪂𑩛𑩻 𑩞𑩴𑩯𑩑𑩛𑩻𑩔𑪁𑩒, 𑩐𑩔𑩫𑪖 𑩐𑩛𑩭𑩛𑩻 𑩞𑩵𑪙𑩶𑩫𑩑𑪜
9. 𑩐𑪁𑩛𑩮𑩛𑩼𑩪𑩴𑩥𑪘𑩔𑪁𑪖, 𑩐𑩵𑩖𑩼𑩛𑪂𑩼𑩪𑩖 𑩯𑩑𑩮𑩑𑪛
𑩜𑩻𑩑𑩼𑩛𑩬 𑩮𑩑𑩛𑩼𑩖 𑩰𑩒𑩥𑪘𑩛𑩯𑩑, 𑩻𑩖 𑩯𑩑𑩮𑩑 𑩐𑩯𑩒𑩞𑩛𑩴𑩑𑩜𑩖𑪜
10. 𑩐𑩔𑪁 𑩭𑩔𑩾𑩴𑩯𑩒𑪁𑪘𑩛𑩯𑪖, 𑪁𑩲𑪘𑩜𑩛𑩴𑩭𑩭𑩖 𑩯𑩑𑩮𑩑𑪛
𑩻𑪖 𑩻𑩭𑩔𑩾𑩛𑩳𑩑𑩰𑩫𑪙𑩬𑩔𑩯𑪙𑩫𑩑, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
11. 𑪁𑩒𑩾𑩪𑪘𑩫𑩛 𑪁𑩒𑪁𑩼𑩫𑩛, 𑪁𑩒𑪁𑩪𑪙𑩧𑩛𑩯𑩛 𑪁𑩒𑩼𑩒𑩛𑩰𑩫𑩛𑪛
𑩐𑩛𑩮𑩑𑩰𑩵𑪘𑩰𑩼𑩑𑩾𑩛𑩼𑩖, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
12. 𑩰𑩭𑩔𑪁𑩼𑩷𑪘𑪖 𑩐𑩑𑪁𑪘𑩼𑩑𑩻𑪖, 𑩵𑩜𑪘𑩾𑩫𑪘𑩑𑪁𑩒𑩝𑪖 𑩰𑩑𑩻𑪖𑪛
𑩭𑩔𑩾𑩼𑩷𑪘𑩴𑪙𑩰𑩑 𑩭𑩑𑩲𑪘𑩔𑪁𑩒, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
13. 𑩴𑩛𑩯𑩒𑪁𑪘𑩑𑩜𑩛 𑩵 𑪁𑩴𑪙𑩰𑩫𑪘𑩑, 𑩭𑩔𑩾𑩽𑩖𑩜𑩔 𑩵 𑩻𑩛 𑩼𑩫𑩑𑪛
𑩻𑩛 𑩵 𑩯𑩑𑩲𑪘𑩛𑩯𑪁𑩴𑪙𑩰𑩫𑪘𑩑, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
14. 𑩴𑩑𑩫𑪘𑪁𑩴𑪙𑩰𑩭𑩴𑩛𑩞𑩴𑪘, 𑩻𑩖𑩯𑩑𑪁𑩖𑩾 𑩰𑩻𑩒𑩥𑪙𑩷𑩫𑩖𑪛
𑩾𑩑𑩷𑪘𑩛 𑩾𑩑𑩴𑩒𑩫𑪘𑩑 𑩾𑪁𑩑𑩛𑩳𑩛𑩾𑩖, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
15. 𑩰𑩦𑩑𑪁𑩴𑪙𑩳𑩑𑩭𑩛 𑩾𑩑𑩴𑩖𑩜𑪙𑩝𑩛 𑩵, 𑩻𑩛 𑩵 𑪁𑩛𑩾𑩜𑩰𑩛𑩼𑩴𑩑𑩛𑪛
𑩰𑩵𑪘𑩔𑩜𑩲𑩖𑩮𑩑 𑩲𑩒𑩭𑪙𑩮𑩳𑩒𑩛𑩴𑩑, 𑪁𑩲𑪘𑩴𑩔𑩫𑩔𑩯 𑩽𑩲𑪙𑩳𑩫𑩑𑪜
16. 𑩐𑩔𑩾𑪖 𑩴𑪂𑩫𑪙𑩬𑩑𑩜𑩛 𑩐𑩔𑪁𑩛, 𑩻𑩭𑩑𑩭𑪖 𑩰𑩒𑩥𑪘𑪁𑩴𑪙𑩰𑩭𑩛𑪛
𑩫𑪁𑪙𑩴𑩛 𑩮𑩑𑩛𑩼𑩛 𑩰𑪁𑪖𑪁𑩯𑪙𑩫𑩑, 𑩰𑩪𑪙𑩨𑩑𑩫𑩛 𑩜𑩫𑩰𑩒𑩥𑪘𑩫𑩯𑪙𑩫𑩑𑪜
𑩯𑩑𑩮𑩑𑩜𑩪𑪙𑩨𑪁𑩒𑩫𑪘𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
9. 𑩴𑩔𑩫𑪘𑪁𑩒𑩫𑪘𑪖
1. 𑩜𑩼𑩪𑩑𑩛𑩻𑩴𑩫𑪙𑩬𑩜𑩒𑪁𑩽𑩔𑩯, 𑩻𑩯𑪙𑩫𑪁𑩯𑪙𑩫𑪖 𑩰𑩭𑪖 𑩐𑩳𑩑𑪁𑩴𑩔𑩵𑪘𑪛
𑪁𑩜𑪘𑩖 𑩐𑩒𑩷𑩒𑩛 𑩵 𑪁𑩒𑪂𑩒𑩷𑩒𑩛 𑩵, 𑪁𑩒𑩾𑩵𑩖 𑩵𑪁𑪘 𑩴𑩒𑩭𑩒 𑩐𑩯𑩫𑩑𑩴𑩛𑩯𑩑𑩛𑪜
2. 𑪁𑩯𑪙𑩫𑩒𑪁𑪘𑩜𑩖 𑩵 𑪁𑩒𑩳𑩼𑩖 𑩵, 𑩐𑩰𑪘𑩜𑩑𑩵𑪘𑩖 𑩵 𑪁𑩽𑪘𑪂𑩒𑩜𑩾𑩒𑩫𑪘𑩑𑪛
𑪁𑩯𑪙𑩫𑩑𑩯𑪙𑩭𑪙𑩼𑩑𑩻𑩖 𑩵 𑩯𑩑𑩰𑩜𑩖 𑩵, 𑩐𑩰𑪘𑩞𑩲𑪙𑩳𑩖 𑩜𑩒𑩽𑩔𑪁𑪙𑩾𑩯𑩯𑩒𑩞𑩑𑩭𑪙𑩮𑩖𑪜
3. 𑩯 𑩵 𑩝𑩒𑩭𑪘𑩴𑩛𑩵𑩼𑩔 𑩜𑩑𑩥𑪙𑩵𑩑, 𑩻𑩔𑩯 𑩾𑩑𑩥𑪘𑩒𑩛 𑩰𑩼𑩔 𑩐𑩒𑩰𑩾𑩭𑩔𑩻𑪘𑩒𑪖𑪛
𑪁𑩒𑩝𑩑𑩯𑩖𑩾 𑩝𑩔𑩴𑩑𑩯𑩖 𑪂𑩖𑩯𑪙𑩫𑩒, 𑪁𑩲𑪘𑪁𑩫𑪘𑩛 𑩳𑩾𑩯𑪙𑩫𑩒 𑪁𑩒𑩝𑩑𑩫𑩫𑪘𑩛𑪜
4. 𑩻𑩔 𑩜𑩔𑩵𑩑 𑩰𑩛𑩪𑩳𑩒𑩛𑩫𑩫𑪙𑩬𑩑, 𑩫𑪁𑩛 𑩾𑩛 𑩬𑩛𑩾𑩼𑩛 𑩾𑩯𑩾𑪁𑩔𑪁𑩛𑪛
𑩭𑩑𑩛𑩟𑩛 𑩾𑩛 𑩻𑩔𑩾 𑩴𑪂𑩯𑪙𑩫𑩛 , 𑩴𑩷𑪙𑩤𑩑𑩴𑩛 𑩼𑪁𑪘𑩜𑩛 𑩐𑩪𑩒𑩜𑩬𑩒𑩛𑩽𑩛𑪜
5. 𑩭𑩑𑩦𑪙𑩧𑩛 𑩾𑩛 𑩻𑩔𑩾 𑩐𑩭𑩑𑩦𑪙𑩧𑩛 𑩭𑩒𑩛𑩼𑩔 𑩾𑪁𑩯𑪙𑩫𑩑 𑩐𑩾𑩑𑩭𑩒𑩛𑩼𑩔𑪛
𑩳𑩒𑩛𑩫𑩛 𑩾 , 𑪁𑩲𑪘𑪁𑩫𑪘𑩛 𑩳𑩾𑩯𑪙𑩫𑩒 𑪁𑩒𑩝𑩑𑩫𑩫𑪘𑩛𑪜
6. 𑩯 𑩰𑩼𑩖 𑩰𑩼𑪖 𑩯𑩑𑩜𑩒𑩲𑪘𑩔𑩬, 𑩯𑩛𑩫𑩑𑩴𑩥𑪘𑩔𑩬 𑩜𑩫𑪙𑩬𑩵𑩑 𑩯 𑩜𑩥𑪙𑩵𑩑𑪛
𑩲𑪙𑩻𑩛𑩼𑩖𑪁𑩯𑩛 𑩰𑩦𑩑𑩟𑪁𑩥𑪘𑩛, 𑩯𑩛𑩥𑪘𑩴𑩥𑪘𑪁𑪘 𑩭𑩒𑩜𑪙𑩝𑩴𑩑𑩵𑪙𑩶𑩔𑩻𑪘𑪜
7. 𑩴𑩛𑩫𑩛 𑩻𑩬𑩛 𑩯𑩑𑩻𑪖 𑩰𑩒𑩫𑪘𑩴𑩛𑩻𑩒𑪁𑩛 𑩐𑩔𑩜𑩰𑩒𑩫𑪘𑩴𑩯𑩒𑩼𑩜𑪙𑩝𑩔𑪛
𑩐𑩔𑩾𑩴𑪙𑩰𑩑 𑪁𑩲𑪘𑩳𑩒𑩛𑩫𑩔𑪁𑩒, 𑩴𑩛𑩯𑪁𑪖 𑩳𑩛𑩾𑩻𑩔 𑩐𑩰𑩼𑩑𑩴𑩛𑩪𑪖𑪜
8. 𑩴𑩔𑩫𑪘𑩥𑪙𑩵 𑪁𑩲𑪘𑩽𑩖𑩜𑪁𑪙𑩴𑩑, 𑩴𑩛𑩯𑪁𑪖 𑩳𑩛𑩾𑩻𑩔 𑩐𑩰𑩼𑩑𑩴𑩛𑩪𑪖𑪛
𑩐𑩒𑩭𑪙𑩮𑪖 𑩐𑩮𑩖 𑩵 𑩫𑩑𑩼𑩑𑩻𑩥𑪙𑩵, 𑩐𑪁𑩴𑪙𑩲𑩛𑩮𑪖 𑩐𑩾𑩔𑩼𑩴𑪁𑩰𑩫𑪘𑪖𑪜
9. 𑩫𑩑𑩦𑪙𑩧𑪖 𑩵𑩼𑪖 𑩯𑩑𑪁𑩑𑩯𑪘𑩖 𑩾, 𑪁𑩻𑩛𑩯𑩖 𑩻𑩛𑩾𑩫𑩛𑪁𑪘 𑩾𑩑𑩫𑩴𑩑𑩭𑪙𑩮𑩖𑪛
𑩐𑩔𑩫𑪖 𑪁𑩫𑩑𑪖 𑩐𑩮𑩑𑩦𑪙𑩧𑩔𑩻𑪘, 𑩲𑪙𑩼𑪂𑪙𑩴𑩴𑩔𑩫𑪖 𑩾𑩑𑪂𑩛𑩼𑩴𑩑𑩮𑩴𑩛𑪂𑩒𑪜
10. 𑩭𑩑𑩦𑪙𑩧𑩑𑩥𑪙𑩵 𑩐𑩯𑩒𑩰𑩞𑪘𑩴𑪘, 𑪁𑩑𑩛𑩽𑩾𑩛 𑩭𑪁𑪘𑩯𑩔𑩯 𑪁𑩴𑪙𑩰𑩯𑪘𑩖𑪛
𑩜𑩛𑩴𑩔𑪁𑩒 𑩾𑩑𑩯𑩻 𑩞𑩔𑩮𑪖, 𑩯 𑪂𑩑 𑩷𑩛𑩫𑩒𑩞𑪘𑩲𑪙𑩳𑪁𑩔𑩻𑪘 𑩰𑩒𑩯 𑩼𑩔𑩫𑩑𑩛𑩫𑩑𑪜
𑩴𑩔𑩫𑪘𑪁𑩒𑩫𑪘𑪖 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑪖𑪛
𑩝𑩒𑩭𑪘𑩜𑩰𑩛𑩧𑩰𑩛𑩽𑩑 𑩯𑩑𑩦𑪙𑩧𑩑𑩫𑩛𑪛
