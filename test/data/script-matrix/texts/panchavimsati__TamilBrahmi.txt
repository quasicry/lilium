𑀧𑀜𑁆𑀘𑀯𑀺𑀫𑁆𑀰𑀢𑀺𑀲𑀸𑀳𑀲𑁆𑀭𑀺𑀓𑀸 𑀧𑁆𑀭𑀚𑁆𑀜𑀸𑀧𑀸𑀭𑀫𑀺𑀢𑀸
𑀧𑀼𑀦𑀭𑁆 𑀅𑀧𑀭𑀫𑁆 𑀲𑀼𑀧𑀽𑀢𑁂 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬 𑀫𑀳𑀸𑀬𑀸𑀦𑀫𑁆. 𑀬𑀢𑀼𑀢 𑀥𑀸𑀭𑀡𑀻𑀫𑀼𑀓𑀸𑀦𑀺. 𑀬𑀢𑀼𑀢𑀸𑀓𑁆𑀱𑀭𑀦𑀬𑀲𑀫𑀢𑀸𑀓𑁆𑀱𑀭𑀫𑀼𑀓𑀫𑀓𑁆𑀱𑀭𑀧𑁆𑀭𑀯𑁂𑀰𑀂. 𑀓𑀢𑀫𑁄'𑀓𑁆𑀱𑀭𑀦𑀬𑀲𑀫𑀢𑀸 𑀅𑀓𑁆𑀱𑀭𑀫𑀼𑀓𑀫𑀓𑁆𑀱𑀭𑀧𑁆𑀭𑀯𑁂𑀰𑀂.
𑀅𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀆𑀢𑁆𑀬𑀦𑀼𑀢𑁆𑀧𑀦𑁆𑀦𑀢𑁆𑀯𑀸𑀢𑁆.
𑀭𑁂𑀧𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀭𑀚𑁄'𑀧𑀓𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀧𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀧𑀭𑀫𑀸𑀭𑁆𑀢𑀦𑀺𑀭𑁆𑀢𑁂𑀰𑀸𑀢𑁆.
𑀘𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀘𑁆𑀬𑀯𑀦𑁄𑀧𑀧𑀢𑁆𑀢𑁆𑀬𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀦𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀦𑀸𑀫𑀸𑀧𑀓𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀮𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀮𑁄𑀓𑁄𑀢𑁆𑀢𑀻𑀭𑁆𑀡𑀢𑁆𑀯𑀸𑀢𑁆 𑀢𑁆𑀭𑀼𑀱𑁆𑀡𑀸𑀮𑀢𑀸𑀳𑁂𑀢𑀼𑀧𑁆𑀭𑀢𑁆𑀬𑀬𑀲𑀫𑀼𑀢𑁆𑀓𑀸𑀢𑀺𑀢𑁆𑀯𑀸𑀢𑁆 𑀘.
𑀢𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀢𑀸𑀦𑁆𑀢𑀢𑀫𑀢𑀧𑀭𑀺𑀘𑁆𑀘𑀺𑀦𑁆𑀦𑀢𑁆𑀯𑀸𑀢𑁆.
𑀧𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀧𑀦𑁆𑀥𑀦𑀯𑀺𑀫𑀼𑀓𑁆𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀝𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀝𑀫𑀭𑀸𑀧𑀓𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀲𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀲𑀗𑁆𑀓𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀯𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀯𑀸𑀓𑁆𑀧𑀢𑀓𑁄𑀱𑀲𑀫𑀼𑀘𑁆𑀘𑀺𑀦𑁆𑀦𑀢𑁆𑀯𑀸𑀢𑁆.
𑀢𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀢𑀢𑀢𑀸'𑀘𑀮𑀺𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀬𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀬𑀢𑀸𑀯𑀢𑁆 𑀅𑀦𑀼𑀢𑁆𑀧𑀸𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀲𑁆𑀢𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀲𑁆𑀢𑀫𑁆𑀧𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀓𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀓𑀸𑀭𑀓𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀲𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀲𑀫𑀢𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀫𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀫𑀫𑀓𑀸𑀭𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑁆𑀯𑀸𑀢𑁆.
𑀓𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀓𑀓𑀦𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀲𑁆𑀢𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀲𑁆𑀢𑀸𑀦𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀚𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀚𑀸𑀢𑁆𑀬𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀰𑁆𑀯𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀰𑁆𑀯𑀸𑀲𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀥𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀥𑀭𑁆𑀫𑀥𑀸𑀢𑁆𑀯𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀰𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀰𑀫𑀢𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀓𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀓𑀲𑀫𑀢𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀓𑁆𑀱𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀡𑀸𑀫𑁆 𑀓𑁆𑀱𑀬𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀢𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀢𑀘𑁆 𑀘𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀚𑁆𑀜𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀲𑀭𑁆𑀯𑀚𑁆𑀜𑀸𑀦𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀳𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀳𑁂𑀢𑁄𑀭𑁆 𑀅𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀘𑁆𑀘𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀘𑀯𑁂𑀭𑁆 𑀅𑀧𑁆𑀬𑁆 𑀅𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀲𑁆𑀫𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀲𑁆𑀫𑀭𑀡𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀢𑁆𑀥𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸 𑀆𑀢𑁆𑀥𑀸𑀦𑀸𑀧𑀓𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀲𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸 𑀉𑀢𑁆𑀲𑀸𑀳𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀓𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀓𑀦𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀝𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀯𑀺𑀝𑀧𑀦𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀡𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀭𑀡𑀯𑀺𑀓𑀢𑀢𑁆𑀯𑀸𑀢𑁆.
𑀧𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀧𑀮𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀲𑁆𑀓𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀲𑁆𑀓𑀦𑁆𑀥𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀚𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀚𑀭𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀘𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀘𑀭𑀡𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀝𑀓𑀸𑀭𑀫𑀼𑀓𑀸𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀝𑀫𑁆𑀓𑀸𑀭𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀝𑀓𑀸𑀭𑁄 𑀫𑀼𑀓𑀂 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸𑀂 𑀝𑀫𑁆𑀓𑀸𑀭𑀸𑀦𑀼𑀧𑀮𑀧𑁆𑀥𑀺𑀢𑀂.
𑀦𑀸𑀲𑁆𑀢𑀺 𑀅𑀢 𑀉𑀢𑁆𑀢𑀭𑀺 𑀅𑀓𑁆𑀱𑀬𑀯𑁆𑀬𑀯𑀳𑀸𑀭𑀂. 𑀢𑀢𑁆𑀓𑀲𑁆𑀬 𑀳𑁂𑀢𑁄𑀂. 𑀢𑀢𑀸 𑀳𑀺 𑀦 𑀓𑀲𑁆𑀬𑀘𑀺𑀦𑁆𑀦𑀸𑀫𑀸𑀲𑁆𑀢𑀺 𑀬𑁂𑀦 𑀲𑀫𑁆𑀯𑁆𑀬𑀯𑀳𑁆𑀭𑀺𑀬𑁂𑀢 𑀬𑁂𑀦 𑀯𑀸𑀧𑀺𑀮𑀧𑁆𑀬𑁂𑀢 𑀬𑁂𑀦 𑀦𑀺𑀭𑁆𑀢𑀺𑀰𑁆𑀬𑁂𑀢 𑀬𑁂𑀦 𑀮𑀓𑁆𑀱𑁆𑀬𑁂𑀢 𑀬𑁂𑀦 𑀧𑀰𑁆𑀬𑁂𑀢𑁆 . 𑀢𑀢𑁆𑀬𑀢𑀸𑀧𑀺 𑀦𑀸𑀫 𑀲𑀼𑀧𑀽𑀢𑁂 𑀆𑀓𑀸𑀰𑀫𑁆. 𑀏𑀯𑀫𑁂𑀯 𑀲𑀭𑁆𑀯𑀥𑀭𑁆𑀫𑀸 𑀅𑀦𑀼𑀓𑀦𑁆𑀢𑀯𑁆𑀬𑀸𑀂. 𑀅𑀬𑀫𑁆 𑀲𑀼𑀧𑀽𑀢𑁂 𑀥𑀸𑀭𑀡𑀻𑀧𑁆𑀭𑀯𑁂𑀰𑁄'𑀓𑀸𑀭𑀸𑀢𑁆𑀬𑀓𑁆𑀱𑀭𑀦𑀺𑀭𑁆𑀢𑁂𑀰𑀧𑁆𑀭𑀯𑁂𑀰𑀂. 𑀬𑀂 𑀓𑀰𑁆𑀘𑀺𑀢𑁆 𑀲𑀼𑀧𑀽𑀢𑁂 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀂 𑀇𑀢𑀫𑀓𑀸𑀭𑀸𑀢𑁆𑀬𑀓𑁆𑀱𑀭𑀓𑁅𑀰𑀮𑁆𑀬𑀧𑁆𑀭𑀯𑁂𑀰𑀫𑁆 𑀚𑁆𑀜𑀸𑀲𑁆𑀬𑀢𑀺 𑀦 𑀲 𑀓𑁆𑀯𑀘𑀺𑀢𑁆𑀭𑀼𑀢𑁂 𑀧𑁆𑀭𑀢𑀺𑀳𑀦𑁆𑀬𑀢𑁆 𑀲𑀭𑁆𑀯𑀫𑁆 𑀢𑀫𑁆 𑀥𑀭𑁆𑀫𑀢𑀬𑀸 𑀲𑀫𑀸𑀥𑀬𑀺𑀱𑁆𑀬𑀢𑀺 𑀭𑀼𑀢𑀚𑁆𑀜𑀸𑀦𑀓𑁅𑀰𑀮𑁆𑀬𑀜𑁆𑀘 𑀧𑁆𑀭𑀢𑀺𑀮𑀧𑁆𑀲𑁆𑀬𑀢𑁆 𑀬𑁄 𑀳𑀺 𑀓𑀰𑁆𑀘𑀺𑀢𑁆𑀲𑀼𑀧𑀽𑀢𑁂 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑁄 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯 𑀇𑀫𑀸𑀫𑀓𑀸𑀭𑀸𑀢𑁆𑀬𑀓𑁆𑀱𑀭𑀫𑀼𑀢𑁆𑀭𑀸𑀫𑁆 𑀰𑁆𑀭𑁄𑀱𑁆𑀬𑀢𑀺 𑀰𑁆𑀭𑀼𑀢𑁆𑀯𑀸 𑀘𑁄𑀢𑁆𑀓𑁆𑀭𑀳𑀻𑀱𑁆𑀬𑀢𑀺 𑀥𑀸𑀭𑀬𑀺𑀱𑁆𑀬𑀢𑀺 𑀯𑀸𑀘𑀬𑀺𑀱𑁆𑀬𑀢𑀺 𑀧𑀭𑁂𑀱𑀸𑀫𑁆 𑀢𑁂𑀰𑀬𑀺𑀱𑁆𑀬𑀢𑀺 𑀭𑀫𑀬𑀢𑀺 𑀢𑀬𑀸 𑀲𑀦𑁆𑀢𑀢𑁆𑀬𑀸 𑀢𑀲𑁆𑀬 𑀯𑀺𑀫𑁆𑀰𑀢𑀺𑀭𑀦𑀼𑀰𑀫𑁆𑀲𑀸𑀂 𑀧𑁆𑀭𑀢𑀺𑀓𑀸𑀫𑁆𑀓𑁆𑀱𑀺𑀢𑀯𑁆𑀬𑀸𑀂. 𑀓𑀢𑀫𑀸 𑀯𑀺𑀫𑁆𑀰𑀢𑀺𑀂. 𑀲𑁆𑀫𑁆𑀭𑀼𑀢𑀺𑀫𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀫𑀢𑀺𑀫𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀓𑀢𑀺𑀫𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀥𑁆𑀭𑀼𑀢𑀺𑀫𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀳𑁆𑀭𑀻𑀫𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑁆𑀭𑀚𑁆𑀜𑀸𑀯𑀸𑀫𑁆𑀰𑁆𑀘 𑀧𑁆𑀭𑀢𑀺𑀧𑀸𑀦𑀯𑀸𑀫𑁆𑀰𑁆𑀘 𑀲𑀢𑁆𑀬𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀇𑀢𑀫𑁆 𑀥𑀸𑀭𑀡𑀻𑀫𑀼𑀓𑀫𑀮𑁆𑀧𑀓𑁆𑀭𑀼𑀘𑁆𑀘𑁆𑀭𑁂𑀡 𑀧𑁆𑀭𑀢𑀺𑀮𑀧𑁆𑀲𑁆𑀬𑀢𑁂 𑀦 𑀘 𑀓𑀢𑀫𑁆𑀓𑀢𑀻 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀯𑀺𑀫𑀢𑀺𑀰𑁆𑀘𑀸𑀲𑁆𑀬 𑀦 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀦 𑀘 𑀰𑁆𑀮𑀓𑁆𑀱𑁆𑀡𑀸𑀫𑁆 𑀯𑀸𑀘𑀫𑁆 𑀰𑁆𑀭𑀼𑀢𑁆𑀯𑀸 𑀧𑀭𑁂𑀱𑁆𑀯𑀦𑀼𑀦𑀻𑀢𑁄 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀦 𑀘 𑀧𑀭𑀼𑀱𑀬𑀸 𑀯𑀸𑀘𑀸 𑀧𑁆𑀭𑀢𑀺𑀳𑀦𑁆𑀬𑀢𑁆 𑀦 𑀘𑁄𑀦𑁆𑀦𑀢𑁄 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀦𑀸𑀯𑀦𑀢𑀂. 𑀬𑀢𑀸𑀲𑁆𑀢𑀺𑀢 𑀏𑀯 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀭𑀼𑀢𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀲𑁆𑀓𑀦𑁆𑀥𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀥𑀸𑀢𑀼𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀆𑀬𑀢𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀲𑀢𑁆𑀬𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀧𑁆𑀭𑀢𑀻𑀢𑁆𑀬𑀲𑀫𑀼𑀢𑁆𑀧𑀸𑀢𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀳𑁂𑀢𑀼𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀧𑁆𑀭𑀢𑁆𑀬𑀬𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀥𑀭𑁆𑀫𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀇𑀦𑁆𑀢𑁆𑀭𑀺𑀬𑀧𑀭𑀺𑀧𑀽𑀭𑁆𑀡𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀧𑀭𑀘𑀺𑀢𑁆𑀢𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀭𑀼𑀢𑁆𑀥𑀺𑀯𑀺𑀥𑀺𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀢𑀺𑀯𑁆𑀬𑀰𑁆𑀭𑁄𑀢𑁆𑀭𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀧𑀽𑀭𑁆𑀯𑀦𑀺𑀯𑀸𑀲𑀸𑀦𑀼𑀲𑁆𑀫𑁆𑀭𑀼𑀢𑀺𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀘𑁆𑀬𑀼𑀢𑁄𑀧𑀧𑀢𑁆𑀢𑀺𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀆𑀲𑁆𑀭𑀯𑀓𑁆𑀱𑀬𑀚𑁆𑀜𑀸𑀦𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀲𑁆𑀢𑀸𑀦𑀸𑀲𑁆𑀢𑀸𑀦𑀦𑀺𑀭𑁆𑀢𑁂𑀰𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀅𑀢𑀺𑀓𑁆𑀭𑀫𑀡𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺 𑀇𑀭𑁆𑀬𑀸𑀧𑀢𑀓𑀼𑀰𑀮𑀰𑁆𑀘 𑀧𑀯𑀺𑀱𑁆𑀬𑀢𑀺. 𑀇𑀫𑀸𑀦𑀦𑀼𑀰𑀫𑁆𑀲𑀸𑀦𑁆 𑀧𑁆𑀭𑀢𑀺𑀮𑀧𑁆𑀲𑁆𑀬𑀢𑁂 𑀇𑀢𑀺. 𑀇𑀢𑀫𑀧𑀺 𑀲𑀼𑀧𑀽𑀢𑁂 𑀥𑀸𑀭𑀡𑀻𑀫𑀼𑀓𑀫𑀓𑁆𑀱𑀭𑀫𑀼𑀓𑀫𑀓𑀸𑀭𑀫𑀼𑀓𑀫𑁆 𑀧𑁄𑀥𑀺𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬 𑀫𑀳𑀸𑀲𑀢𑁆𑀢𑁆𑀯𑀲𑁆𑀬 𑀫𑀳𑀸𑀬𑀸𑀦𑀫𑁆. 𑀇𑀢𑀺 𑀥𑀸𑀭𑀡𑀻𑀲𑀫𑁆𑀧𑀸𑀭𑀂..
