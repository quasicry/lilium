༡. 𑲍𑲊𑱽𑱺𑲚𑲉𑲵
𑲀𑲲𑱼𑲜𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑱼𑲮𑲁𑲡𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑲍𑱵𑲔𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱𑱱
𑱼𑲲𑱺𑲱𑲉𑲁𑲞𑲱 𑲀𑲲𑱼𑲜𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑱼𑲲𑱺𑲱𑲉𑲁𑲞𑲱 𑱼𑲮𑲁𑲡𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑱼𑲲𑱺𑲱𑲉𑲁𑲞𑲱 𑲍𑱵𑲔𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱𑱱
𑱺𑱺𑲱𑲉𑲁𑲞𑲱 𑲀𑲲𑱼𑲜𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑱺𑱺𑲱𑲉𑲁𑲞𑲱 𑱼𑲮𑲁𑲡𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱
𑱺𑱺𑲱𑲉𑲁𑲞𑲱 𑲍𑱵𑲔𑲮𑲵 𑲍𑲊𑱽𑲵 𑱴𑲂𑲣𑲰𑲁𑲱𑱱𑱱
𑲍𑲊𑱽𑱺𑲚𑲉𑲵 𑱽𑲱𑱺𑲛𑲱𑱺𑲵𑱱
༢. 𑱼𑲍𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵
༡. 𑱾𑲰𑱽𑲰𑱺𑲱𑱾𑲰𑱺𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༢. 𑲏𑱼𑲱𑱽𑲝𑲰𑱼𑲰𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༣. 𑲏𑲀𑲪𑲎𑲡𑲂𑲊𑲱𑲉𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༤. 𑲁𑲲𑲍𑲰𑲅𑲰𑱼𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༥. 𑲍𑲲𑲊𑲰𑲁𑲳𑲊𑲉𑲁𑲄𑲤𑱾𑲁𑲰𑱼𑱺𑲛𑲰𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༦. 𑲅𑲱𑱲𑲰𑲋𑲀𑲮𑲴𑲄𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༧. 𑱽𑲂𑲢-𑱴𑲱𑲰𑱺-𑲅𑲰𑱼𑲱𑱺-𑲅𑲱𑲍𑲲𑲰𑱲𑱼𑲍𑲭𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༨. 𑲁𑲰𑲋𑲰-𑱴𑱽𑲜𑲮-𑲅𑲱𑲋𑲳𑱾𑱽-𑱼𑲮𑲰𑲊𑱽-𑲁𑱽𑲜𑱽-𑲅𑲱𑲀𑲮𑲲𑲰𑲍𑱽𑱺𑲛𑲰𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༩. 𑲏𑲲𑲂𑲢𑲰𑲍𑲉𑱽-𑲁𑲎𑲰𑲍𑲉𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
༡༠. 𑲄𑲰𑱺𑲊𑲲𑲰𑱾-𑲊𑲄𑱺𑱾𑱺𑲱𑱴𑲔𑲎𑱽𑲰 𑲅𑲳𑲊𑲁𑱽𑲱𑲰-𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑲍𑲁𑲰𑱼𑲱𑲉𑲰𑲁𑲱𑱱
𑱼𑲍𑲍𑲱𑱲𑲓𑲰𑱾𑱼𑲵 𑱽𑲱𑱺𑲛𑲱𑱺𑲵𑱱
༣. 𑱼𑲥𑱺𑲚𑲱𑲵𑲍𑲰𑱲𑲰𑲊𑲴
𑲏𑱺𑲛𑲱 𑲏𑲱𑲁𑲍𑲡𑲱𑲵 𑱲𑲰𑲉𑲳 –
𑱲𑲳𑲍𑲰 𑲋𑲴𑲁𑲰 𑱽𑱳𑲰 𑱼𑱽𑲚𑲰 𑱺𑲂𑲴,
𑲁𑲵𑲍𑲵 𑱽𑲮𑲰𑲊𑲲 𑲏𑱺𑲛𑲱 𑲏𑱺𑲛𑲱𑲁𑲱𑱹𑲤𑲵 𑲅𑱲𑲒𑲵,
𑲎𑱼𑲉𑲵 𑲉𑱲𑱽𑲵 𑱲𑲱𑲋𑲴𑲁𑱲𑲵 𑱾𑲱𑲎𑱲𑲵 𑱾𑱾𑲟𑲰𑲍𑲵,
𑲏𑱽𑲚𑲵 𑲏𑱽𑲚𑱴𑲲𑱽𑲵 𑲏𑲲𑱼𑲊𑲱𑲉𑲵 𑱲𑲊𑲱𑲰𑲍𑲵 𑲁𑱺𑲛𑲋𑲲𑱵𑲔𑲵,
𑱾𑲱𑱺𑲚𑲵 𑲍𑲳𑲁𑲮𑲵 𑱾𑲲𑲀𑲠𑲴 𑲋𑲴𑲎𑲱𑱺𑲵 𑲍𑲳𑱼𑲴 𑲁𑲳𑱼𑲴,
𑲏𑲍𑲭𑲲 𑲅𑲍𑲰 𑱳𑲳𑲋𑲴 𑲍𑲱𑱵𑲔𑲮𑲰𑱽𑲱𑱲𑲰 𑲋𑲍𑲱𑱲𑲰 𑲁𑲲𑱺𑲚𑱽𑲚𑲱 𑱱𑱱
𑱼𑲥𑱺𑲚𑲱𑲵𑲍𑲰𑱲𑲰𑲊𑲴 𑱽𑲱𑱺𑲛𑲱𑱺𑲴𑱱
༤. 𑱲𑲲𑲁𑲰𑲊𑱾𑱹𑲮𑲰
༡. ‘‘𑲏𑲳𑱲𑲵 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑲍𑲀𑲠𑲳 𑲍𑱺𑲚𑲰 𑲏𑲰𑲎𑲰𑲊𑱺𑲛𑲱𑱺𑲱𑱲𑲰’’𑱱
༢. ‘‘𑱼𑲥𑲳 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑱽𑲰𑲁𑱹𑲢 𑲊𑲲𑲰𑱾𑱹𑲢’’𑱱
༣. ‘‘𑱺𑲱𑲰𑱽𑲱 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑱺𑲱𑲍𑲭𑲴 𑲅𑲳𑱼𑱽𑲰’’𑱱
༤. ‘‘𑲂𑱺𑲚𑲰𑲊𑲱 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑲂𑱺𑲚𑲰𑲊𑲱 𑲏𑲊𑲱𑲉𑲍𑲂𑲢𑲰𑱽𑲱’’𑱱
༥. ‘‘𑱾𑱹𑲢 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑱾𑱹𑲢𑲲𑱾𑲰𑱼𑲰𑱽𑱲𑲓𑱽𑲜𑲮𑲰’’𑱱
༦. ‘‘𑲃 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑲃 𑲏𑲄𑲤𑲮𑱺𑲚𑲱𑱲𑲰𑱽𑲱 𑲏𑲰𑲉𑱺𑱽𑲰𑱽𑲱’’𑱱
༧. ‘‘𑲍𑱺𑲚 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑲍𑱺𑲚 𑲀𑲴𑲄𑲤𑲮𑱵𑲔𑲰’’𑱱
༨. ‘‘𑲏𑱺𑲛 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑲏𑲊𑲱𑲉𑲴 𑲏𑱺𑲛𑱵𑲔𑲱𑱲𑲴 𑲁𑱴𑲔𑲴’’𑱱
༩. ‘‘𑱽𑲅 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑱽𑲅 𑲍𑱺𑲚𑲰𑲅𑲰𑲍𑲰’’𑱱
༡༠. ‘‘𑱼𑲍 𑱽𑲰𑲁 𑱲𑲱𑲵’’? ‘‘𑱼𑲍𑲎𑱵𑲔𑲳𑲎𑲱 𑲍𑲁𑱽𑲝𑲰𑱴𑱺𑲴 ‘𑲏𑲊𑲎𑲰’𑱺𑲱 𑲅𑲲𑲂𑲢𑱺𑲱𑲰’’𑱺𑲱𑱱
𑱲𑲲𑲁𑲰𑲊𑱾𑱹𑲮𑲰 𑱽𑲱𑱺𑲛𑲱𑱺𑲰𑱱
༥. 𑲁𑱵𑲔𑲋𑲍𑲲𑱺𑲚𑲵
༡. 𑲏𑲳𑲅𑲵 𑲁𑲳 𑲍𑲲𑱺𑲵 – 𑲏𑲳𑱲𑲵 𑲍𑲁𑲉𑲵 𑲀𑲮𑱴𑲅𑲰 𑲍𑲰𑲅𑱺𑲛𑲱𑲉𑲵 𑲅𑲱𑲎𑲊𑱺𑲱 𑲄𑲳𑱺𑲅𑱽𑲳 𑲏𑱽𑲰𑱻𑱾𑲱𑱽𑲜𑲱𑱲𑲍𑲭 𑲏𑲰𑲊𑲰𑲁𑲳𑱱 𑲏𑱻 𑱳𑲴 𑲏𑱹𑲙𑱺𑲊𑲰 𑱼𑲳𑲅𑱺𑲰 𑲏𑲀𑲮𑲱𑱲𑲒𑱽𑲚𑲰𑲉 𑲊𑱺𑲚𑲱𑲉𑲰 𑲏𑲀𑲮𑲱𑱲𑲒𑱽𑲚𑲅𑱽𑲝𑲰 𑱲𑲳𑲅𑲋𑱲𑱾𑲞𑲵 𑲄𑲳𑱺𑲅𑱽𑲵 𑲏𑲴𑲀𑲮𑲰𑲍𑲳𑱺𑲥𑲰 𑲉𑲳𑱽 𑲀𑲮𑱴𑲅𑲰 𑱺𑲳𑱽𑲲𑱾𑲍𑱵𑲒𑲁𑲱; 𑲏𑲲𑱾𑲍𑱵𑲒𑲁𑲱𑱺𑲥𑲰 𑲀𑲮𑱴𑲅𑱽𑲚𑲵 𑲏𑲀𑲮𑲱𑲅𑲰𑱼𑲳𑱺𑲥𑲰 𑲏𑲳𑱲𑲁𑱽𑲚𑲵 𑲏𑱺𑲛𑲰𑲍𑲱𑱱 𑲏𑲳𑱲𑲁𑱽𑲚𑲵 𑱻𑲱𑱺𑲰 𑱳𑲴 𑲍𑲰 𑱼𑲳𑲅𑱺𑲰 𑲀𑲮𑱴𑲅𑱽𑲚𑲵 𑱴𑲰𑱻𑲰𑲉 𑲏𑲄𑲤𑲮𑲀𑲮𑲰𑲍𑲱 –
༢. ‘‘𑲀𑲎𑲲𑲰 𑱼𑲳𑲅𑲰 𑲁𑱽𑲲𑲍𑲭𑲰 𑲂, 𑲁𑱵𑲔𑲋𑲰𑱽𑲱 𑲏𑲂𑲱𑱽𑲚𑲉𑲲𑲵𑱱
𑲏𑲰𑱲𑱵𑲓𑲁𑲰𑱽𑲰 𑲍𑲴𑱺𑲛𑲰𑱽𑲵, 𑲀𑲪𑲲𑲰𑲎𑲱 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵’’𑱱𑱱
༣. ‘‘𑲏𑲍𑲳𑲅𑱽𑲰 𑲂 𑲀𑲰𑲋𑲰𑱽𑲵, 𑱾𑱽𑲜𑲱𑱺𑲰𑱽𑱹𑲢 𑲍𑲳𑲅𑱽𑲰𑱱
𑱾𑲲𑲰𑲄𑲰 𑲂 𑱾𑲲𑲰𑲄𑱽𑲳𑲉𑲩𑲰𑱽𑲵 , 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱
༤. ‘‘𑱾𑱺𑲱𑲊𑲲𑲰𑱾𑱼𑲳𑲍𑲅𑲰𑲍𑲴 𑲂, 𑱾𑲲𑲀𑲠𑲳 𑲂 𑱲𑱺𑱾𑲲𑱹𑲙𑱺𑲰𑱱
𑲏𑱺𑲚𑲍𑲁𑲡𑲰𑱾𑱽𑲱𑱼𑲮𑲱 𑲂, 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱
༥. ‘‘𑲀𑲰𑲎𑲲𑲍𑲂𑲢𑱹𑲢 𑲍𑲱𑱾𑲞𑱹𑲢, 𑲅𑲱𑱽𑲉𑲴 𑲂 𑲍𑲲𑲍𑲱𑱲𑲓𑲱𑱺𑲴𑱱
𑲍𑲲𑲀𑲮𑲰𑲍𑲱𑱺𑲰 𑲂 𑲉𑲰 𑲅𑲰𑲂𑲰, 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱
༦. ‘‘𑲁𑲰𑱺𑲰𑱾𑲱𑱺𑲲 𑲏𑲲𑱾𑱺𑲛𑲰𑱽𑲵, 𑱾𑲲𑱺𑲚𑱼𑲰𑲊𑲍𑲭 𑲍𑱵𑲔𑲎𑲴𑱱
𑲏𑱽𑲰𑱲𑲲𑲋𑲰 𑲂 𑱲𑲁𑲡𑱽𑲚𑲰, 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱
༧. ‘‘𑱼𑲰𑱽𑱹𑲢 𑱼𑲮𑲁𑲡𑲂𑲊𑲱𑲉𑲰 𑲂, 𑱹𑲰𑱺𑱲𑲰𑱽𑱹𑲢 𑲍𑱵𑲔𑲎𑲴𑱱
𑲏𑱽𑲅𑲄𑲤𑲰𑱽𑲱 𑱲𑲁𑲡𑲰𑱽𑲱, 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱
༨. ‘‘𑲏𑲰𑲊𑱺𑲱𑲰 𑲅𑲱𑲊𑱺𑲱𑲰 𑱾𑲰𑱾𑲰, 𑲁𑲄𑲤𑱾𑲰𑱽𑲰 𑲂 𑲍𑲵𑲉𑲁𑲴𑱱
𑲏𑱾𑲞𑲁𑲰𑱼𑲴 𑲂 𑱼𑲮𑲁𑲡𑲳𑲍𑲲, 𑲏𑲳𑱺𑲵 𑲁𑱵𑲔𑲋𑲁𑲲𑱺𑲚𑲁𑲵𑱱𑱱

𑲏𑲳𑱺𑲳𑱽 𑲍𑲂𑲢𑲳𑱽 𑲍𑲲𑲅𑱺𑲛𑲱 𑲎𑲴𑱺𑲲𑱱𑱱

༡༨. 𑲉𑲰𑱽𑲱𑲰𑱼𑲮 𑲀𑲮𑲲𑲰𑱺𑲰𑱽𑲱 𑲍𑲁𑲰𑱴𑱺𑲰𑱽𑲱, 𑲀𑲮𑲲𑲁𑲡𑲰𑱽𑲱 𑲅𑲰 𑲉𑲰𑱽𑲱 𑲅 𑲏𑱽𑲚𑲋𑲱𑱲𑲓𑲳𑱱

༧. 𑱺𑲱𑲊𑲴𑱲𑲲𑱺𑲚𑲍𑲲𑱺𑲚𑲵
༡. 𑱺𑲱𑲊𑲴𑱲𑲲𑱺𑲚𑲳𑲍𑲲 𑱺𑲱𑱺𑲛𑱽𑲚𑲱, 𑲍𑱽𑲜𑲮𑲱𑲍𑲱𑱵𑲔𑲮𑲰𑱺𑱲𑲳𑲍𑲲 𑲂𑱱

༣. 𑲏𑲳𑲅𑲵 𑱼𑱼𑱽𑲚𑲱 𑱹𑲰𑱺𑲱𑲰𑱽𑲵, 𑲉𑲳 𑲎𑲴𑱽𑲚𑲱 𑲏𑱽𑲲𑱲𑲁𑲞𑱲𑲰𑱱
𑲍𑲲𑲂𑲱𑲵 𑱾𑱽𑲱𑲰𑱺𑲵 𑱲𑲰𑲋𑲳𑱽, 𑱲𑱾𑲞𑲱𑲉𑲵 𑱾𑲰𑱽𑲀𑲮𑲴𑲄𑱽𑲵𑱱
𑲏𑲱𑱼𑲵 𑲅𑲴 𑱹𑲰𑱺𑲱𑲰𑱽𑲵 𑲎𑲴𑱺𑲲, 𑲍𑲲𑱳𑲱𑱺𑲰 𑲎𑲴𑱽𑲚𑲲 𑱹𑲰𑱺𑲉𑲴𑱱𑱱
༤. 𑱺𑲳 𑲂 𑱺𑱺𑲛 𑲍𑲁𑲰𑱴𑱽𑲚𑲥𑲰, 𑱹𑲰𑱺𑲱𑱾𑲳𑱺𑲰 𑲍𑲁𑲰𑱴𑱺𑲰𑱱

༥. 𑲂𑲱𑲊𑲵 𑲄𑲱𑲰𑲅𑱽𑲚𑲲 𑱽𑲴 𑱹𑲰𑱺𑲱𑲰, 𑲉𑲳𑲍𑲵 𑲎𑲳𑱺𑲲 𑲋𑲀𑲮𑲰𑲁𑲍𑲳𑱱
𑲏𑲁𑲮𑲰𑱲𑱹𑲢 𑱲𑱺𑲰 𑱾𑲲𑲰𑲄𑲰, 𑱼𑲰𑲉𑱲𑲰 𑲂 𑲏𑱽𑲱𑱾𑲟𑲋𑲰𑱱𑱱
༦. 𑱽 𑲎𑲱 𑱺𑱺𑲛 𑱲𑲍𑲱 𑲏𑱺𑲛𑲱, 𑱴𑲴𑲊𑱲𑲓𑲳𑱺𑲛 𑱽 𑲅𑲱𑲄𑲤𑱺𑲱𑱱

༡༤. 𑲁𑲱𑱺𑲚𑲍𑲁𑲞𑱼𑲁𑲰𑱴𑲁𑲡, 𑲉𑲴𑱽𑲱𑲍𑲴𑲅 𑱾𑲉𑲲𑱹𑲤𑱺𑲴𑱱

༡༦. 𑲏𑲳𑲅𑲵 𑲁𑲎𑱺𑲛𑲱𑱲𑲰 𑲏𑲳𑲍𑲰, 𑲉𑱼𑲱𑱼𑲵 𑱾𑲲𑱹𑲙𑲍𑲁𑲞𑱼𑲰𑱱

༩. 𑲁𑲳𑱺𑲚𑲍𑲲𑱺𑲚𑲵
